  92. 92def accepted_json_field(raw: bytes, field: str = "result") -> AcceptedJSON:
  93. 93 # 先校验整个文档并拒绝重复键，再利用解码偏移复制原始值；
  94. 94 # 不要经过 float 转换而改变 Go 已接受的数值。
  95. 95 """从已严格校验的 Go 回调中提取 result 原始切片，避免大整数或小数经过 Python 重编码失真。"""
  96. 96 parsed = strict_json(raw)
  97. 97 if not isinstance(parsed, dict) or field not in parsed:
  98. 98 raise ProtocolError()
  99. 99 source = raw.decode("utf-8")
  100. 100 decoder = json.JSONDecoder(parse_int=lambda _: None, parse_float=lambda _: None)
  101. 101 index = 0
  102. 102
  103. 103 def whitespace(position: int) -> int:
  104. 104 while position < len(source) and source[position] in " \t\r\n":
  105. 105 position += 1
  106. 106 return position
  107. 107
  108. 108 index = whitespace(index) + 1
  109. 109 while True:
  110. 110 index = whitespace(index)
  111. 111 key, index = decoder.raw_decode(source, index)
  112. 112 index = whitespace(index)
  113. 113 if source[index] != ":":
  114. 114 raise ProtocolError()
  115. 115 start = whitespace(index + 1)
  116. 116 _, end = decoder.raw_decode(source, start)
  117. 117 if key == field:
  118. 118 return AcceptedJSON(source[start:end].encode("utf-8"))
  119. 119 index = whitespace(end)
  120. 120 if source[index] != ",":
  121. 121 raise ProtocolError()
  122. 122 index += 1

  162. 162async def read_http(
  163. 163 reader: asyncio.StreamReader,
  164. 164 *,
  165. 165 response: bool = False,
  166. 166 max_body_bytes: int = MAX_BODY_BYTES,

  168. 168) -> tuple[str, dict, bytes]:
  169. 169 """读取一条固定 HTTP/1.1 请求或响应，返回首行、头和原始正文。
  170. 170
  171. 171 头部和正文分别限长；拒绝重复头与 transfer-encoding，避免私有协议出现歧义或无界读取。
  172. 172 """
  173. 173 try:
  174. 174 header = await reader.readuntil(b"\r\n\r\n")
  175. 175 except (asyncio.IncompleteReadError, asyncio.LimitOverrunError):
  176. 176 raise ProtocolError() from None
  177. 177 if len(header) > MAX_HEADER_BYTES:
  178. 178 raise ProtocolError()
  179. 179 try:
  180. 180 lines = header[:-4].decode("ascii").split("\r\n")
  181. 181 except UnicodeError:
  182. 182 raise ProtocolError() from None
  183. 183 first = lines[0]
  184. 184 headers = {}
  185. 185 for line in lines[1:]:
  186. 186 key, separator, value = line.partition(":")
  187. 187 key = key.lower()
  188. 188 if not separator or key in headers or not key or any(c.isspace() for c in key):
  189. 189 raise ProtocolError()
  190. 190 headers[key] = value.strip()
  191. 191 length_text = headers.get(
  192. 192 "content-length", "0" if first == "GET /health HTTP/1.1" else ""
  193. 193 )
  194. 194 if (
  195. 195 not length_text.isascii()
  196. 196 or not length_text.isdigit()
  197. 197 or len(length_text) > 7
  198. 198 or "transfer-encoding" in headers
  199. 199 ):
  200. 200 raise ProtocolError()
  201. 201 size = int(length_text)
  202. 202 if size > max_body_bytes:
  203. 203 raise ProtocolError("AGENT_RUNTIME_BODY_TOO_LARGE")
  204. 204 if response:
  205. 205 if first != "HTTP/1.1 200 OK":
  206. 206 raise ProtocolError("AGENT_RUNTIME_GATEWAY_REJECTED")
  207. 207 elif first not in allowed_first:
  208. 208 raise ProtocolError()
  209. 209 try:
  210. 210 body = await reader.readexactly(size)
  211. 211 except asyncio.IncompleteReadError:
  212. 212 raise ProtocolError() from None
  213. 213 return first, headers, body

  259. 259 self.path, limit=MAX_HEADER_BYTES

  281. 281 return result
