  203. 203def external_monitoring_metrics(path):
  204. 204 """可选宿主计数器失败时，不应破坏主指标端点。"""
  205. 205 try:
  206. 206 descriptor = os.open(path, os.O_RDONLY | getattr(os, "O_NONBLOCK", 0) | getattr(os, "O_NOFOLLOW", 0))
  207. 207 with os.fdopen(descriptor, "rb") as stream:
  208. 208 info = os.fstat(stream.fileno())
  209. 209 if not stat.S_ISREG(info.st_mode) or info.st_size > 4096:
  210. 210 return b""
  211. 211 data = stream.read(4097)
  212. 212 if len(data) > 4096:
  213. 213 return b""
  214. 214 lines = []
  215. 215 for line in data.decode("ascii").splitlines():
  216. 216 match = re.fullmatch(r"drop_(backup_last_success|backup_restore_duration_seconds|public_certificate_expiry_seconds) ((?:0|[1-9][0-9]*)(?:\.[0-9]+)?)", line)
  217. 217 if match and math.isfinite(float(match.group(2))):
  218. 218 lines.append(line)
  219. 219 return ("\n".join(lines) + ("\n" if lines else "")).encode("ascii")
  220. 220 except FileNotFoundError:
  221. 221 return b""
  222. 222 except (OSError, UnicodeError, ValueError) as error:
  223. 223 log("monitoring_status_unavailable", level="warning", stage="metrics",
  224. 224 error_code="MONITORING_STATUS_UNAVAILABLE", error_type=type(error).__name__)
  225. 225 return b""

  315. 315def main(argv=None):
  316. 316 server = None
  317. 317 server_started = False
  318. 318 try:
  319. 319 config = Config.load(argv)
  320. 320 config.log_startup("drop-maintenance")
  321. 321 endpoint = urlsplit(config["S3_ENDPOINT"])
  322. 322 client = Minio(
  323. 323 endpoint.netloc,
  324. 324 access_key=config["S3_ACCESS_KEY"],
  325. 325 secret_key=config["S3_SECRET_KEY"],
  326. 326 secure=endpoint.scheme == "https",
  327. 327 )
  328. 328 bucket = config["S3_BUCKET"]
  329. 329 stop = threading.Event()
  330. 330 for sig in (signal.SIGTERM, signal.SIGINT):
  331. 331 signal.signal(sig, lambda *_: stop.set())
  332. 332 configure_lifecycle(client, bucket)
  333. 333 server = SafeHTTPServer(
  334. 334 (config["MAINTENANCE_METRICS_HOST"], config["MAINTENANCE_METRICS_PORT"]),
  335. 335 Metrics,
  336. 336 )
  337. 337 server.config = config
  338. 338 threading.Thread(target=server.serve_forever, daemon=True).start()
  339. 339 server_started = True
  340. 340 while not stop.is_set():
  341. 341 started = time.monotonic()
  342. 342 try:
  343. 343 with psycopg.connect(
  344. 344 config["DATABASE_URL"], autocommit=True, row_factory=dict_row
  345. 345 ) as conn:
  346. 346 locked = conn.execute(
  347. 347 "SELECT pg_try_advisory_lock(hashtextextended('drop-maintenance-v1',0)) AS yes"
  348. 348 ).fetchone()["yes"]
  349. 349 if locked:
  350. 350 outcome = reconcile(
  351. 351 SQLRepository(conn), S3Storage(client, bucket)
  352. 352 )
  353. 353 for key, value in outcome.items():
  354. 354 state[key] += value
  355. 355 state["last_success"] = time.time()
  356. 356 log(
  357. 357 "storage_reconciled",
  358. 358 stage="reconcile",
  359. 359 duration_ms=int((time.monotonic() - started) * 1000),
  360. 360 **outcome,
  361. 361 )
  362. 362 except Exception as exc:
  363. 363 state["errors"] += 1
  364. 364 log(
  365. 365 "storage_reconcile_failed",
  366. 366 level="error",
  367. 367 stage="reconcile",
  368. 368 error_code="STORAGE_OR_DATABASE_FAILURE",
  369. 369 error_type=type(exc).__name__,
  370. 370 duration_ms=int((time.monotonic() - started) * 1000),
  371. 371 )
  372. 372 stop.wait(config["MAINTENANCE_INTERVAL_SECONDS"])
  373. 373 log("maintenance_stopped", stage="shutdown")
  374. 374 return 0
  375. 375 except ConfigurationError:
  376. 376 log(
  377. 377 "configuration_invalid",
  378. 378 level="error",
  379. 379 stage="startup",
  380. 380 error_code="CONFIGURATION_INVALID",
  381. 381 )
  382. 382 return 1
  383. 383 except Exception:
  384. 384 log(
  385. 385 "maintenance_startup_failed",
  386. 386 level="error",
  387. 387 stage="startup",
  388. 388 error_code="STARTUP_FAILED",
  389. 389 )
  390. 390 return 1
  391. 391 finally:
  392. 392 if server is not None:
  393. 393 if server_started:
  394. 394 server.shutdown()
  395. 395 server.server_close()
  396. 396
  397. 397
  398. 398if __name__ == "__main__":
  399. 399 raise SystemExit(main())
