  113. 113def validate_regression_tests(tests, manifest):
  114. 114 """模型只能新增独立 unittest 文件；不得替换原测试、跳过检查或通过反射/外部 I/O 观察宿主。

  118. 118 """

  124. 124 for test in tests:
  125. 125 require(isinstance(test, dict) and set(test) == {'path', 'content'}, 'REGRESSION_TEST_FIELDS')

  139. 139 if isinstance(node, ast.ImportFrom):

  150. 150 require(not any(isinstance(target, ast.Attribute) for target in targets), 'REGRESSION_TEST_MUTATION')
  151. 151 assertions += isinstance(node, ast.Assert)
  152. 152 require(assertions > 0 and any(isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name.startswith('test_') for node in ast.walk(tree)), 'REGRESSION_TEST_ASSERTION_REQUIRED')
  153. 153 result[name] = content
  154. 154 return result

  158. 158 """输出可审阅的标准 unified diff；交付物不含发布动作或原仓库写入能力。"""
  159. 159 def unified(before, after, old_name, new_name):
  160. 160 # difflib 不自动标识缺少末尾换行；补标准标记，保证交付物可供 git apply/PR 审阅。
  161. 161 return ''.join(line if line.endswith('\n') else line + '\n\\ No newline at end of file\n'
  162. 162 for line in difflib.unified_diff(before, after, fromfile=old_name, tofile=new_name))
  163. 163 code = ''.join(unified((source / name).read_text(encoding='utf-8').splitlines(keepends=True), text.splitlines(keepends=True), 'a/' + name, 'b/' + name) for name, text in sorted(changed.items()))
  164. 164 test_diff = ''.join(unified([], text.splitlines(keepends=True), '/dev/null', 'b/' + name) for name, text in sorted(tests.items()))
  165. 165 content = code + test_diff
  166. 166 require(len(content.encode()) <= 24576, 'PATCH_BUNDLE_LIMIT')
  167. 167 return {'schema_version': 1, 'format': 'unified-diff', 'source_revision': manifest['source_revision'],
  168. 168 'content': content, 'sha256': hashlib.sha256(content.encode()).hexdigest(),
  169. 169 'code_files': sorted(changed), 'test_files': sorted(tests),
  170. 170 'applied_to': 'isolated_snapshot', 'production_deployed': False}

  220. 220def run_sandbox(argv, container, key, image, log, canceled, timeout=240):
  221. 221 """先 create、核对归属，再 start，确保取消操作持有不可变容器 ID。
  222. 222
  223. 223 不进行全局进程清理，也不只凭容器名删除；Docker CLI 是本请求的子进程，
  224. 224 取消时只移除已核对 job 标签与镜像的本次容器，避免影响其他作业。
  225. 225 """
  226. 226 require(not canceled(), 'EXECUTION_CANCELED')
  227. 227 old = owned_container(container, key, image)
  228. 228 if old:
  229. 229 subprocess.run(['docker', 'rm', '-f', old], check=True, capture_output=True, timeout=15)
  230. 230 require(not canceled(), 'EXECUTION_CANCELED')
  231. 231 created = subprocess.run(argv, capture_output=True, timeout=30)
  232. 232 require(created.returncode == 0, 'SANDBOX_CREATE_FAILED')
  233. 233 container_id = created.stdout.decode('ascii').strip()
  234. 234 require(re.fullmatch(r'[0-9a-f]{64}', container_id), 'SANDBOX_CREATE_FAILED')
  235. 235 process = None
  236. 236 try:
  237. 237 require(owned_container(container_id, key, image) == container_id, 'CONTAINER_OWNERSHIP_MISMATCH')
  238. 238 require(not canceled(), 'EXECUTION_CANCELED')
  239. 239 process = subprocess.Popen(['docker', 'start', '--attach', container_id], stdout=log, stderr=subprocess.STDOUT)
  240. 240 deadline = time.monotonic() + timeout
  241. 241 while process.poll() is None:
  242. 242 require(not canceled(), 'EXECUTION_CANCELED')
  243. 243 if time.monotonic() >= deadline:
  244. 244 return -1
  245. 245 time.sleep(0.05)
  246. 246 require(not canceled(), 'EXECUTION_CANCELED')
  247. 247 return process.returncode
  248. 248 finally:
  249. 249 try:
  250. 250 if owned_container(container_id, key, image) == container_id:
  251. 251 subprocess.run(['docker', 'rm', '-f', container_id], check=True, capture_output=True, timeout=15)
  252. 252 finally:
  253. 253 if process is not None and process.poll() is None:
  254. 254 process.terminate()
  255. 255 try:
  256. 256 process.wait(timeout=2)
  257. 257 except subprocess.TimeoutExpired:
  258. 258 process.kill()
  259. 259 process.wait(timeout=2)

  268. 268def validate_patch(base, edits, manifest):
  269. 269 """校验精确上下文补丁并返回修改后的文件内容，不直接写原业务源码。
  270. 270
  271. 271 只允许登记的 editable_files，先核对原 SHA 再按序替换；AST 限制导入、反射与缓存规模，
  272. 272 Saber 额外冻结 SQL/租户相关节点。语法通过不等于行为正确，仍需独立回归与性能对照。
  273. 273 """
  274. 274 require(isinstance(edits, list) and 1 <= len(edits) <= 6, 'PATCH_EDIT_LIMIT')
  275. 275 require(len(json.dumps(edits).encode()) <= 24576, 'PATCH_BYTE_LIMIT')
  276. 276 changed = {}
  277. 277 ranges = {}
  278. 278 for edit in edits:
  279. 279 require(isinstance(edit, dict) and set(edit) == {'path', 'sha256', 'old', 'new'}, 'PATCH_FIELDS_REJECTED')
  280. 280 name = edit['path']
  281. 281 require(name in manifest['editable_files'] and name in manifest['files'], 'PATCH_PATH_NOT_ENROLLED')
  282. 282 target = (base / name)
  283. 283 require(target.is_file() and not target.is_symlink() and target.resolve().is_relative_to(base.resolve()), 'PATCH_PATH_ESCAPE')
  284. 284 require(file_hash(target) == edit['sha256'] == manifest['files'][name], 'PATCH_SOURCE_CHANGED')
  285. 285 old, new = edit['old'], edit['new']
  286. 286 require(isinstance(old, str) and isinstance(new, str) and 0 < len(old) <= 16384 and len(new) <= 16384, 'PATCH_CONTENT_LIMIT')
  287. 287 original = target.read_text(encoding='utf-8')
  288. 288 require(original.count(old) == 1, 'PATCH_CONTEXT_NOT_UNIQUE')
  289. 289 start = original.index(old)
  290. 290 end = start + len(old)
  291. 291 require(all(end <= a or start >= b for a, b in ranges.get(name, [])), 'PATCH_CONTEXT_OVERLAP')
  292. 292 ranges.setdefault(name, []).append((start, end))
  293. 293 # 同一文件可有多次精确上下文编辑；每次都绑定原始登记 SHA，
  294. 294 # 再按声明顺序应用，避免补丁目标漂移。
  295. 295 text = changed.get(name, original)
  296. 296 require(text.count(old) == 1 and old != new, 'PATCH_CONTEXT_NOT_UNIQUE')
  297. 297 candidate = text.replace(old, new, 1)
  298. 298 require(len(candidate.encode()) <= 131072, 'PATCH_RESULT_LIMIT')
  299. 299 try:
  300. 300 tree = ast.parse(candidate)
  301. 301 except SyntaxError:
  302. 302 raise Rejected('PATCH_SYNTAX_INVALID') from None
  303. 303 # 此 preset 只登记纯计算模块；测试、授权、harness 以及网络/OS 导入
  304. 304 # 不能通过模型工具修改。
  305. 305 forbidden = {'eval', 'exec', 'compile', '__import__', 'open', 'getattr', 'setattr', 'globals', 'locals', 'breakpoint', 'input'}
  306. 306 preset = manifest.get('test_preset', 'crm-brand-palette-v1')
  307. 307 require(preset in PRESET_IMPORTS or preset == GENERIC_PRESET, 'PROJECT_PRESET_INVALID')
  308. 308 imports = set(manifest['allowed_imports']) if preset == GENERIC_PRESET else PRESET_IMPORTS[preset]
  309. 309 if preset == 'saber-fts5-v1':
  310. 310 require(name == 'internal/rag/fts5_index.py', 'PATCH_PATH_NOT_ENROLLED')
  311. 311 require(frozen_saber_nodes(ast.parse(original)) == frozen_saber_nodes(tree), 'PATCH_FROZEN_SQL_CONTRACT')
  312. 312 for node in ast.walk(tree):
  313. 313 if isinstance(node, ast.Import):
  314. 314 require(all(alias.name in imports for alias in node.names), 'PATCH_IMPORT_REJECTED')
  315. 315 if isinstance(node, ast.ImportFrom):
  316. 316 require(node.level == 0 and node.module in imports, 'PATCH_IMPORT_REJECTED')
  317. 317 if isinstance(node, ast.Name):
  318. 318 require(node.id not in forbidden and not node.id.startswith('__'), 'PATCH_REFLECTION_REJECTED')
  319. 319 if isinstance(node, ast.Attribute):
  320. 320 require(not node.attr.startswith('__'), 'PATCH_REFLECTION_REJECTED')
  321. 321 if isinstance(node, ast.Call) and isinstance(node.func, (ast.Name, ast.Attribute)):
  322. 322 function = node.func.id if isinstance(node.func, ast.Name) else node.func.attr
  323. 323 if function in {'lru_cache', 'cache'}:
  324. 324 require(function == 'lru_cache', 'PATCH_UNBOUNDED_CACHE')
  325. 325 size = next((arg.value for arg in node.keywords if arg.arg == 'maxsize'), None)
  326. 326 require(isinstance(size, ast.Constant) and type(size.value) is int and 1 <= size.value <= 4096, 'PATCH_UNBOUNDED_CACHE')
  327. 327 changed[name] = candidate
  328. 328 return changed

  337. 337 def project(self, name, revision=None):
  338. 338 """读取 root 所有且不可被组/其他用户修改的登记 manifest，校验源码修订与固定镜像摘要。"""
  339. 339 require(isinstance(name, str) and re.fullmatch(r'[a-z0-9][a-z0-9_-]{0,63}', name), 'PROJECT_ID_INVALID')
  340. 340 root = self.root / 'registry' / name
  341. 341 require(root.is_dir() and not root.is_symlink() and root.resolve().is_relative_to(self.root), 'PROJECT_NOT_ENROLLED')
  342. 342 path = root / 'manifest.json'
  343. 343 require(not path.is_symlink() and path.stat().st_uid == 0 and not path.stat().st_mode & 0o022 and path.stat().st_size <= 131072, 'PROJECT_MANIFEST_UNSAFE')
  344. 344 manifest = json.loads(path.read_text())
  345. 345 require(manifest.get('id') == name and manifest.get('test_preset') in {*PRESET_IMPORTS, GENERIC_PRESET}, 'PROJECT_PRESET_INVALID')
  346. 346 if manifest['test_preset'] == GENERIC_PRESET:
  347. 347 validate_generic_manifest(manifest)
  348. 348 digest = hashlib.sha256(json.dumps(manifest['files'], sort_keys=True).encode()).hexdigest()
  349. 349 require(digest == manifest['source_revision'] and (revision is None or revision == digest), 'PROJECT_REVISION_MISMATCH')
  350. 350 require(re.fullmatch(r'sha256:[0-9a-f]{64}', manifest['runner_image_id']), 'PROJECT_IMAGE_NOT_PINNED')
  351. 351 return root, manifest

  353. 353 def invoke(self, request, canceled=lambda: False):
  354. 354 """分派闭合动作集合并校验字段；执行型动作必须绑定 source_revision，且受单并发槽限制。"""
  355. 355 require(not canceled(), 'EXECUTION_CANCELED')
  356. 356 require(isinstance(request, dict) and set(request) <= {'action', 'project_id', 'source_revision', 'execution_id', 'arguments'}, 'REQUEST_FIELDS_REJECTED')
  357. 357 action = request.get('action')
  358. 358 args = request.get('arguments') or {}
  359. 359 require(isinstance(args, dict), 'ARGUMENTS_INVALID')
  360. 360 if action == 'projects':
  361. 361 require(not args, 'ARGUMENTS_INVALID')
  362. 362 return {'projects': [self.describe(path.name) for path in sorted((self.root / 'registry').iterdir()) if path.is_dir() and not path.is_symlink()]}
  363. 363 revision = request.get('source_revision') or None
  364. 364 if action != 'describe':
  365. 365 require(isinstance(revision, str) and re.fullmatch(r'[0-9a-f]{64}', revision), 'PROJECT_REVISION_REQUIRED')
  366. 366 root, manifest = self.project(request.get('project_id'), revision)
  367. 367 source = root / 'source'
  368. 368 if action == 'describe':
  369. 369 require(not args, 'ARGUMENTS_INVALID')
  370. 370 return self.describe(manifest['id'])
  371. 371 if action == 'read_source':
  372. 372 require(set(args) <= {'path', 'start_line', 'end_line'}, 'SOURCE_ARGUMENTS_INVALID')
  373. 373 name = args.get('path')
  374. 374 require(name in manifest['readable_files'], 'SOURCE_PATH_NOT_ENROLLED')
  375. 375 path = source / name
  376. 376 require(path.is_file() and not path.is_symlink() and path.resolve().is_relative_to(source.resolve()) and file_hash(path) == manifest['files'][name], 'SOURCE_REVISION_MISMATCH')
  377. 377 start, end = args.get('start_line', 1), args.get('end_line', 160)
  378. 378 require(type(start) is int and type(end) is int and 1 <= start <= end <= start + 159 and end <= 10000, 'SOURCE_LINE_LIMIT')
  379. 379 lines = path.read_text(encoding='utf-8').splitlines(keepends=True)
  380. 380 return {'path': name, 'start_line': start, 'end_line': min(end, len(lines)), 'sha256': manifest['files'][name],
  381. 381 'source_revision': manifest['source_revision'], 'content': ''.join(lines[start - 1:end]), 'total_lines': len(lines)}
  382. 382 if action in {'profile_business_flow', 'test_patch'}:
  383. 383 generic = manifest['test_preset'] == GENERIC_PRESET
  384. 384 require(set(args) <= ({'edits', 'regression_tests'} if action == 'test_patch' and generic else {'edits'} if action == 'test_patch' else set()), 'EXECUTION_ARGUMENTS_INVALID')
  385. 385 changed = validate_patch(source, args.get('edits'), manifest) if action == 'test_patch' else {}
  386. 386 tests = validate_regression_tests(args.get('regression_tests'), manifest) if action == 'test_patch' and generic else {}
  387. 387 if generic and changed:
  388. 388 require(len(json.dumps(args.get('edits')).encode()) <= 12288, 'PATCH_BYTE_LIMIT')
  389. 389 require(self.slots.acquire(blocking=False), 'EXECUTOR_BUSY')
  390. 390 try:
  391. 391 return self.execute(source, manifest, changed, args.get('edits', []), request.get('execution_id', ''), canceled, tests)
  392. 392 finally:
  393. 393 self.slots.release()
  394. 394 raise Rejected('ACTION_NOT_ALLOWED')

  396. 396 def describe(self, name):
  397. 397 """返回已登记的项目能力；存在工作负载绑定时，再以实际容器镜像、标签和 /proc 身份核对目标。"""
  398. 398 _, manifest = self.project(name)
  399. 399 description = {key: manifest[key] for key in ('id', 'label', 'source_revision', 'editable_files', 'readable_files', 'test_preset', 'workload')}
  400. 400 if manifest['test_preset'] == GENERIC_PRESET:
  401. 401 description.update({key: manifest[key] for key in ('schema_version', 'language', 'output_contract', 'symbol_mapping', 'frozen_files', 'allowed_imports', 'regression_test_dir', 'performance_gate')})
  402. 402 description['regression_tests_required'] = True
  403. 403 description['supported_languages'] = ['python']
  404. 404 # 运行容器与源码的绑定由运维登记，不接受模型提供的绑定关系。
  405. 405 container = manifest.get('workload_container')
  406. 406 if container:
  407. 407 require(re.fullmatch(r'[a-z0-9][a-z0-9_-]{0,63}', container), 'RUNTIME_BINDING_INVALID')
  408. 408 inspected = subprocess.run(['docker', 'inspect', container], capture_output=True, timeout=5)
  409. 409 if inspected.returncode == 0:
  410. 410 info = json.loads(inspected.stdout)[0]
  411. 411 labels = info['Config'].get('Labels') or {}
  412. 412 if (info['State']['Running'] and info['Image'] == manifest.get('workload_image_id', manifest['runner_image_id'])
  413. 413 and labels.get('drop.code-project') == name
  414. 414 and labels.get('drop.source-revision') == manifest['source_revision']):
  415. 415 host_pid = int(info['State']['Pid'])
  416. 416 proc = Path('/proc') / str(host_pid)
  417. 417 status = dict(line.split(':', 1) for line in (proc / 'status').read_text().splitlines() if ':' in line)
  418. 418 start_ticks = int((proc / 'stat').read_text().rsplit(')', 1)[1].split()[19])
  419. 419 description['runtime'] = {'agent_id': manifest['agent_id'], 'pid': int(status['NSpid'].split()[-1]),
  420. 420 'uid': int(status['Uid'].split()[0]), 'start_time_ticks': start_ticks,
  421. 421 'source_revision': manifest['source_revision']}
  422. 422 return description

  424. 424 def execute(self, source, manifest, changed, edits, execution_id='', canceled=lambda: False, tests=None):
  425. 425 # 执行前验证整个已登记快照，不能只验证模型点名的单个文件。
  426. 426 """在作业私有目录复制基线与候选，并在固定、无网络、无凭据容器中测量。
  427. 427
  428. 428 先验证完整登记快照，再以执行键和文件锁隔离并发；仅完整收据可命中缓存。
  429. 429 上次中断产生的 result.json 必须清除，新的成功/失败结果通过临时文件替换发布，避免半成品恢复误判。
  430. 430 """
  431. 431 tests = tests or {}
  432. 432 generic = manifest['test_preset'] == GENERIC_PRESET
  433. 433 for name, digest in manifest['files'].items():
  434. 434 require(not canceled(), 'EXECUTION_CANCELED')
  435. 435 path = source / name
  436. 436 require(path.is_file() and not path.is_symlink() and path.resolve().is_relative_to(source.resolve())
  437. 437 and (not generic or path.stat().st_size <= 131072)
  438. 438 and file_hash(path) == digest, 'PROJECT_SOURCE_CHANGED')
  439. 439 kind = 'test' if changed else 'profile'
  440. 440 # 旧 preset 的作业键及收据保持不变；新协议把独立测试与完整 manifest 一起绑定。
  441. 441 key_edits = {'code_edits': edits, 'regression_tests': tests, 'manifest_sha256': hashlib.sha256(json.dumps(manifest, sort_keys=True).encode()).hexdigest()} if generic else edits
  442. 442 key = execution_key(manifest, kind, key_edits, execution_id)
  443. 443 job = self.root / 'jobs' / key
  444. 444 job.mkdir(mode=0o755, parents=True, exist_ok=True)
  445. 445 require(not job.is_symlink() and job.resolve().parent == self.root / 'jobs', 'JOB_PATH_ESCAPE')
  446. 446 # systemd 使用严格 UMask；挂载根目录仍需允许沙箱 UID10001 穿越，
  447. 447 # 而宿主上层目录继续保持仅 root 可访问的 0700。
  448. 448 job.chmod(0o755)
  449. 449 import fcntl
  450. 450 with (job / 'lock').open('a') as lock:
  451. 451 require(not canceled(), 'EXECUTION_CANCELED')
  452. 452 try:
  453. 453 fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
  454. 454 except BlockingIOError:
  455. 455 raise Rejected('EXECUTOR_BUSY') from None
  456. 456 receipt = job / 'receipt.json'
  457. 457 if receipt.exists():
  458. 458 return {**json.loads(receipt.read_text()), 'cache_hit': True}
  459. 459 for side in ('baseline', 'candidate') if changed else ('baseline',):
  460. 460 # 中断恢复先清除本作业的私有副本，不能保留上次失败留下的文件或测试。
  461. 461 if generic and (job / side).exists():
  462. 462 shutil.rmtree(job / side)
  463. 463 if generic:
  464. 464 destination = job / side
  465. 465 destination.mkdir(mode=0o755, exist_ok=True)
  466. 466 for name in manifest['files']:
  467. 467 target = destination / name
  468. 468 target.parent.mkdir(mode=0o755, parents=True, exist_ok=True)
  469. 469 shutil.copyfile(source / name, target)
  470. 470 target.chmod(0o644)
  471. 471 for directory in [destination, *(path for path in destination.rglob('*') if path.is_dir())]:
  472. 472 directory.chmod(0o755)
  473. 473 else:
  474. 474 shutil.copytree(source, job / side, dirs_exist_ok=True)
  475. 475 # JSON 路由回归需要静态挂载目录；此集成不会因此发布 CRM 的公共 UI。
  476. 476 if manifest['test_preset'] == 'crm-brand-palette-v1':
  477. 477 (job / side / 'crm/static').mkdir(exist_ok=True)
  478. 478 for name, text in changed.items():
  479. 479 (job / 'candidate' / name).write_text(text, encoding='utf-8')
  480. 480 if generic:
  481. 481 (job / 'manifest.json').write_text(json.dumps(manifest), encoding='utf-8')
  482. 482 (job / 'manifest.json').chmod(0o644)
  483. 483 regression = job / 'regression-tests'
  484. 484 if regression.exists():
  485. 485 shutil.rmtree(regression)
  486. 486 regression.mkdir(mode=0o755)
  487. 487 regression.chmod(0o755)
  488. 488 for name, content in tests.items():
  489. 489 # 只挂载独立测试目录；两边读取相同文件，不能随 baseline/candidate 替换断言。
  490. 490 (regression / name.rsplit('/', 1)[-1]).write_text(content, encoding='utf-8')
  491. 491 (regression / name.rsplit('/', 1)[-1]).chmod(0o644)
  492. 492 output = job / 'output'
  493. 493 output.mkdir(mode=0o700, exist_ok=True)
  494. 494 require(not output.is_symlink() and output.resolve().parent == job.resolve(), 'JOB_PATH_ESCAPE')
  495. 495 os.chown(output, 10001, 10001)
  496. 496 # 上次中断的尝试没有有效收据；恢复时必须删除半成品结果，
  497. 497 # 不能把它当成本轮的新测量。
  498. 498 previous_result = output / 'result.json'
  499. 499 if previous_result.exists() or previous_result.is_symlink():
  500. 500 previous_result.unlink()
  501. 501 (job / 'job.json').write_text(json.dumps({'seed': secrets.randbits(32)}))
  502. 502 (job / 'job.json').chmod(0o644)
  503. 503 container = 'drop-code-' + key[:24]
  504. 504 argv = ['docker', 'create', '--name', container, '--label', 'drop.code-job=' + key,
  505. 505 '--pull', 'never', '--network', 'none', '--user', '10001:10001', '--cap-drop', 'ALL',
  506. 506 '--security-opt', 'no-new-privileges:true', '--read-only', '--memory', '512m', '--memory-swap', '512m',
  507. 507 '--cpus', '1', '--pids-limit', '64', '--ulimit', 'fsize=1048576:1048576',
  508. 508 '--tmpfs', '/tmp:rw,nosuid,nodev,noexec,size=64m,mode=1777',
  509. 509 '--mount', f'type=bind,src={job},dst=/workspace,readonly',
  510. 510 '--mount', f'type=bind,src={output},dst=/output',
  511. 511 '--entrypoint', '/opt/drop-tools/venv/bin/python', manifest['runner_image_id'], '/harness/run_job.py', '--action', kind]
  512. 512 with (job / 'runner.private.log').open('wb') as log:
  513. 513 code = run_sandbox(argv, container, key, manifest['runner_image_id'], log, canceled)
  514. 514 require(not canceled(), 'EXECUTION_CANCELED')
  515. 515 result = output / 'result.json'
  516. 516 require(result.is_file() and not result.is_symlink() and result.stat().st_size <= 65536, 'SANDBOX_RESULT_MISSING')
  517. 517 data = json.loads(result.read_text())
  518. 518 require(isinstance(data, dict), 'SANDBOX_RESULT_INVALID')
  519. 519 if code != 0:
  520. 520 data = {'success': False, 'error_code': data.get('error_code', 'SANDBOX_FAILED')}
  521. 521 data.update({'job_key': key, 'source_revision': manifest['source_revision'], 'project_id': manifest['id'],
  522. 522 'executor_contract': 3, 'cache_hit': False,
  523. 523 'runner_image_id': manifest['runner_image_id'], 'network': 'none', 'uid': 10001, 'credentials_mounted': False})
  524. 524 if changed:
  525. 525 data['patch'] = edits
  526. 526 data['candidate_files'] = {name: file_hash(job / 'candidate' / name) for name in changed}
  527. 527 if generic:
  528. 528 data['executor_contract'] = 4
  529. 529 data['output_contract'] = GENERIC_CONTRACT
  530. 530 data['language'] = 'python'
  531. 531 if changed:
  532. 532 data['regression_tests'] = [{'path': name, 'sha256': hashlib.sha256(content.encode()).hexdigest()} for name, content in sorted(tests.items())]
  533. 533 data['patch_bundle'] = patch_bundle(source, changed, tests, manifest)
  534. 534 require(len(json.dumps(data).encode()) <= 49152, 'SANDBOX_RESULT_OUTPUT_LIMIT')
  535. 535 temporary = receipt.with_suffix('.next')
  536. 536 require(not canceled(), 'EXECUTION_CANCELED')
  537. 537 temporary.write_text(json.dumps(data), encoding='utf-8')
  538. 538 temporary.replace(receipt)
  539. 539 return data

  543. 543 """在本机 Unix socket 暴露固定 RPC，并用 SO_PEERCRED、请求体大小和响应大小限制保护入口。"""
  544. 544 executor = Executor(root)

  550. 550 def do_POST(self):
  551. 551 try:
  552. 552 require(self.path == '/rpc', 'ENDPOINT_NOT_ALLOWED')
  553. 553 _, uid, _ = struct.unpack('3i', self.connection.getsockopt(socket.SOL_SOCKET, socket.SO_PEERCRED, 12))
  554. 554 require(uid in {0, 10001}, 'PEER_NOT_AUTHORIZED')
  555. 555 length = int(self.headers.get('Content-Length', '0'))
  556. 556 require(0 < length <= 49152, 'REQUEST_BYTE_LIMIT')
  557. 557 data = executor.invoke(json.loads(self.rfile.read(length)), lambda: peer_disconnected(self.connection))
  558. 558 body = json.dumps({'data': data}).encode()
  559. 559 require(len(body) <= 131072, 'RESPONSE_BYTE_LIMIT')
  560. 560 self.send_response(200)
  561. 561 except Exception as error:
  562. 562 body = json.dumps({'error_code': str(error) if isinstance(error, Rejected) else 'EXECUTOR_FAILED'}).encode()
  563. 563 self.send_response(409)
  564. 564 try:
  565. 565 self.send_header('Content-Type', 'application/json')
  566. 566 self.send_header('Content-Length', str(len(body)))
  567. 567 self.end_headers()
  568. 568 self.wfile.write(body)
  569. 569 except (BrokenPipeError, ConnectionResetError):
  570. 570 pass
