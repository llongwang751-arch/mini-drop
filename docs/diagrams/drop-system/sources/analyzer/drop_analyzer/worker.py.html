  83. 83 resolved.parent == root
  84. 84 and resolved.is_dir()

  90. 90def cleanup_indexes(
  91. 91 root_path: str, *, max_age_seconds=24 * 3600, max_bytes=512 * 1024 * 1024
  92. 92):
  93. 93 root = Path(root_path).resolve() / "indexes"
  94. 94 if not root.is_dir() or root.is_symlink():
  95. 95 return
  96. 96 entries = []
  97. 97 for candidate in root.iterdir():
  98. 98 if (
  99. 99 not candidate.is_symlink()
  100. 100 and candidate.is_file()
  101. 101 and re.fullmatch(r"hprof-[a-f0-9]{64}\.sqlite3", candidate.name)
  102. 102 and candidate.resolve().parent == root
  103. 103 ):
  104. 104 entries.append(
  105. 105 (candidate.stat().st_mtime, candidate.stat().st_size, candidate)
  106. 106 )
  107. 107 remaining = sum(size for _, size, _ in entries)
  108. 108 for modified, size, candidate in sorted(entries):
  109. 109 if modified < time.time() - max_age_seconds or remaining > max_bytes:
  110. 110 # 清理仅发生在作业之间；只删除已核对摘要命名数据库对应的 SQLite sidecar，
  111. 111 # 不处理任意路径。
  112. 112 candidate.unlink(missing_ok=True)
  113. 113 for suffix in ("-wal", "-shm"):
  114. 114 sidecar = candidate.with_name(candidate.name + suffix)
  115. 115 if not sidecar.is_symlink():
  116. 116 sidecar.unlink(missing_ok=True)
  117. 117 remaining -= size

  120. 120class LeaseGuard:
  121. 121 """为当前 Job 后台续约的生命周期对象。
  122. 122
  123. 123 续约异常与明确拒绝都立即标记 lost；check 用于提前停止下载/解析，SQL 写入仍会再次核对归属。
  124. 124 """
  125. 125 def __init__(self, repository: Repository, job: Job, renew_seconds: float):
  126. 126 self.repository, self.job, self.renew_seconds = repository, job, renew_seconds
  127. 127 self.stop = threading.Event()
  128. 128 self.lost = threading.Event()
  129. 129 self.thread = threading.Thread(
  130. 130 target=self._renew, name="analysis-lease", daemon=True
  131. 131 )
  132. 132
  133. 133 def _renew(self):
  134. 134 while not self.stop.wait(self.renew_seconds):
  135. 135 try:
  136. 136 renewed = self.repository.renew(self.job)
  137. 137 except Exception: # noqa: BLE001 -- 续期失败即禁止任何结果发布。
  138. 138 renewed = False
  139. 139 if not renewed:
  140. 140 self.lost.set()
  141. 141 return
  142. 142
  143. 143 def check(self):
  144. 144 if self.lost.is_set():
  145. 145 raise LeaseLost()
  146. 146
  147. 147 def __enter__(self):
  148. 148 self.thread.start()
  149. 149 return self
  150. 150
  151. 151 def __exit__(self, *_):
  152. 152 self.stop.set()
  153. 153 self.thread.join(timeout=15)

  156. 156def process_job(job: Job, repository: Repository, storage: ObjectStore, config: Config):
  157. 157 """处理一个已领取 Job，成功时返回公开分析结果，失败交由外层 run 决定重试与状态落库。
  158. 158
  159. 159 先收紧任务预算、核验 RAW 与辅助 ELF，再将解析放入受限子进程。
  160. 160 输出对象键包含采集和分析尝试，逐个上传后才调用 Repository.complete 原子发布数据库引用。
  161. 161 上传与数据库之间不是分布式事务，途中失去租约可能留下未引用对象，但不能发布过期结果。
  162. 162 """
  163. 163 pipeline = get_pipeline(job.pipeline)
  164. 164 task_budget = job.params.get("budget", {})
  165. 165 if not isinstance(task_budget, dict):
  166. 166 raise AnalysisError("INVALID_INPUT", "Task budget must be an object")
  167. 167
  168. 168 def budget(name, maximum):
  169. 169 value = task_budget.get(name, maximum)
  170. 170 if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
  171. 171 raise AnalysisError(
  172. 172 "INVALID_INPUT", "Task budget must contain positive integers"
  173. 173 )
  174. 174 return min(value, maximum)
  175. 175
  176. 176 limits = Limits(max_input_bytes=budget("max_file_bytes", 32 * 1024 * 1024))
  177. 177 with LeaseGuard(repository, job, config.renew_seconds) as lease:
  178. 178 artifact = repository.input_artifact(job)
  179. 179 auxiliary = (
  180. 180 repository.auxiliary_artifacts(job)
  181. 181 if hasattr(repository, "auxiliary_artifacts")
  182. 182 else []
  183. 183 )
  184. 184 inputs = [artifact, *auxiliary]
  185. 185 if (
  186. 186 any(
  187. 187 isinstance(item["size"], bool)
  188. 188 or not isinstance(item["size"], int)
  189. 189 or not 0 < item["size"] <= limits.max_input_bytes
  190. 190 for item in inputs
  191. 191 )
  192. 192 or sum(item["size"] for item in inputs) > config.analysis_download_bytes
  193. 193 ):
  194. 194 raise AnalysisError(
  195. 195 "INPUT_LIMIT_EXCEEDED",
  196. 196 "Combined raw and auxiliary downloads exceed the analysis byte budget",
  197. 197 )
  198. 198 work_root = Path(config.work_dir).resolve()
  199. 199 work_root.mkdir(mode=0o700, parents=True, exist_ok=True)
  200. 200 with tempfile.TemporaryDirectory(
  201. 201 prefix=f"job-{job.id}-", dir=work_root
  202. 202 ) as work_dir:
  203. 203 disk = WorkspaceBudget(
  204. 204 work_dir,
  205. 205 max_bytes=config.analysis_temp_bytes,
  206. 206 min_free_bytes=config.analysis_min_free_bytes,
  207. 207 index_dir=work_root / "indexes" if job.kind == "JAVA_HEAP" else None,
  208. 208 index_digest=artifact["sha256"],
  209. 209 )
  210. 210
  211. 211 def before_write(size):
  212. 212 lease.check()
  213. 213 disk.check(additional_bytes=size)
  214. 214
  215. 215 raw_path = Path(work_dir) / "raw.profile"
  216. 216 disk.check(additional_bytes=artifact["size"])
  217. 217 raw = storage.download(
  218. 218 artifact["object_key"],
  219. 219 raw_path,
  220. 220 expected_size=artifact["size"],
  221. 221 max_bytes=limits.max_input_bytes,
  222. 222 before_write=before_write,
  223. 223 )
  224. 224 lease.check()
  225. 225 if len(raw) > limits.max_input_bytes:
  226. 226 raise AnalysisError(
  227. 227 "INPUT_LIMIT_EXCEEDED", "Raw artifact exceeds the task byte budget"
  228. 228 )
  229. 229 if (
  230. 230 len(raw) != artifact["size"]
  231. 231 or hashlib.sha256(raw).hexdigest() != artifact["sha256"].lower()
  232. 232 ):
  233. 233 raise AnalysisError(
  234. 234 "ARTIFACT_INTEGRITY_ERROR",
  235. 235 "Raw artifact size or SHA-256 does not match metadata",
  236. 236 )
  237. 237 context = {
  238. 238 "pipeline": job.pipeline,
  239. 239 "kind": job.kind,
  240. 240 "format": artifact.get("format") or pipeline.formats[0],
  241. 241 "artifact": artifact,
  242. 242 "inputs": [artifact],
  243. 243 "params": job.params,
  244. 244 "raw_path": str(raw_path),
  245. 245 "work_dir": work_dir,
  246. 246 "limits": {
  247. 247 "input_bytes": limits.max_input_bytes,
  248. 248 "download_bytes": config.analysis_download_bytes,
  249. 249 "address_space_bytes": budget(
  250. 250 "max_address_space_bytes", config.analysis_memory_bytes
  251. 251 ),
  252. 252 "cpu_seconds": budget(
  253. 253 "max_cpu_seconds", config.analysis_cpu_seconds
  254. 254 ),
  255. 255 "processes": budget("max_processes", 16),
  256. 256 "wall_seconds": config.analysis_wall_seconds,
  257. 257 "output_bytes": budget("max_file_bytes", 32 * 1024 * 1024),
  258. 258 "temp_bytes": config.analysis_temp_bytes,
  259. 259 "min_free_bytes": config.analysis_min_free_bytes,
  260. 260 },
  261. 261 "index_dir": str(work_root / "indexes"),
  262. 262 }
  263. 263 for binary in auxiliary:
  264. 264 binary_path = Path(work_dir) / "target.elf"
  265. 265 disk.check(additional_bytes=binary["size"])
  266. 266 binary_raw = storage.download(
  267. 267 binary["object_key"],
  268. 268 binary_path,
  269. 269 expected_size=binary["size"],
  270. 270 max_bytes=limits.max_input_bytes,
  271. 271 before_write=before_write,
  272. 272 )
  273. 273 if hashlib.sha256(binary_raw).hexdigest() != binary["sha256"].lower():
  274. 274 raise AnalysisError(
  275. 275 "ARTIFACT_INTEGRITY_ERROR",
  276. 276 "Auxiliary binary SHA-256 mismatches metadata",
  277. 277 )
  278. 278 context["binary_path"] = str(binary_path)
  279. 279 context["inputs"].append(binary)
  280. 280 disk.check()
  281. 281 pipeline.validate(context)
  282. 282 isolated = analyze_isolated(
  283. 283 context,
  284. 284 check_lease=lease.check,
  285. 285 on_progress=(lambda stage: repository.progress(job, stage))
  286. 286 if hasattr(repository, "progress")
  287. 287 else None,
  288. 288 )
  289. 289 isolated["metrics"]["downloaded_bytes"] = sum(
  290. 290 item["size"] for item in context["inputs"]
  291. 291 )
  292. 292 telemetry.observe(
  293. 293 "duration_seconds",
  294. 294 isolated["metrics"]["duration_ms"] / 1000,
  295. 295 kind=job.kind,
  296. 296 )
  297. 297 telemetry.observe(
  298. 298 "cpu_seconds", isolated["metrics"]["cpu_seconds"], kind=job.kind
  299. 299 )
  300. 300 telemetry.observe(
  301. 301 "temp_disk_bytes", isolated["metrics"]["temp_disk_bytes"], kind=job.kind
  302. 302 )
  303. 303 if isolated["metrics"]["max_rss_bytes"] is not None:
  304. 304 telemetry.observe(
  305. 305 "max_rss_bytes", isolated["metrics"]["max_rss_bytes"], kind=job.kind
  306. 306 )
  307. 307 parsed = isolated["parsed"]
  308. 308 partial = build_partial(
  309. 309 parsed, job, artifact, private_paths=(work_dir, str(work_root))
  310. 310 )
  311. 311 if partial is not None:
  312. 312 lease.check()
  313. 313 repository.save_partial(job, partial, artifact)
  314. 314 parsed["metadata"]["resource_usage"] = isolated["metrics"]
  315. 315 parsed["metadata"]["input_digest"] = artifact["sha256"]
  316. 316 settings = (
  317. 317 repository.rule_settings()
  318. 318 if hasattr(repository, "rule_settings")
  319. 319 else []
  320. 320 )
  321. 321 result = make_result(
  322. 322 parsed, job.kind, {**job.params, "_diagnostic_rules": settings}
  323. 323 )
  324. 324 prefix = f"results/tasks/{job.task_id}/attempts/{job.collection_attempt}/analysis/{job.id}/runs/{job.attempt}"
  325. 325 outputs = build_outputs(
  326. 326 parsed,
  327. 327 task_id=str(job.task_id),
  328. 328 job_id=str(job.id),
  329. 329 input_sha256=artifact["sha256"],
  330. 330 public_result=result,
  331. 331 object_prefix=prefix,
  332. 332 attempt_id=str(job.attempt_id),
  333. 333 input_artifact_id=str(artifact["id"]) if "id" in artifact else None,
  334. 334 )
  335. 335 manifest = json.loads(outputs["manifest.json"])
  336. 336 manifest.update(isolated["manifest"])
  337. 337 manifest["metadata"] = parsed["metadata"]
  338. 338 manifest["resource_usage"] = isolated["metrics"]
  339. 339 for extra in parsed.get("extra_outputs", []):
  340. 340 name = extra["name"]
  341. 341 if name != "optimized.elf":
  342. 342 raise AnalysisError(
  343. 343 "INVALID_OUTPUT", "Unregistered analysis output name"
  344. 344 )
  345. 345 body = (Path(work_dir) / name).read_bytes()
  346. 346 if len(body) > 32 * 1024 * 1024:
  347. 347 raise AnalysisError(
  348. 348 "OUTPUT_LIMIT_EXCEEDED", "Analysis artifact exceeds byte limit"
  349. 349 )
  350. 350 outputs[name] = body
  351. 351 manifest["artifacts"].append(
  352. 352 {
  353. 353 "name": name,
  354. 354 "size": len(body),
  355. 355 "sha256": hashlib.sha256(body).hexdigest(),
  356. 356 "object_key": f"{prefix}/{name}",
  357. 357 }
  358. 358 )
  359. 359 outputs["manifest.json"] = json_bytes(manifest)
  360. 360 disk.check()
  361. 361 lease.check()
  362. 362 artifacts = []
  363. 363 for name, body in outputs.items():
  364. 364 lease.check()
  365. 365 disk.check()
  366. 366 content_type = (
  367. 367 "application/x-elf"
  368. 368 if name.endswith(".elf")
  369. 369 else "application/gzip"
  370. 370 if name.endswith(".gz")
  371. 371 else "application/json"
  372. 372 )
  373. 373 metadata = storage.upload(f"{prefix}/{name}", body, content_type)
  374. 374 artifacts.append(
  375. 375 {
  376. 376 **metadata,
  377. 377 "name": name,
  378. 378 "kind": "MANIFEST" if name == "manifest.json" else "RESULT",
  379. 379 "format": "elf"
  380. 380 if name.endswith(".elf")
  381. 381 else "json_gzip"
  382. 382 if name.endswith(".gz")
  383. 383 else "json",
  384. 384 "metadata": {
  385. 385 "analyzer_version": __version__,
  386. 386 "pipeline": job.pipeline,
  387. 387 "source": result["summary"]["source"],
  388. 388 "input_digest": artifact["sha256"],
  389. 389 },
  390. 390 }
  391. 391 )
  392. 392 lease.check()
  393. 393 repository.complete(job, result, artifacts)
  394. 394 return result

  397. 397class Heartbeat:
  398. 398 """向本地健康文件原子写入存活时间与依赖就绪状态，容器探针无需开放管理网络端口。"""
  399. 399 def __init__(self, path: str):
  400. 400 self.path = Path(path)
  401. 401 self.ready = False
  402. 402 self.stop = threading.Event()
  403. 403 self.thread = threading.Thread(
  404. 404 target=self._run, name="analysis-health", daemon=True
  405. 405 )
  406. 406
  407. 407 def _run(self):
  408. 408 self.path.parent.mkdir(parents=True, exist_ok=True)
  409. 409 while not self.stop.is_set():
  410. 410 try:
  411. 411 temporary = self.path.with_suffix(".tmp")
  412. 412 temporary.write_text(
  413. 413 json.dumps(
  414. 414 {
  415. 415 "updated_at": time.time(),
  416. 416 "ready": self.ready,
  417. 417 "version": __version__,
  418. 418 }
  419. 419 ),
  420. 420 encoding="utf-8",
  421. 421 )
  422. 422 os.replace(temporary, self.path)
  423. 423 except OSError:
  424. 424 pass
  425. 425 self.stop.wait(2)

  428. 428def run(config: Config):
  429. 429 """维护单作业领取循环与依赖健康状态，收到停机信号后停止领取新任务。
  430. 430
  431. 431 LeaseLost 只记归属失效，不能覆盖新持有者状态；失败落库不可用时保留租约到期恢复路径，
  432. 432 而不是伪造终态或将数据库异常无限重试在当前调用栈中。
  433. 433 """
  434. 434 install_logging("drop-analyzer", __version__)
  435. 435 stop = threading.Event()
  436. 436 for signum in (signal.SIGTERM, signal.SIGINT):
  437. 437 signal.signal(signum, lambda *_: stop.set())
  438. 438 visible, digest = config.snapshot()
  439. 439 emit(
  440. 440 "drop-analyzer",
  441. 441 __version__,
  442. 442 "configuration_loaded",
  443. 443 stage="startup",
  444. 444 effective_config=visible,
  445. 445 config_hash=digest,
  446. 446 )
  447. 447 telemetry.start(config)
  448. 448 repository = Repository(
  449. 449 config.database_url, config.worker_id, config.lease_seconds, config.queue
  450. 450 )
  451. 451 storage = ObjectStore(config)
  452. 452 heartbeat = Heartbeat(config.health_file)
  453. 453 heartbeat.thread.start()
  454. 454 cleanup_stale_workdirs(config.work_dir)
  455. 455 cleanup_indexes(config.work_dir)
  456. 456 last_dependency_check = 0.0
  457. 457 log(
  458. 458 "worker_started",
  459. 459 worker_id=config.worker_id,
  460. 460 lease_seconds=config.lease_seconds,
  461. 461 concurrency=1,
  462. 462 )
  463. 463 try:
  464. 464 while not stop.is_set():
  465. 465 job = None
  466. 466 started = time.monotonic()
  467. 467 try:
  468. 468 cleanup_indexes(config.work_dir)
  469. 469 if not heartbeat.ready or time.monotonic() - last_dependency_check > 30:
  470. 470 heartbeat.ready = repository.ready() and storage.ready()
  471. 471 if heartbeat.ready:
  472. 472 statuses = {
  473. 473 row["status"]: row["count"]
  474. 474 for row in repository.queue_stats()
  475. 475 }
  476. 476 for status in (
  477. 477 "pending",
  478. 478 "running",
  479. 479 "retry",
  480. 480 "success",
  481. 481 "failed",
  482. 482 ):
  483. 483 telemetry.gauge(
  484. 484 "analysis_jobs", statuses.get(status, 0), status=status
  485. 485 )
  486. 486 last_dependency_check = time.monotonic()
  487. 487 if not heartbeat.ready:
  488. 488 stop.wait(config.poll_seconds)
  489. 489 continue
  490. 490 job = repository.claim()
  491. 491 if job is None:
  492. 492 stop.wait(config.poll_seconds)
  493. 493 continue
  494. 494 log("analysis_started", job=job)
  495. 495 with telemetry.analysis_span(job):
  496. 496 result = process_job(job, repository, storage, config)
  497. 497 telemetry.count("jobs_finished_total", kind=job.kind, status="success")
  498. 498 cleanup_indexes(config.work_dir)
  499. 499 log(
  500. 500 "analysis_succeeded",
  501. 501 job=job,
  502. 502 duration_ms=int((time.monotonic() - started) * 1000),
  503. 503 total_samples=result["total_samples"],
  504. 504 node_count=result["node_count"],
  505. 505 )
  506. 506 except LeaseLost:
  507. 507 telemetry.count("lease_lost_total", kind=job.kind if job else "all")
  508. 508 log(
  509. 509 "analysis_lease_lost",
  510. 510 job=job,
  511. 511 level="warning",
  512. 512 error_code="ANALYSIS_LEASE_LOST",
  513. 513 duration_ms=int((time.monotonic() - started) * 1000),
  514. 514 )
  515. 515 except Exception as exc: # noqa: BLE001 -- 统一不可信工具失败，不输出原始诊断
  516. 516 error = (
  517. 517 exc
  518. 518 if isinstance(exc, AnalysisError)
  519. 519 else AnalysisError(
  520. 520 "ANALYSIS_INTERNAL_ERROR",
  521. 521 "Analysis service encountered a temporary failure",
  522. 522 retryable=True,
  523. 523 )
  524. 524 )
  525. 525 telemetry.count(
  526. 526 "jobs_finished_total" if job else "dependency_failures_total",
  527. 527 kind=job.kind if job else "all",
  528. 528 status="failure",
  529. 529 )
  530. 530 log(
  531. 531 "analysis_failed" if job else "worker_dependency_unavailable",
  532. 532 job=job,
  533. 533 level="error",
  534. 534 error_code=error.code,
  535. 535 retryable=error.retryable,
  536. 536 error_type=type(exc).__name__,
  537. 537 duration_ms=int((time.monotonic() - started) * 1000),
  538. 538 )
  539. 539 heartbeat.ready = False
  540. 540 if job:
  541. 541 try:
  542. 542 repository.fail(job, error)

  545. 545 except Exception as failure: # noqa: BLE001 -- 数据库写入失败后交由租约恢复
