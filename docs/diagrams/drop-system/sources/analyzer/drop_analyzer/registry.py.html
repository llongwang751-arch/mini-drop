  60. 60 def analyze(self, raw, context):

  62. 62
  63. 63 BOLT 可处理受控 ELF 并生成优化文件，但这里不会执行上传二进制；

  65. 65 """
  66. 66 kind, fmt = context["kind"], context["format"]

  68. 68 **context.get("params", {}),
  69. 69 **context.get("params", {}).get("options", {}),
  70. 70 }

  84. 84 else analyze_folded(decoded)

  88. 88 "artifact"

  91. 91 metadata.get("perf")

  94. 94 else None

  97. 97 elif fmt == "perf_data":

  99. 99 elif fmt == "pprof":
  100. 100 parsed = analyze_pprof(raw, kind=kind, params=params)
  101. 101 elif fmt == "jsonl":

  103. 103 elif fmt == "json":

  105. 105 analyze_resources(raw)
  106. 106 if kind == "RESOURCE_SNAPSHOT"
  107. 107 else analyze_script(raw)
  108. 108 )
  109. 109 elif fmt == "memray":

  111. 111 elif fmt == "hprof":

  113. 113 raw,
  114. 114 work_dir=context.get("index_dir"),

  116. 116 on_progress=lambda stage: write_stage(work_dir, stage),

  120. 120 elif fmt == "gperf_cpu":

  122. 122 elif fmt in {"gperf_heap", "jeprof_heap"}:

  131. 131 parsed["extra_outputs"] = [

  134. 134 elif not context.get("binary_path"):

  136. 136 "未附目标 ELF，仅分析 fdata，未生成优化二进制。"
  137. 137 )
  138. 138 else:
  139. 139 raise AnalysisError(
  140. 140 "UNSUPPORTED_FORMAT", "No decoder is registered for this raw format"
  141. 141 )
  142. 142 if fmt in {"gperf_cpu", "gperf_heap", "jeprof_heap"} and context.get(
  143. 143 "binary_path"
  144. 144 ):
  145. 145 parsed = symbolize_native(parsed, context["binary_path"], work_dir=work_dir)
  146. 146 source = {
  147. 147 "PERF_CPU": "perf",
  148. 148 "PROC_CPU": "procfs",
  149. 149 "IMPORT_COLLAPSED": "import",
  150. 150 }.get(kind, self.source)
  151. 151 if fmt == "jfr":
  152. 152 source = "jfr"
  153. 153 imported = (
  154. 154 kind == "IMPORT_COLLAPSED"
  155. 155 or context.get("params", {}).get("source") == "import"

  163. 163 "input_format": fmt,
  164. 164 "pipeline": self.pipeline,

  167. 167 "collector": context["artifact"].get("runner_metadata")
