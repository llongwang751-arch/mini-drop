  17. 17def run_tool(
  18. 18 argv, *, cwd, output_name, timeout=60, max_bytes=32 * 1024 * 1024, environment=None
  19. 19):
  20. 20 executable = shutil.which(argv[0])
  21. 21 if not executable:
  22. 22 raise AnalysisError(
  23. 23 "ANALYZER_TOOL_MISSING",
  24. 24 f"Required analysis tool {Path(argv[0]).name} is unavailable",
  25. 25 )
  26. 26 executable = str(Path(executable).resolve())
  27. 27 destination = Path(cwd) / output_name
  28. 28 error_file = Path(cwd) / (output_name + ".stderr")
  29. 29 env = {
  30. 30 "PATH": os.environ.get("PATH", ""),
  31. 31 "HOME": str(cwd),
  32. 32 "TMPDIR": str(cwd),
  33. 33 "TMP": str(cwd),
  34. 34 "TEMP": str(cwd),
  35. 35 "LANG": "C.UTF-8",
  36. 36 "LC_ALL": "C.UTF-8",
  37. 37 **(environment or {}),
  38. 38 }
  39. 39 try:
  40. 40 with destination.open("wb") as output, error_file.open("wb") as errors:
  41. 41 process = subprocess.run(
  42. 42 [executable, *map(str, argv[1:])],
  43. 43 cwd=cwd,
  44. 44 stdin=subprocess.DEVNULL,
  45. 45 stdout=output,
  46. 46 stderr=errors,
  47. 47 env=env,
  48. 48 timeout=timeout,
  49. 49 check=False,
  50. 50 )
  51. 51 except subprocess.TimeoutExpired as exc:
  52. 52 raise AnalysisError(
  53. 53 "ANALYZER_TOOL_TIMEOUT", "Analysis tool exceeded its execution time budget"
  54. 54 ) from exc
  55. 55 if (
  56. 56 destination.stat().st_size > max_bytes
  57. 57 or error_file.stat().st_size > 1024 * 1024
  58. 58 ):
  59. 59 raise AnalysisError(
  60. 60 "OUTPUT_LIMIT_EXCEEDED", "Analysis tool output exceeded its byte budget"
  61. 61 )
  62. 62 if process.returncode:
  63. 63 raise AnalysisError(
  64. 64 "ANALYZER_TOOL_FAILED",
  65. 65 f"Analysis tool {Path(argv[0]).name} rejected the profile",
  66. 66 )
  67. 67 return destination.read_bytes()

  70. 70def analyze_jfr(path, *, work_dir):
  71. 71 # jfr print 是 OpenJDK 解码器；转换后的 JSON 调用栈仍来自真实 JFR 事件。
  72. 72 raw = run_tool(
  73. 73 [
  74. 74 "jfr",
  75. 75 "print",
  76. 76 "--json",
  77. 77 "--stack-depth",
  78. 78 "128",
  79. 79 "--events",
  80. 80 "jdk.ExecutionSample,jdk.NativeMethodSample,profiler.ExecutionSample",
  81. 81 path,
  82. 82 ],
  83. 83 cwd=work_dir,
  84. 84 output_name="jfr.json",
  85. 85 environment={
  86. 86 "JAVA_TOOL_OPTIONS": "-Xms16m -Xmx128m -Xss256k -XX:+UseSerialGC -XX:MaxMetaspaceSize=64m -XX:MaxDirectMemorySize=16m -XX:ReservedCodeCacheSize=32m -XX:ActiveProcessorCount=1 -XX:-UseCompressedOops -XX:-UseCompressedClassPointers -XX:-UsePerfData",
  87. 87 "MALLOC_ARENA_MAX": "2",
  88. 88 },
  89. 89 )
  90. 90 try:
  91. 91 value = json.loads(raw)
  92. 92 except (ValueError, RecursionError):
  93. 93 raise AnalysisError("INVALID_INPUT", "JFR tool produced invalid JSON")
  94. 94 events = value.get("recording", {}).get("events", [])
  95. 95 if not isinstance(events, list) or len(events) > 200_000:
  96. 96 raise AnalysisError(
  97. 97 "INPUT_LIMIT_EXCEEDED", "JFR execution sample count exceeds limit"
  98. 98 )
  99. 99 stacks = []
  100. 100 for event in events:
  101. 101 fields = event.get("values", {})
  102. 102 stack = fields.get("stackTrace") or {}
  103. 103 frames = stack.get("frames", [])
  104. 104 names = []
  105. 105 for frame in reversed(frames):
  106. 106 method = frame.get("method", {})
  107. 107 cls = method.get("type", {}).get("name", "[unknown]").replace("/", ".")
  108. 108 names.append(cls + "." + method.get("name", "[unknown]"))
  109. 109 thread = fields.get("sampledThread") or fields.get("eventThread") or {}
  110. 110 if thread.get("javaName"):
  111. 111 names.insert(0, "thread:" + thread["javaName"])
  112. 112 stacks.append((names, 1))
  113. 113 if not stacks:
  114. 114 raise AnalysisError(
  115. 115 "INVALID_INPUT",
  116. 116 "JFR recording contains no supported execution sample events",
  117. 117 )
  118. 118 return stack_result(
  119. 119 stacks,
  120. 120 metadata={
  121. 121 "format": "jfr",
  122. 122 "decoded_by": "OpenJDK jfr print",
  123. 123 "event_count": len(events),
  124. 124 },
  125. 125 )

  128. 128def analyze_memray(path):
  129. 129 try:
  130. 130 import memray
  131. 131 except ImportError as exc:
  132. 132 raise AnalysisError(
  133. 133 "ANALYZER_TOOL_MISSING", "memray FileReader is unavailable on this worker"
  134. 134 ) from exc
  135. 135 try:
  136. 136 with memray.FileReader(
  137. 137 str(path), report_progress=False, max_memory_records=3600
  138. 138 ) as reader:
  139. 139 metadata = reader.metadata
  140. 140 leaked, high, allocations = [], [], 0
  141. 141 for index, record in enumerate(
  142. 142 reader.get_leaked_allocation_records(merge_threads=True)
  143. 143 ):
  144. 144 if index >= 200_000:
  145. 145 raise AnalysisError(
  146. 146 "INPUT_LIMIT_EXCEEDED",
  147. 147 "memray allocation stack count exceeds limit",
  148. 148 )
  149. 149 frames = [
  150. 150 f"{function} ({filename}:{line})"
  151. 151 for function, filename, line in reversed(
  152. 152 record.stack_trace(max_stacks=128)
  153. 153 )
  154. 154 ]
  155. 155 leaked.append(
  156. 156 {
  157. 157 "bytes": record.size,
  158. 158 "allocations": record.n_allocations,
  159. 159 "stack": frames,
  160. 160 "classification": "unfreed_at_capture_end",
  161. 161 }
  162. 162 )
  163. 163 for index, record in enumerate(
  164. 164 reader.get_high_watermark_allocation_records(merge_threads=True)
  165. 165 ):
  166. 166 if index >= 200_000:
  167. 167 raise AnalysisError(
  168. 168 "INPUT_LIMIT_EXCEEDED",
  169. 169 "memray high watermark stack count exceeds limit",
  170. 170 )
  171. 171 frames = [
  172. 172 f"{function} ({filename}:{line})"
  173. 173 for function, filename, line in reversed(
  174. 174 record.stack_trace(max_stacks=128)
  175. 175 )
  176. 176 ]
  177. 177 high.append((frames, record.size))
  178. 178 snapshots = [
  179. 179 {
  180. 180 "time_ns": snapshot.time * 1_000_000,
  181. 181 "rss_bytes": snapshot.rss,
  182. 182 "live_bytes": snapshot.heap,
  183. 183 }
  184. 184 for snapshot in reader.get_memory_snapshots()
  185. 185 ][:3600]
  186. 186 total_allocated, total_freed, free_count = None, None, None
  187. 187 # 聚合 memray 文件本来就不包含完整事件流；无法取得的总量用 null 表示，
  188. 188 # 不能把缺失信息当作零。
  189. 189 if (
  190. 190 getattr(metadata, "file_format", None)
  191. 191 == memray.FileFormat.ALL_ALLOCATIONS
  192. 192 ):
  193. 193 total_allocated, total_freed, free_count = 0, 0, 0
  194. 194 live = {}
  195. 195 free_types = {
  196. 196 memray.AllocatorType.FREE,
  197. 197 memray.AllocatorType.MUNMAP,
  198. 198 memray.AllocatorType.PYMALLOC_FREE,
  199. 199 }
  200. 200 for index, record in enumerate(reader.get_allocation_records()):
  201. 201 if index >= 500_000:
  202. 202 raise AnalysisError(
  203. 203 "INPUT_LIMIT_EXCEEDED", "memray event count exceeds limit"
  204. 204 )
  205. 205 if record.allocator in free_types:
  206. 206 total_freed += live.pop(record.address, 0)
  207. 207 free_count += 1
  208. 208 else:
  209. 209 total_allocated += record.size
  210. 210 allocations += record.n_allocations
  211. 211 live[record.address] = record.size
  212. 212 parsed = stack_result(
  213. 213 high,
  214. 214 unit="bytes",
  215. 215 metadata={
  216. 216 "format": "memray",
  217. 217 "profile_type": "high_watermark",
  218. 218 "tool_version": memray.__version__,
  219. 219 "native_traces": bool(
  220. 220 getattr(metadata, "has_native_traces", False)
  221. 221 ),
  222. 222 "start_time": str(metadata.start_time),
  223. 223 "end_time": str(metadata.end_time),
  224. 224 },
  225. 225 )
  226. 226 parsed["sections"] = {
  227. 227 "memory": {
  228. 228 "allocated_bytes": total_allocated,
  229. 229 "freed_bytes": total_freed,
  230. 230 "live_bytes": sum(row["bytes"] for row in leaked),
  231. 231 "peak_live_bytes": getattr(
  232. 232 metadata, "peak_memory", sum(weight for _, weight in high)
  233. 233 ),
  234. 234 "allocations": allocations if total_allocated is not None else None,
  235. 235 "frees": free_count,
  236. 236 "unknown_frees": None,
  237. 237 "pagefaults": {"minor": None, "major": None},
  238. 238 "lost_events": None,
  239. 239 "survivors": sorted(leaked, key=lambda row: -row["bytes"])[:100],
  240. 240 "timeseries": snapshots,
  241. 241 "limitations": [
  242. 242 "memray 记录分配，不记录 page fault；捕获结束仍存活不等于泄漏。",
  243. 243 "聚合格式没有逐次 alloc/free 流，对应总量为 null。",
  244. 244 ],
  245. 245 }
  246. 246 }
  247. 247 return parsed
  248. 248 except AnalysisError:
  249. 249 raise
  250. 250 except (OSError, RuntimeError, ValueError) as exc:
  251. 251 raise AnalysisError(
  252. 252 "INVALID_INPUT", "memray FileReader rejected the allocation capture"
  253. 253 ) from exc

  256. 256def symbolize_native(parsed, binary_path, *, work_dir):
  257. 257 addresses = sorted(
  258. 258 {
  259. 259 row["name"]
  260. 260 for row in parsed["top"]["inclusive"]
  261. 261 if row["name"].startswith("0x")
  262. 262 }
  263. 263 )
  264. 264 if not addresses:
  265. 265 return parsed
  266. 266 # 按完整树中的地址进行符号化，而非只看 TopN；地址数量另设上限。
  267. 267 addresses = set()
  268. 268 queue = [parsed["flamegraph"]]
  269. 269 while queue:
  270. 270 node = queue.pop()
  271. 271 if node["name"].startswith("0x"):
  272. 272 addresses.add(node["name"])
  273. 273 queue.extend(node["children"])
  274. 274 if len(addresses) > 10_000:
  275. 275 raise AnalysisError(
  276. 276 "INPUT_LIMIT_EXCEEDED", "Symbolization address count exceeds limit"
  277. 277 )
  278. 278 ordered = sorted(addresses)
  279. 279 output = run_tool(
  280. 280 ["addr2line", "-f", "-C", "-e", binary_path, *ordered],
  281. 281 cwd=work_dir,
  282. 282 output_name="symbols.txt",
  283. 283 max_bytes=4 * 1024 * 1024,
  284. 284 )
  285. 285 lines = output.decode("utf8", errors="replace").splitlines()
  286. 286 if len(lines) != 2 * len(ordered):
  287. 287 raise AnalysisError(
  288. 288 "ANALYZER_TOOL_FAILED", "Symbolizer output length mismatches addresses"
  289. 289 )
  290. 290 symbols = {
  291. 291 address: lines[2 * index]
  292. 292 for index, address in enumerate(ordered)
  293. 293 if lines[2 * index] != "??"
  294. 294 }
  295. 295 stacks, queue = [], [(parsed["flamegraph"], [])]
  296. 296 while queue:
  297. 297 node, frames = queue.pop()
  298. 298 if node["name"] != "all":
  299. 299 frames = [*frames, symbols.get(node["name"], node["name"])]
  300. 300 if node["self"]:
  301. 301 stacks.append((frames, node["self"]))
  302. 302 queue.extend((child, frames) for child in node["children"])
  303. 303 result = stack_result(
  304. 304 stacks,
  305. 305 unit=parsed["metadata"]["unit"],
  306. 306 metadata={
  307. 307 **parsed["metadata"],
  308. 308 "symbolization": "addr2line_partial",
  309. 309 "resolved_addresses": len(symbols),
  310. 310 "unresolved_addresses": len(ordered) - len(symbols),
  311. 311 },
  312. 312 )
  313. 313 result["sections"] = parsed.get("sections", {})
  314. 314 return result
