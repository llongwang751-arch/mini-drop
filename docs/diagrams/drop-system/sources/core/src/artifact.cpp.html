  36. 36std::string sha256_file(const std::string &file) {
  37. 37 std::ifstream input(file, std::ios::binary);
  38. 38 if (!input)
  39. 39 throw std::runtime_error("Cannot open artifact for hashing");

  41. 41 EVP_MD_CTX_new(), EVP_MD_CTX_free);

  43. 43 throw std::runtime_error("SHA-256 initialization failed");
  44. 44 std::array<char, 32768> buffer;
  45. 45 while (input) {
  46. 46 input.read(buffer.data(), buffer.size());
  47. 47 if (EVP_DigestUpdate(context.get(), buffer.data(),
  48. 48 static_cast<std::size_t>(input.gcount())) != 1)
  49. 49 throw std::runtime_error("SHA-256 failed");
  50. 50 }
  51. 51 if (!input.eof())
  52. 52 throw std::runtime_error("Artifact read failed");
  53. 53 std::array<unsigned char, EVP_MAX_MD_SIZE> digest;
  54. 54 unsigned int length = 0;
  55. 55 EVP_DigestFinal_ex(context.get(), digest.data(), &length);
  56. 56 std::ostringstream hex;
  57. 57 for (unsigned int i = 0; i < length; ++i)
  58. 58 hex << std::hex << std::setw(2) << std::setfill('0')
  59. 59 << static_cast<int>(digest[i]);
  60. 60 return hex.str();

  84. 84std::size_t upload_header(char *data, std::size_t size, std::size_t count,
  85. 85 void *user) {
  86. 86 const auto bytes = size * count;
  87. 87 if (bytes > 4096)
  88. 88 return 0;
  89. 89 const std::string line(data, bytes);
  90. 90 auto &headers = *static_cast<UploadHeaders *>(user);
  91. 91 if (line.rfind("HTTP/", 0) == 0)
  92. 92 headers = UploadHeaders{};
  93. 93 auto colon = line.find(':');
  94. 94 if (colon != std::string::npos) {
  95. 95 auto name = line.substr(0, colon);
  96. 96 std::transform(
  97. 97 name.begin(), name.end(), name.begin(),
  98. 98 [](unsigned char ch) { return static_cast<char>(std::tolower(ch)); });
  99. 99 if (name == "etag") {
  100. 100 const auto start = line.find_first_not_of(" \t", colon + 1);
  101. 101 const auto end = line.find_last_not_of(" \t\r\n");
  102. 102 if (start != std::string::npos && end >= start &&
  103. 103 end - start + 1 <= 1024) {
  104. 104 const auto value = line.substr(start, end - start + 1);
  105. 105 if (std::all_of(value.begin(), value.end(), [](unsigned char ch) {
  106. 106 return ch >= 0x20 && ch < 0x7f;
  107. 107 }))
  108. 108 headers.etag = value;
  109. 109 }
  110. 110 }
  111. 111 if (name == "retry-after") {
  112. 112 const auto start = line.find_first_not_of(" \t", colon + 1);
  113. 113 if (start != std::string::npos) {
  114. 114 unsigned seconds = 0;
  115. 115 const auto result = std::from_chars(line.data() + start,
  116. 116 line.data() + line.size(), seconds);
  117. 117 if (result.ec == std::errc{})
  118. 118 headers.retry_after = std::min(seconds, 60U);
  119. 119 }
  120. 120 }
  121. 121 }
  122. 122 return bytes;
  123. 123}

  136. 136UploadReceipt upload_file(const std::string &url, const std::string &file,
  137. 137 const std::string &content_type,
  138. 138 std::uint64_t max_bytes) {
  139. 139 if (url.rfind("http://", 0) != 0 && url.rfind("https://", 0) != 0)
  140. 140 throw std::invalid_argument("Upload URL must use HTTP(S)");
  141. 141 auto size = std::filesystem::file_size(file);
  142. 142 if (size == 0 || size > max_bytes || max_bytes > 512ULL * 1024 * 1024)
  143. 143 throw std::runtime_error("Artifact size outside upload budget");
  144. 144 struct stat statbuf {};
  145. 145 if (lstat(file.c_str(), &statbuf) != 0 || !S_ISREG(statbuf.st_mode))
  146. 146 throw std::runtime_error("Artifact must be a regular file");
  147. 147 std::unique_ptr<FILE, CloseFile> input(std::fopen(file.c_str(), "rb"));
  148. 148 if (!input)
  149. 149 throw std::runtime_error("Cannot open artifact");
  150. 150 std::unique_ptr<CURL, decltype(&curl_easy_cleanup)> curl(curl_easy_init(),
  151. 151 curl_easy_cleanup);
  152. 152 if (!curl)
  153. 153 throw std::runtime_error("Upload initialization failed");
  154. 154 curl_slist *raw_headers = nullptr;
  155. 155 raw_headers =
  156. 156 curl_slist_append(raw_headers, ("Content-Type: " + content_type).c_str());
  157. 157 raw_headers = curl_slist_append(raw_headers, "Expect:");
  158. 158 std::unique_ptr<curl_slist, decltype(&curl_slist_free_all)> headers(
  159. 159 raw_headers, curl_slist_free_all);
  160. 160 curl_easy_setopt(curl.get(), CURLOPT_URL, url.c_str());
  161. 161 curl_easy_setopt(curl.get(), CURLOPT_UPLOAD, 1L);
  162. 162 curl_easy_setopt(curl.get(), CURLOPT_READDATA, input.get());
  163. 163 curl_easy_setopt(curl.get(), CURLOPT_INFILESIZE_LARGE,
  164. 164 static_cast<curl_off_t>(size));
  165. 165 curl_easy_setopt(curl.get(), CURLOPT_HTTPHEADER, headers.get());
  166. 166 curl_easy_setopt(curl.get(), CURLOPT_CONNECTTIMEOUT, 5L);
  167. 167 curl_easy_setopt(curl.get(), CURLOPT_TIMEOUT, 20L);
  168. 168 curl_easy_setopt(curl.get(), CURLOPT_NOSIGNAL, 1L);
  169. 169 curl_easy_setopt(curl.get(), CURLOPT_PROTOCOLS_STR, "http,https");
  170. 170 curl_easy_setopt(curl.get(), CURLOPT_WRITEFUNCTION, discard_response);
  171. 171 UploadHeaders response_headers;
  172. 172 curl_easy_setopt(curl.get(), CURLOPT_HEADERFUNCTION, upload_header);
  173. 173 curl_easy_setopt(curl.get(), CURLOPT_HEADERDATA, &response_headers);
  174. 174 // 上传不跟随重定向：预签名URL只授权一个确定的
  175. 175 // 目标地址，不能让重定向把制品或授权发送到另一处。
  176. 176 auto result = curl_easy_perform(curl.get());
  177. 177 long status = 0;
  178. 178 curl_easy_getinfo(curl.get(), CURLINFO_RESPONSE_CODE, &status);
  179. 179 if (result != CURLE_OK)
  180. 180 throw UploadFailure(transient_curl(result), status,
  181. 181 response_headers.retry_after);
  182. 182 if (status < 200 || status > 299)
  183. 183 throw UploadFailure(retryable_upload_http(status), status,
  184. 184 response_headers.retry_after);
  185. 185 return {response_headers.etag,
  186. 186 std::chrono::duration_cast<std::chrono::milliseconds>(
  187. 187 std::chrono::system_clock::now().time_since_epoch())
  188. 188 .count()};
  189. 189}

  191. 191void atomic_write(const std::string &path, const std::string &content) {
  192. 192 const std::string temporary = path + ".tmp";
  193. 193 int fd = open(temporary.c_str(),
  194. 194 O_WRONLY | O_CREAT | O_NONBLOCK | O_NOFOLLOW | O_CLOEXEC, 0600);
  195. 195 if (fd < 0)
  196. 196 throw std::runtime_error("Could not open local recovery journal");
  197. 197 struct stat target {};
  198. 198 if (fstat(fd, &target) || !S_ISREG(target.st_mode) ||
  199. 199 target.st_uid != geteuid() || target.st_nlink != 1 || ftruncate(fd, 0)) {
  200. 200 close(fd);
  201. 201 throw std::runtime_error(
  202. 202 "Local recovery journal must be an owned regular file");
  203. 203 }
  204. 204 std::size_t written = 0;
  205. 205 while (written < content.size()) {
  206. 206 auto count = write(fd, content.data() + written, content.size() - written);
  207. 207 if (count <= 0) {
  208. 208 close(fd);
  209. 209 throw std::runtime_error("Local recovery journal write failed");
  210. 210 }
  211. 211 written += static_cast<std::size_t>(count);
  212. 212 }
  213. 213 if (fsync(fd) != 0) {
  214. 214 close(fd);
  215. 215 throw std::runtime_error("Local recovery journal sync failed");
  216. 216 }
  217. 217 close(fd);
  218. 218 if (rename(temporary.c_str(), path.c_str()) != 0)
  219. 219 throw std::runtime_error("Local recovery journal commit failed");
  220. 220 const auto parent = std::filesystem::path(path).parent_path();
  221. 221 int directory_fd = open(parent.c_str(), O_RDONLY | O_DIRECTORY | O_CLOEXEC);
  222. 222 if (directory_fd < 0)
  223. 223 throw std::runtime_error("Local recovery directory could not be opened");
  224. 224 const auto synced = fsync(directory_fd);
  225. 225 close(directory_fd);
  226. 226 if (synced != 0)
  227. 227 throw std::runtime_error("Local recovery directory sync failed");
  228. 228}
