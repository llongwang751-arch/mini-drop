  193. 193void native_control(const RunnerContext &c, const std::string &operation,
  194. 194 const fs::path &output) {
  195. 195 const auto deadline = std::chrono::steady_clock::now() + 5s;
  196. 196 const auto root = fs::canonical(
  197. 197 secret("NATIVE_CONTROL_ROOT",
  198. 198 c.output_directory.empty()
  199. 199 ? "/run/drop-profilers"
  200. 200 : (fs::path(secret("COLLECTOR_HELPER_INCOMING_DIR",
  201. 201 "/run/drop-helper-incoming")) /
  202. 202 "native-control")
  203. 203 .string()));
  204. 204 std::uint64_t scratch_bound = 0;
  205. 205 if (!c.output_directory.empty()) {
  206. 206 scratch_bound =
  207. 207 std::stoull(secret("COLLECTOR_HELPER_DISK_BYTES", "2147483648"));
  208. 208 struct stat control {
  209. 209 }, output{};
  210. 210 if (stat(root.c_str(), &control) ||
  211. 211 stat(c.output_directory.c_str(), &output) ||
  212. 212 !S_ISDIR(control.st_mode) || control.st_uid != c.target.uid ||
  213. 213 (control.st_mode & 0077) || control.st_dev != output.st_dev ||
  214. 214 fs::space(root).capacity > scratch_bound)
  215. 215 throw std::runtime_error("Native profiler scratch must share the "
  216. 216 "protected bounded RAW filesystem");
  217. 217 }
  218. 218 // 协作程序用自身PID命名空间里的getpid()生成socket名称；
  219. 219 // 下方仍通过SO_PEERCRED核对控制器可见PID，不能混淆两个编号空间。
  220. 220 const auto helper_pid =
  221. 221 c.target.namespace_pid > 0 ? c.target.namespace_pid : c.target.pid;
  222. 222 const auto socket_path = root / (std::to_string(helper_pid) + ".sock");
  223. 223 struct stat info {};
  224. 224 if (lstat(socket_path.c_str(), &info) != 0 || !S_ISSOCK(info.st_mode) ||
  225. 225 info.st_uid != c.target.uid)
  226. 226 throw std::runtime_error(
  227. 227 "Native profiler control socket is missing or has the wrong owner");
  228. 228 int fd = socket(AF_UNIX, SOCK_STREAM | SOCK_CLOEXEC, 0);
  229. 229 if (fd < 0)
  230. 230 throw std::runtime_error("Native control socket failed");
  231. 231 struct Guard {
  232. 232 int fd;
  233. 233 ~Guard() { close(fd); }
  234. 234 } guard{fd};
  235. 235 timeval timeout{5, 0};
  236. 236 setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout));
  237. 237 setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &timeout, sizeof(timeout));
  238. 238 sockaddr_un address{};
  239. 239 address.sun_family = AF_UNIX;
  240. 240 const auto path = socket_path.string();
  241. 241 if (path.size() >= sizeof(address.sun_path))
  242. 242 throw std::runtime_error("Control root path too long");
  243. 243 std::copy(path.begin(), path.end(), address.sun_path);
  244. 244 if (connect(fd, reinterpret_cast<sockaddr *>(&address), sizeof(address)) != 0)
  245. 245 throw std::runtime_error(
  246. 246 "Native profiler target did not accept control connection");
  247. 247 struct ucred peer {};
  248. 248 socklen_t length = sizeof(peer);
  249. 249 if (getsockopt(fd, SOL_SOCKET, SO_PEERCRED, &peer, &length) != 0 ||
  250. 250 peer.pid != c.target.pid || peer.uid != c.target.uid)
  251. 251 throw std::runtime_error(
  252. 252 "Native profiler peer identity does not match selected target");
  253. 253 const auto request = json{{"schema_version", 2},
  254. 254 {"operation", operation},
  255. 255 {"scratch_filesystem_max_bytes", scratch_bound},
  256. 256 {"mode", c.kind == "GPERF_CPU" ? "cpu"
  257. 257 : c.kind == "GPERF_HEAP" ? "heap"
  258. 258 : "jemalloc"}}
  259. 259 .dump() +
  260. 260 "\n";
  261. 261 if (send(fd, request.data(), request.size(), MSG_NOSIGNAL) !=
  262. 262 static_cast<ssize_t>(request.size()))
  263. 263 throw std::runtime_error("Native profiler command failed");
  264. 264 std::string header;
  265. 265 char ch = 0;
  266. 266 while (header.size() < 128 && recv(fd, &ch, 1, 0) == 1 && ch != '\n')
  267. 267 header += ch;
  268. 268 if (header.rfind("OK2 ", 0) != 0)
  269. 269 throw std::runtime_error("Native target rejected profiling operation");
  270. 270 auto size = std::stoull(header.substr(4));
  271. 271 if (size > c.max_file_bytes)
  272. 272 throw std::runtime_error("Native profile exceeds task budget");
  273. 273 Guard file{-1};
  274. 274 if (!output.empty()) {
  275. 275 file.fd =
  276. 276 open(output.c_str(),
  277. 277 O_WRONLY | O_CREAT | O_NONBLOCK | O_NOFOLLOW | O_CLOEXEC, 0600);
  278. 278 struct stat status {};
  279. 279 if (file.fd < 0 || fstat(file.fd, &status) || !S_ISREG(status.st_mode) ||
  280. 280 status.st_uid != geteuid() || status.st_nlink != 1 ||
  281. 281 ftruncate(file.fd, 0))
  282. 282 throw std::runtime_error(
  283. 283 "Native RAW must be a private owned regular file");
  284. 284 }
  285. 285 std::array<char, 65536> bytes{};
  286. 286 std::uint64_t received = 0;
  287. 287 while (received < size) {
  288. 288 if (std::chrono::steady_clock::now() > deadline)
  289. 289 throw std::runtime_error("Native profile transfer deadline exceeded");
  290. 290 if (c.canceled() && operation != "stop")
  291. 291 throw std::runtime_error("TASK_CANCELED");
  292. 292 auto count =
  293. 293 recv(fd, bytes.data(),
  294. 294 std::min<std::uint64_t>(bytes.size(), size - received), 0);
  295. 295 if (count <= 0)
  296. 296 throw std::runtime_error("Truncated native profile response");
  297. 297 if (file.fd >= 0) {
  298. 298 std::size_t written = 0;
  299. 299 while (written < static_cast<std::size_t>(count)) {
  300. 300 const auto size = write(file.fd, bytes.data() + written,
  301. 301 static_cast<std::size_t>(count) - written);
  302. 302 if (size < 0 && errno == EINTR)
  303. 303 continue;
  304. 304 if (size <= 0)
  305. 305 throw std::runtime_error("Native profile write failed");
  306. 306 written += static_cast<std::size_t>(size);
  307. 307 }
  308. 308 }
  309. 309 received += static_cast<std::uint64_t>(count);
  310. 310 }
  311. 311 if (file.fd >= 0 && fsync(file.fd))
  312. 312 throw std::runtime_error("Native profile sync failed");
  313. 313}

  393. 393tracepoint:syscalls:sys_exit_munmap /pid == DROP_PID && @unmap_len[tid]/ { if(args->ret == 0) { printf("{\"schema_version\":1,\"event\":\"munmap\",\"time_ns\":%llu,\"pid\":%d,\"tid\":%d,\"address\":%llu,\"size\":%llu}\n",nsecs,pid,tid,@unmap_addr[tid],@unmap_len[tid]); @events=@events+1; } delete(@unmap_addr[tid]); delete(@unmap_len[tid]); }
  394. 394)BPF";
  395. 395 else
  396. 396 program =
  397. 397 R"BPF(software:page-faults:1 /pid == DROP_PID/ { printf("{\"schema_version\":1,\"event\":\"pagefault\",\"time_ns\":%llu,\"pid\":%d,\"tid\":%d}\n",nsecs,pid,tid); @events=@events+1; }
  398. 398)BPF";
  399. 399 }
  400. 400 // 每条事件写入前都检查预算，free事件也不能例外。
  401. 401 // 若只周期性检查退出条件，两次检查之间仍可能输出无界数据。
  402. 402 const std::string increment = " @events=@events+1;";
  403. 403 for (std::size_t at = 0;
  404. 404 (at = program.find(increment, at)) != std::string::npos;)
  405. 405 program.erase(at, increment.size());
  406. 406 for (std::size_t at = 0;
  407. 407 (at = program.find("printf(", at)) != std::string::npos;) {
  408. 408 const auto end = program.find(");", at);
  409. 409 if (end == std::string::npos)
  410. 410 throw std::logic_error("Invalid fixed trace event");
  411. 411 const auto statement = program.substr(at, end + 2 - at);
  412. 412 const auto bounded =
  413. 413 "if(@events < DROP_LIMIT) { " + statement + " @events=@events+1; }";
  414. 414 program.replace(at, statement.size(), bounded);
  415. 415 at += bounded.size();
  416. 416 }
  417. 417 // 一条printf记录同时携带JSON与真实用户栈，随后由有界解码器
  418. 418 // 把栈行转换成JSON数组；不能把bpftrace原始的
  419. 419 // 多行输出直接标记为合法JSONL。
  420. 420 const std::string marker = "}\\n\",";
  421. 421 for (std::size_t at = 0; (at = program.find(marker, at)) != std::string::npos;
  422. 422 at += 50)
  423. 423 program.replace(at, marker.size(),
  424. 424 "}\\nDROP_STACK_BEGIN\\n%s\\nDROP_STACK_END\\n\",");
  425. 425 for (std::size_t at = 0;
  426. 426 (at = program.find("printf(", at)) != std::string::npos;) {
  427. 427 const auto end = program.find(");", at);
  428. 428 if (end == std::string::npos)
  429. 429 throw std::logic_error("Invalid fixed trace template");
  430. 430 const auto stack =
  431. 431 c.kind == "IO_TRACE" &&
  432. 432 c.options.value("category", "syscalls") == "block"
  433. 433 ? ",@stack[$key]"
  434. 434 : ",ustack(perf,DROP_DEPTH)";
  435. 435 program.insert(end, stack);
  436. 436 at = end + std::strlen(stack) + 2;
  437. 437 }
  438. 438 // 排空异步用户栈记录前先停止事件生产；某些
  439. 439 // bpftrace版本在退出时仍保留multi-uprobe链接，若没有
  440. 440 // 持续生效的停止门控，新事件会不断产生，导致收尾永远无法完成。
  441. 441 for (std::size_t at = 0;
  442. 442 (at = program.find(" /", at)) != std::string::npos;) {
  443. 443 const auto end = program.find("/ {", at + 2);
  444. 444 if (end == std::string::npos || program.find('\n', at) < end)
  445. 445 throw std::logic_error("Invalid fixed trace predicate");
  446. 446 program.insert(end, ")");
  447. 447 program.insert(at + 2, "!@drop_done && (");
  448. 448 at = end + 19;
  449. 449 }
  450. 450 program +=
  451. 451 R"BPF(interval:s:1 /@events >= DROP_LIMIT/ { @drop_done=1; exit(); } interval:s:DROP_DURATION { @drop_done=1; exit(); } END { @drop_done=1; clear(@events); )BPF";
  452. 452 if (c.kind == "IO_TRACE") {
  453. 453 const auto category = c.options.value("category", "syscalls");
  454. 454 if (category == "block")
  455. 455 program += "clear(@begin); clear(@owner); clear(@thread); clear(@bytes); "
  456. 456 "clear(@sector); clear(@major); clear(@minor); clear(@stack);";
  457. 457 else {
  458. 458 program += "clear(@start); clear(@op);";
  459. 459 if (category == "vfs")
  460. 460 program += "clear(@inode); clear(@dev);";
  461. 461 }
  462. 462 } else if (c.options.value("category", "pagefault") == "alloc")
  463. 463 program += "clear(@size);";
  464. 464 else if (c.options.value("category", "pagefault") == "mmap")
  465. 465 program += "clear(@len); clear(@unmap_addr); clear(@unmap_len);";
  466. 466 program += " }";
  467. 467 if (!c.target.pid_namespace.empty()) {
  468. 468 const auto inode =
  469. 469 c.target.pid_namespace.substr(5, c.target.pid_namespace.size() - 6);
  470. 470 if (inode.empty() ||
  471. 471 inode.find_first_not_of("0123456789") != std::string::npos)
  472. 472 throw std::invalid_argument("Invalid pinned PID namespace identity");
  473. 473 const std::string
  474. 474 leader = "((struct task_struct*)curtask)->group_leader->thread_pid",
  475. 475 thread = "((struct task_struct*)curtask)->thread_pid";
  476. 476 // bpftrace 0.20只支持字面量数组下标；通过BTF中的upid
  477. 477 // 偏移确定真实命名空间层级，而不是猜测宿主嵌套深度。
  478. 478 const auto upid = [](const std::string &identity) {
  479. 479 return "((struct upid*)((uint64)" + identity +
  480. 480 " + offsetof(struct pid,numbers) + (uint64)" + identity +
  481. 481 "->level * sizeof(struct upid)))";
  482. 482 };
  483. 483 const auto number = upid(leader), thread_number = upid(thread);
  484. 484 const std::map<std::string, std::string> local{
  485. 485 {"pid == DROP_PID", "((struct pid_namespace*)" + number +
  486. 486 "->ns)->ns.inum == " + inode + " && " + number +
  487. 487 "->nr == DROP_NAMESPACE_PID"},
  488. 488 {"nsecs,pid,tid,", "nsecs,DROP_PID," + thread_number + "->nr,"},
  489. 489 {"nsecs,pid,tid)", "nsecs,DROP_PID," + thread_number + "->nr)"},
  490. 490 {"@owner[$key]=pid; @thread[$key]=tid;",
  491. 491 "@owner[$key]=DROP_PID; @thread[$key]=" + thread_number + "->nr;"}};
  492. 492 for (const auto &[key, value] : local)
  493. 493 for (std::size_t at = 0;
  494. 494 (at = program.find(key, at)) != std::string::npos;
  495. 495 at += value.size())
  496. 496 program.replace(at, key.size(), value);
  497. 497 }
  498. 498 const std::map<std::string, std::string> replacements{
  499. 499 {"DROP_PID", std::to_string(c.target.pid)},
  500. 500 {"DROP_NAMESPACE_PID",
  501. 501 std::to_string(c.target.namespace_pid > 0 ? c.target.namespace_pid
  502. 502 : c.target.pid)},
  503. 503 {"DROP_MINIMUM",
  504. 504 std::to_string(c.options.value("min_duration_us", 0LL) * 1000)},
  505. 505 {"DROP_LIMIT", std::to_string(c.options.value("max_events", 10000))},
  506. 506 {"DROP_DEPTH", std::to_string(c.options.value("stack_depth", 32))},
  507. 507 {"DROP_DURATION", std::to_string(c.duration_seconds)}};
  508. 508 for (const auto &[key, value] : replacements)
  509. 509 for (std::size_t at = 0; (at = program.find(key, at)) != std::string::npos;
  510. 510 at += value.size())
  511. 511 program.replace(at, key.size(), value);
  512. 512 return program;
  513. 513}

  529. 529 };
  530. 530 const auto output = ((c.output_directory.empty() ? c.directory

  567. 567 "--method", "gdb",
  568. 568 "--output", output,
  569. 569 "--duration", std::to_string(c.duration_seconds)};

  571. 571 result.push_back("--native");
  572. 572 result.push_back(pid);
  573. 573 return result;
  574. 574 }
  575. 575 if (c.kind == "JAVA_HEAP") {
  576. 576 // 即使PID命名空间共享，原生jattach仍能解析目标的挂载命名空间；
  577. 577 // JDK17的jcmd在这种拓扑下却会假设/tmp共享，因此这里不能混用。
  578. 578 // 诊断命令还有独立的分词规则，只允许程序持有的简单路径，
  579. 579 // 调用方不能追加第二个参数或命令来扩大执行范围。

  581. 581 output.find_first_not_of(
  582. 582 "/._-"
  583. 583 "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ") !=
  584. 584 std::string::npos)

  587. 587 "GC.heap_dump " +
  588. 588 std::string(c.options.value("live", true) ? "" : "-all ") +
  589. 589 output};
  590. 590 }
  591. 591 if (c.kind == "IO_TRACE" || c.kind == "MEMORY_TRACE") {
  592. 592 if (c.kind == "MEMORY_TRACE" &&
  593. 593 c.options.value("category", "pagefault") == "alloc")
  594. 594 return {tool("bpftrace"), "-q", "-p", pid, "-e", trace_program(c)};
  595. 595 return {tool("bpftrace"), "-q", "-e", trace_program(c)};
  596. 596 }
  597. 597 if (c.kind == "SCRIPT") {
  598. 598 const auto id = c.options.value("template_id", "process_status");
  599. 599 if (id != "process_status" && id != "process_limits" &&
  600. 600 id != "open_files" && id != "mapped_libraries" &&
  601. 601 id != "network_counters")
  602. 602 throw std::invalid_argument("Unknown script template");
  603. 603 return {};
  604. 604 }
  605. 605 if (c.kind == "BOLT_PROFILE")
  606. 606 return {tool("perf2bolt"),
  607. 607 "/proc/" + pid + "/exe",
  608. 608 "-p",
  609. 609 (c.directory / "bolt.perf.data").string(),
  610. 610 "-o",
  611. 611 output,
  612. 612 "-nl",
  613. 613 "-aggregate-only"};
  614. 614 if (c.kind == "RESOURCE_SNAPSHOT" || c.kind == "GO_CPU" ||
  615. 615 c.kind == "GO_HEAP" || c.kind == "GPERF_CPU" || c.kind == "GPERF_HEAP" ||
  616. 616 c.kind == "JEPROF_HEAP")
  617. 617 return {};
  618. 618 throw std::invalid_argument("Unsupported external runner");
  619. 619}

  621. 621RunnerRegistry::RunnerRegistry(const Catalog &catalog, std::string scratch,
  622. 622 std::string perf)
  623. 623 : catalog_(catalog) {
  624. 624 for (const auto &[kind, spec] : catalog.entries()) {
  625. 625 if (spec.value("agent_runner", true) == false ||
  626. 626 spec.value("runner", "") == "import")
  627. 627 continue;
  628. 628 std::string reason;
  629. 629 json versions = json::array();
  630. 630 for (const auto &requirement :
  631. 631 spec.value("tool_requirements", json::array())) {
  632. 632 const auto name = requirement.at("name").get<std::string>();
  633. 633 if (!tools_.contains(name)) {
  634. 634 ToolInfo info;
  635. 635 info.name = name;
  636. 636 info.path =
  637. 637 name == "perf"
  638. 638 ? perf
  639. 639 : secret(requirement.at("executable_env")
  640. 640 .get<std::string>()
  641. 641 .c_str(),
  642. 642 requirement.at("default_path").get<std::string>());
  643. 643 try {
  644. 644 // 保留argv[0]别名：LLVM的perf2bolt可能是多功能二进制的软链接，
  645. 645 // 执行模式依赖配置名称。校验解析后的真实文件哈希，
  646. 646 // 执行时仍使用配置中的绝对别名，避免工具进入错误模式。
  647. 647 info.path = fs::absolute(info.path).lexically_normal().string();
  648. 648 const auto resolved = fs::canonical(info.path);
  649. 649 for (auto parent = resolved.parent_path(); !parent.empty();
  650. 650 parent = parent.parent_path()) {
  651. 651 struct stat directory {};
  652. 652 if (::stat(parent.c_str(), &directory) != 0 ||
  653. 653 (directory.st_mode & 0022) ||
  654. 654 (directory.st_uid != 0 && directory.st_uid != getuid()))
  655. 655 throw std::runtime_error("Tool resolved parent directory is "
  656. 656 "writable by an untrusted principal");
  657. 657 if (parent == parent.root_path())
  658. 658 break;
  659. 659 }
  660. 660 for (auto parent = fs::path(info.path).parent_path(); !parent.empty();
  661. 661 parent = parent.parent_path()) {
  662. 662 struct stat directory {};
  663. 663 if (::stat(parent.c_str(), &directory) != 0 ||
  664. 664 (directory.st_mode & 0022) ||
  665. 665 (directory.st_uid != 0 && directory.st_uid != getuid()))
  666. 666 throw std::runtime_error("Tool parent directory is writable by "
  667. 667 "an untrusted principal");
  668. 668 if (parent == parent.root_path())
  669. 669 break;
  670. 670 }
  671. 671 struct stat stat {};
  672. 672 if (::stat(info.path.c_str(), &stat) != 0 || !S_ISREG(stat.st_mode) ||
  673. 673 (stat.st_mode & 0022) || access(info.path.c_str(), X_OK) != 0)
  674. 674 throw std::runtime_error(
  675. 675 "Tool is missing, nonregular, nonexecutable or writable by an "
  676. 676 "untrusted principal");
  677. 677 info.sha256 = sha256_file(info.path);
  678. 678 auto expected =
  679. 679 secret(requirement.at("sha256_env").get<std::string>().c_str());
  680. 680 const auto manifest =
  681. 681 secret("TOOL_MANIFEST_FILE", "/opt/drop-tools/manifest.json");
  682. 682 if (fs::is_regular_file(manifest)) {
  683. 683 const auto pinned = parse_json(read_file(manifest, 1024 * 1024));
  684. 684 if (pinned.at("tools").contains(name)) {
  685. 685 const auto &entry = pinned.at("tools").at(name);
  686. 686 if (expected.empty())
  687. 687 expected = entry.at("sha256").get<std::string>();
  688. 688 info.support_files = entry.value(
  689. 689 "support_files", std::map<std::string, std::string>{});
  690. 690 }
  691. 691 }
  692. 692 if (expected.empty())
  693. 693 throw std::runtime_error(
  694. 694 "No trusted tool hash configured; set SHA256 or install the "
  695. 695 "trusted build tool manifest");
  696. 696 for (const auto &[path, hash] : info.support_files)
  697. 697 if (sha256_file(path) != hash)
  698. 698 throw std::runtime_error("Tool support library SHA256 mismatch");
  699. 699 if (name == "gdb") {
  700. 700 const std::string launcher = "/usr/local/bin/drop_gdb_ptrace";
  701. 701 const std::string alias = "/opt/drop-tools/gdb-bin/gdb";
  702. 702 if (!info.support_files.contains(launcher) ||
  703. 703 !info.support_files.contains(alias) ||
  704. 704 fs::canonical(alias) != fs::canonical(launcher) ||
  705. 705 parse_json(read_file(manifest, 1024 * 1024))
  706. 706 .at("tools")
  707. 707 .at("gdb")
  708. 708 .at("path") != info.path)
  709. 709 throw std::runtime_error("GDB requires its pinned ptrace-only "
  710. 710 "launcher and matching manifest path");
  711. 711 }
  712. 712 if (!expected.empty() && expected != info.sha256)
  713. 713 throw std::runtime_error(
  714. 714 "Tool SHA256 differs from configured allowlist");
  715. 715 std::vector<std::string> argv{info.path};
  716. 716 for (const auto &argument : requirement.at("probe_argv"))
  717. 717 argv.push_back(argument.get<std::string>());
  718. 718 // I/O受限节点上，LLVM/JVM即使执行--version也可能初始化超过5秒；
  719. 719 // 仍设置硬时限，但不能仅因为共享库冷启动就禁用
  720. 720 // 原本健康的采集工具。
  721. 721 const auto probe_timeout =
  722. 722 name == "bpftrace" || name == "jcmd" || name == "memray" ? 15s
  723. 723 : 5s;
  724. 724 auto probe =
  725. 725 execute(argv, scratch + "/tool-" + name + ".log", probe_timeout);
  726. 726 if (probe.exit_code != 0)
  727. 727 throw std::runtime_error(probe.timed_out
  728. 728 ? "Tool version probe timed out"
  729. 729 : "Tool version probe failed");
  730. 730 info.version = probe.output.substr(0, 1024);
  731. 731 info.available = true;
  732. 732 } catch (const std::exception &error) {
  733. 733 info.reason = error.what();
  734. 734 }
  735. 735 tools_.emplace(name, std::move(info));
  736. 736 }
  737. 737 const auto &info = tools_.at(name);
  738. 738 versions.push_back({{"name", name},
  739. 739 {"path", info.path},
  740. 740 {"sha256", info.sha256},
  741. 741 {"version", info.version},
  742. 742 {"available", info.available},
  743. 743 {"reason", info.reason}});
  744. 744 if (!info.available)
  745. 745 reason = "Required tool unavailable: " + name;
  746. 746 }
  747. 747 if ((kind == "PYTHON_CPU" || kind == "PYTHON_MEMORY") && !capability(19)) {
  748. 748 try {
  749. 749 if (std::stoi(read_file("/proc/sys/kernel/yama/ptrace_scope", 32)) != 0)
  750. 750 reason = "SYS_PTRACE or permissive ptrace policy is required for "
  751. 751 "attaching";
  752. 752 } catch (const std::exception &) {
  753. 753 reason = "Cannot verify ptrace permission";
  754. 754 }
  755. 755 }
  756. 756 if ((kind == "IO_TRACE" || kind == "MEMORY_TRACE") &&
  757. 757 (!capability(39) || !capability(38)))
  758. 758 reason = "CAP_BPF and CAP_PERFMON are required by this trace runner";
  759. 759 capabilities_[kind] = {{"available", reason.empty()},
  760. 760 {"reason", reason},
  761. 761 {"tools", versions},
  762. 762 {"target_probe_required", true},
  763. 763 {"runner", spec.at("runner")}};
  764. 764 if (kind == "IO_TRACE" || kind == "MEMORY_TRACE") {
  765. 765 auto &categories = capabilities_[kind]["category_capabilities"];
  766. 766 categories = json::object();
  767. 767 for (const auto &category :
  768. 768 kind == "IO_TRACE"
  769. 769 ? std::vector<std::string>{"syscalls", "vfs", "block"}
  770. 770 : std::vector<std::string>{"alloc", "mmap", "pagefault"})
  771. 771 categories[category] = {
  772. 772 {"available", reason.empty() ? json(nullptr) : json(false)},
  773. 773 {"status",
  774. 774 reason.empty() ? "target_probe_required" : "unavailable"},
  775. 775 {"reason",
  776. 776 reason.empty()
  777. 777 ? "Kernel attachment must be verified for the selected target"
  778. 778 : reason}};
  779. 779 }
  780. 780 runners_[kind] = [this] {
  781. 781 return make_managed_runner(
  782. 782 [this](const RunnerContext &c) {
  783. 783 catalog_.validate(c.kind, c.parameters);
  784. 784 if (c.target.pid &&
  785. 785 !same_identity(c.target, process_identity(c.target.pid)))
  786. 786 throw std::runtime_error("TARGET_CHANGED");
  787. 787 if (!available(c.kind))
  788. 788 throw std::runtime_error(
  789. 789 "Required profiler capability is not available");
  790. 790 // GDB需要暂停目标新建线程，但具备ptrace权限
  791. 791 // 并不等于具备跨UID发送信号的权限。信号0
  792. 792 // 只做权限检查，不暂停或改变目标任务。
  793. 793 if (c.kind == "PYTHON_MEMORY" && c.target.pid > 1 &&
  794. 794 c.target.uid != geteuid() && ::kill(c.target.pid, 0) != 0)
  795. 795 throw ProfilerExecutionFailure(1, "PROFILER_PERMISSION_DENIED");
  796. 796 },
  797. 797 [this](const RunnerContext &c) {
