  202. 202 throw std::invalid_argument("Invalid resources");
  203. 203 std::lock_guard lock(db_.mutex);
  204. 204 Transaction transaction(db_);
  205. 205 auto found = db_.exec(
  206. 206 "UPDATE agents SET "
  207. 207 "last_seen_at=now(),status='online',cpu_percent=$2::float8,memory_"
  208. 208 "percent=$3::float8,load_average=$4::float8,resources=$5::jsonb "
  209. 209 "WHERE id=$1 AND status<>'disabled' RETURNING id",
  210. 210 {request->agent_id(), std::to_string(request->cpu_percent()),
  211. 211 std::to_string(request->memory_percent()),
  212. 212 std::to_string(request->load_average()), resources.dump()});
  213. 213 if (!found.count())
  214. 214 throw grpc::Status(grpc::StatusCode::PERMISSION_DENIED,
  215. 215 "Agent is not registered or is disabled");
  216. 216 db_.exec("INSERT INTO "
  217. 217 "agent_samples(agent_id,cpu_percent,memory_percent,load_average)"
  218. 218 " VALUES($1,$2::float8,$3::float8,$4::float8)",
  219. 219 {request->agent_id(), std::to_string(request->cpu_percent()),
  220. 220 std::to_string(request->memory_percent()),
  221. 221 std::to_string(request->load_average())});
  222. 222 if (request->events_size() > 32)
  223. 223 throw std::invalid_argument("Too many heartbeat events");
  224. 224 for (const auto &item : request->events())
  225. 225 store_agent_event(item, request->agent_id());
  226. 226 for (auto attempt : request->running_attempt_ids()) {
  227. 227 auto owner = db_.exec("SELECT task_id FROM task_attempts WHERE "
  228. 228 "id=$1::bigint AND agent_id=$2",
  229. 229 {std::to_string(attempt), request->agent_id()});
  230. 230 if (!owner.count()) {
  231. 231 response->add_cancel_attempt_ids(attempt);
  232. 232 continue;
  233. 233 }
  234. 234 // 先锁任务，再用新的SQL语句读取最新attempt，保证读到锁等待之后的状态。

  237. 237 {owner.get(0, "task_id")});
  238. 238 auto rows =
  239. 239 db_.exec("SELECT t.id,t.collection_status,a.id AS attempt_id FROM "
  240. 240 "task_attempts a JOIN tasks t ON t.id=a.task_id WHERE "
  241. 241 "a.id=$1::bigint AND a.agent_id=$2 AND a.attempt=(SELECT "
  242. 242 "max(x.attempt) FROM task_attempts x WHERE "
  243. 243 "x.task_id=t.id) FOR UPDATE OF t,a",
  244. 244 {std::to_string(attempt), request->agent_id()});
  245. 245 if (!rows.count()) {
  246. 246 response->add_cancel_attempt_ids(attempt);
  247. 247 continue;
  248. 248 }
  249. 249 const auto state = rows.get(0, "collection_status");
  250. 250 if (state == "CANCELED" || state == "FAILED" || state == "COLLECTED")
  251. 251 response->add_cancel_attempt_ids(attempt);
  252. 252 else
  253. 253 db_.exec("UPDATE tasks SET updated_at=now() WHERE id=$1::bigint",
  254. 254 {rows.get(0, "id")});
  255. 255 }
  256. 256 if (request->running_attempt_ids().empty())
  257. 257 dispatch(request->agent_id(), response);
  258. 258 transaction.commit();
  259. 259 trace_dispatched(*response, started);
  260. 260 metrics_.add("drop_heartbeats_total");
  261. 261 });
  262. 262 }

  753. 753 void store_log_artifacts(const Rows &rows, const dv::ResultRequest &request) {
  754. 754 if (request.artifacts_size() > 4)
  755. 755 throw std::invalid_argument("Too many auxiliary artifacts");
  756. 756 const auto grants = parse_json(rows.get(0, "upload_targets"));
  757. 757 const auto state = rows.get(0, "collection_status");
  758. 758 const bool terminal =
  759. 759 state == "COLLECTED" || state == "CANCELED" || state == "FAILED";
  760. 760 const auto expected_key = "raw/tasks/" + rows.get(0, "id") + "/attempts/" +
  761. 761 rows.get(0, "attempt_number") + "/runner.log";
  762. 762 bool seen = false;
  763. 763 for (const auto &artifact : request.artifacts()) {
  764. 764 if (artifact.kind() != "LOG")
  765. 765 continue;
  766. 766 if (seen)
  767. 767 throw std::invalid_argument("Duplicate diagnostic artifact");
  768. 768 seen = true;
  769. 769 if (artifact.name() != "runner.log" || artifact.format() != "log_jsonl" ||
  770. 770 artifact.content_type() != "text/plain" ||
  771. 771 artifact.object_key() != expected_key || artifact.size() <= 0 ||
  772. 772 artifact.size() > 16384 || artifact.sha256().size() != 64 ||
  773. 773 artifact.sha256().find_first_not_of("0123456789abcdef") !=
  774. 774 std::string::npos)
  775. 775 throw std::invalid_argument("Invalid bounded diagnostic manifest");
  776. 776 const auto existing =
  777. 777 db_.exec("SELECT sha256,size FROM artifacts WHERE task_id=$1::bigint "
  778. 778 "AND attempt_id=$2::bigint AND object_key=$3 AND kind='LOG'",
  779. 779 {rows.get(0, "id"), std::to_string(request.attempt_id()),
  780. 780 artifact.object_key()});
  781. 781 if (existing.count()) {
  782. 782 if (existing.get(0, "sha256") != artifact.sha256() ||
  783. 783 std::stoll(existing.get(0, "size")) != artifact.size())
  784. 784 throw std::invalid_argument(
  785. 785 "Diagnostic replay differs from stored evidence");
  786. 786 continue;
  787. 787 }
  788. 788 if (terminal) {
  789. 789 const auto at = rows.get(0, "terminal_at_ms");
  790. 790 const auto now =
  791. 791 std::chrono::duration_cast<std::chrono::milliseconds>(
  792. 792 std::chrono::system_clock::now().time_since_epoch())
  793. 793 .count();
  794. 794 if (at.empty() || now > std::stoll(at) + 120000)
  795. 795 throw std::invalid_argument(
  796. 796 "Terminal diagnostic upload grace expired");
  797. 797 }
  798. 798 bool allowed = false;
  799. 799 for (const auto &grant : grants)
  800. 800 if (grant.value("kind", "") == "LOG" &&
  801. 801 grant.value("name", "") == "runner.log" &&
  802. 802 grant.value("object_key", "") == artifact.object_key() &&
  803. 803 grant.value("max_file_bytes", 0LL) == 16384 &&
  804. 804 grant.value("format", "") == "log_jsonl" &&
  805. 805 grant.value("content_type", "") == "text/plain")
  806. 806 allowed = true;
  807. 807 if (!allowed)
  808. 808 throw std::invalid_argument("Diagnostic lacks current task grant");
  809. 809 store_artifact(rows.get(0, "id"), std::to_string(request.attempt_id()),
  810. 810 artifact, "LOG", "log_jsonl");
  811. 811 event(db_, rows.get(0, "id"), std::to_string(request.attempt_id()),
  812. 812 "LOG_ARTIFACT_UPLOADED",
  813. 813 {{"size", artifact.size()}, {"sha256", artifact.sha256()}});
  814. 814 }

  860. 860 } catch (...) {
  861. 861 // 追踪上报属于尽力而为的旁路，不能改变已经提交的RPC
  862. 862 // 结果，更不能因为遥测失败让调用方误以为业务事务未提交。
  863. 863 }
  864. 864 }
  865. 865 void trace_dispatched(const dv::HeartbeatResponse &response,
  866. 866 std::int64_t started) {
  867. 867 for (const auto &task : response.tasks())
  868. 868 trace_committed("dispatch.task", task.request_id(), task.task_id(),
  869. 869 started, task.traceparent());
  870. 870 }
  871. 871 // 为一个Agent串行分配任务，再按优先级与先入先出选择候选。
  872. 872 // Task的SKIP LOCKED避免争抢同一任务，Agent行锁另外约束节点总并发。
  873. 873 void dispatch(const std::string &agent, dv::HeartbeatResponse *response) {
  874. 874 // 每个任务下发入口都必须锁同一Agent行。心跳路径已持有该锁，
  875. 875 // 但直接Collect可能进入另一个Server副本；只对Task使用
  876. 876 // SKIP LOCKED会让两个副本分别领到不同任务，突破同节点并发限制。
  877. 877 auto registered = db_.exec(
  878. 878 "SELECT id,labels FROM agents WHERE id=$1 AND status='online' AND "
  879. 879 "last_seen_at>now()-interval '15 seconds' FOR UPDATE",
  880. 880 {agent});
  881. 881 if (!registered.count())
  882. 882 return;
  883. 883 auto active = db_.exec(
  884. 884 "SELECT id FROM tasks WHERE target_agent_id=$1 AND collection_status "
  885. 885 "IN ('DELIVERED','RUNNING','UPLOADING') LIMIT 1",
  886. 886 {agent});
  887. 887 if (active.count())
  888. 888 return;
  889. 889 auto rows = db_.exec(
  890. 890 "SELECT t.id,t.kind,t.params,t.priority,t.request_id,(extract(epoch "
  891. 891 "FROM "
  892. 892 "t.collection_deadline)*1000)::bigint AS deadline,a.id AS "
  893. 893 "attempt_id,a.raw_upload_url,a.raw_object_key,a.upload_targets FROM "
  894. 894 "tasks t JOIN task_attempts a ON a.task_id=t.id AND a.attempt=(SELECT "
  895. 895 "max(x.attempt) FROM task_attempts x WHERE x.task_id=t.id) WHERE "
  896. 896 "t.target_agent_id=$1 AND t.collection_status='QUEUED' AND "
  897. 897 "t.collection_deadline>now() AND a.raw_upload_url<>'' ORDER BY "
  898. 898 "t.priority DESC,t.created_at,t.id LIMIT 1 FOR UPDATE OF t,a SKIP "
  899. 899 "LOCKED",
  900. 900 {agent});
  901. 901 if (!rows.count())
  902. 902 return;
  903. 903 const auto task = rows.get(0, "id"), attempt = rows.get(0, "attempt_id");
  904. 904 auto parameters = parse_json(rows.get(0, "params"));
  905. 905 auto *desc = response->add_tasks();
  906. 906 desc->set_task_id(std::stoll(task));
  907. 907 desc->set_attempt_id(std::stoll(attempt));
  908. 908 desc->set_priority(std::stoi(rows.get(0, "priority")));
  909. 909 desc->set_kind(rows.get(0, "kind"));
  910. 910 desc->set_target_agent_id(agent);
  911. 911 desc->set_deadline_unix_ms(std::stoll(rows.get(0, "deadline")));
  912. 912 desc->set_raw_upload_url(rows.get(0, "raw_upload_url"));
  913. 913 desc->set_raw_object_key(rows.get(0, "raw_object_key"));
  914. 914 const auto &spec = catalog_.get(rows.get(0, "kind"));
  915. 915 catalog_.validate(rows.get(0, "kind"), parameters);
  916. 916 const auto defaults = spec;
  917. 917 const auto options = parameters.value("options", json::object());
  918. 918 const auto budget = parameters.value("budget", json::object());
  919. 919 desc->set_request_id(rows.get(0, "request_id"));
  920. 920 desc->set_max_file_bytes(
  921. 921 std::min(budget.value("max_file_bytes",
  922. 922 spec.at("max_file_bytes").get<std::int64_t>()),
  923. 923 spec.at("max_file_bytes").get<std::int64_t>()));
  924. 924 desc->set_protocol_version(2);
  925. 925 desc->set_extra_parameters_json(options.dump());
  926. 926 desc->set_raw_name(spec.at("raw_name"));
  927. 927 desc->set_raw_format(spec.at("input_format"));
  928. 928 desc->set_raw_content_type(spec.at("raw_content_type"));
  929. 929 desc->set_analysis_pipeline(spec.at("analysis_pipeline"));
  930. 930 desc->set_traceparent(parameters.value("traceparent", ""));
  931. 931 auto target =
  932. 932 parameters.value("target", json{{"type", "process"},
  933. 933 {"pid", parameters.value("pid", 0)}});
  934. 934 auto *typed = desc->mutable_target();
  935. 935 typed->set_type(target.value("type", "process"));
  936. 936 typed->set_pid(target.value("pid", parameters.value("pid", 0)));
  937. 937 typed->set_start_time_ticks(target.value("start_time_ticks", 0ULL));
  938. 938 typed->set_container_id(target.value("container_id", ""));
  939. 939 typed->set_runtime(target.value("runtime", "docker"));
  940. 940 typed->set_container_pid(target.value("container_pid", 0));
  941. 941 typed->set_expected_uid(target.value("expected_uid", 0U));
  942. 942 typed->set_check_uid(target.value("check_uid", false));
  943. 943 auto *limits = desc->mutable_budget();
  944. 944 limits->set_max_file_bytes(desc->max_file_bytes());
  945. 945 limits->set_max_address_space_bytes(
  946. 946 budget.value("max_address_space_bytes", 8LL * 1024 * 1024 * 1024));
  947. 947 limits->set_max_cpu_seconds(budget.value("max_cpu_seconds", 120));
  948. 948 limits->set_max_processes(budget.value("max_processes", 32));
  949. 949 limits->set_max_concurrency(1);
  950. 950 auto granted_targets = parse_json(rows.get(0, "upload_targets"));
  951. 951 if (parse_json(registered.get(0, "labels"))
  952. 952 .value("collection_manifest_version", 0) != 1) {
  953. 953 for (auto item = granted_targets.begin(); item != granted_targets.end();)
  954. 954 if (item->value("kind", "") == "MANIFEST")
  955. 955 item = granted_targets.erase(item);
  956. 956 else
  957. 957 ++item;
  958. 958 db_.exec("UPDATE task_attempts SET upload_targets=$2::jsonb WHERE "
  959. 959 "id=$1::bigint",
  960. 960 {attempt, granted_targets.dump()});
  961. 961 }
  962. 962 for (const auto &grant : granted_targets) {
  963. 963 auto *upload = desc->add_upload_targets();
  964. 964 upload->set_name(grant.at("name"));
  965. 965 upload->set_object_key(grant.at("object_key"));
  966. 966 upload->set_put_url(grant.at("put_url"));
  967. 967 upload->set_kind(grant.at("kind"));
  968. 968 upload->set_max_file_bytes(grant.at("max_file_bytes"));
  969. 969 }
  970. 970 auto *params = desc->mutable_parameters();
  971. 971 params->set_pid(parameters.value("pid", typed->pid()));
  972. 972 params->set_duration_seconds(parameters.value(
  973. 973 "duration_seconds", defaults.value("default_duration_seconds", 10)));
  974. 974 params->set_frequency_hz(parameters.value(
  975. 975 "frequency_hz", defaults.value("default_frequency_hz", 99)));
  976. 976 params->set_scope(parameters.value("scope", "process"));
  977. 977 const auto kind = rows.get(0, "kind");
  978. 978 if (kind == "PERF_CPU") {
  979. 979 desc->mutable_perf()->set_event(options.value("event", "cpu-clock"));
  980. 980 desc->mutable_perf()->set_call_graph(options.value("call_graph", "fp"));
  981. 981 } else if (kind == "JAVA_CPU") {
  982. 982 desc->mutable_java_profiler()->set_event(options.value("event", "cpu"));
  983. 983 desc->mutable_java_profiler()->set_output_format("collapsed");
  984. 984 } else if (kind == "GO_CPU" || kind == "GO_HEAP") {
  985. 985 desc->mutable_go_profile()->set_port(options.value("port", 6060));
  986. 986 desc->mutable_go_profile()->set_path(
  987. 987 options.value("path", kind == "GO_CPU" ? "/debug/pprof/profile"
  988. 988 : "/debug/pprof/heap"));
  989. 989 desc->mutable_go_profile()->set_profile_type(kind == "GO_CPU" ? "cpu"
  990. 990 : "heap");
  991. 991 } else if (kind == "PYTHON_CPU" || kind == "PYTHON_MEMORY") {
  992. 992 desc->mutable_python_profiler()->set_mode(
  993. 993 kind == "PYTHON_CPU" ? "cpu" : "memory");
  994. 994 desc->mutable_python_profiler()->set_native(
  995. 995 options.value("native", false));
  996. 996 } else if (kind == "JAVA_HEAP") {
  997. 997 desc->mutable_heap()->set_format("hprof");
  998. 998 desc->mutable_heap()->set_live(options.value("live", true));
  999. 999 } else if (kind == "IO_TRACE" || kind == "MEMORY_TRACE") {
  1000. 1000 desc->mutable_trace()->set_category(options.value(
  1001. 1001 "category", kind == "IO_TRACE" ? "read_write" : "alloc"));
  1002. 1002 desc->mutable_trace()->set_min_duration_us(
  1003. 1003 options.value("min_duration_us", 0));
  1004. 1004 } else if (kind == "GPERF_CPU" || kind == "GPERF_HEAP" ||
  1005. 1005 kind == "JEPROF_HEAP") {
  1006. 1006 desc->mutable_allocator()->set_mode(kind == "GPERF_CPU" ? "cpu"
  1007. 1007 : kind == "GPERF_HEAP" ? "heap"
  1008. 1008 : "jemalloc");
  1009. 1009 } else if (kind == "BOLT_PROFILE") {
  1010. 1010 desc->mutable_bolt()->set_binary_path(
  1011. 1011 "/proc/" + std::to_string(typed->pid()) + "/exe");
  1012. 1012 desc->mutable_bolt()->set_optimize(options.value("optimize", false));
  1013. 1013 } else if (kind == "SCRIPT")
  1014. 1014 desc->mutable_script()->set_template_id(
  1015. 1015 options.value("template_id", "process_status"));
  1016. 1016 else
  1017. 1017 desc->mutable_native()->set_category(kind);
  1018. 1018 db_.exec("UPDATE tasks SET "
  1019. 1019 "collection_status='DELIVERED',updated_at=now(),version=version+1 "
  1020. 1020 "WHERE id=$1::bigint",
  1021. 1021 {task});
  1022. 1022 db_.exec("UPDATE task_attempts SET status='DELIVERED' WHERE id=$1::bigint",
  1023. 1023 {attempt});
  1024. 1024 event(db_, task, attempt, "TASK_DELIVERED");
  1025. 1025 metrics_.add("drop_dispatch_total");
  1026. 1026 }
  1027. 1027 Database &db_;
  1028. 1028 Catalog catalog_;
  1029. 1029 Telemetry metrics_;
  1030. 1030};
