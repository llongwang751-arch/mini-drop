  533. 533 }
  534. 534 }

  711. 711 UploadReceipt upload_retry(const fs::path &directory,
  712. 712 const dv::TaskDesc &task,
  713. 713 const dv::ArtifactInfo &artifact,
  714. 714 const std::string &url, std::uint64_t maximum,
  715. 715 bool renew, unsigned max_attempts = 0) {
  716. 716 const auto file = artifact_path(directory, artifact);
  717. 717 if (artifact.size() <= 0 ||
  718. 718 fs::file_size(file) != static_cast<std::uint64_t>(artifact.size()) ||
  719. 719 sha256_file(file.string()) != artifact.sha256())
  720. 720 throw UploadFailure(false);
  721. 721 const auto marker = directory / ("uploaded-" + artifact.name());
  722. 722 const auto receipt_file =
  723. 723 directory / ("receipt-" + artifact.name() + ".json");
  724. 724 const auto saved_receipt = [&]() -> UploadReceipt {
  725. 725 if (!fs::is_regular_file(receipt_file))
  726. 726 return {};
  727. 727 const auto value = parse_json(read_file(receipt_file.string(), 4096));
  728. 728 if (value.value("sha256", "") != artifact.sha256())
  729. 729 throw UploadFailure(false);
  730. 730 return {value.value("etag", ""),
  731. 731 value.value("completed_unix_ms", std::int64_t{0})};
  732. 732 };
  733. 733 if ((artifact.kind() == "RAW" || artifact.kind() == "BINARY") &&
  734. 734 fs::is_regular_file(directory / "uploaded"))
  735. 735 return saved_receipt(); // 兼容上一协议版本的恢复记录，不重复已经完成的
  736. 736 // 数据PUT；避免把网络重试变成再次采集或重复上传。
  737. 737 if (fs::is_regular_file(marker) &&
  738. 738 read_file(marker.string(), 128) == artifact.sha256())
  739. 739 return saved_receipt();
  740. 740 const auto retry_state = directory / ("retry-" + artifact.name() + ".json");
  741. 741 unsigned retry = 0;
  742. 742 if (fs::is_regular_file(retry_state)) {
  743. 743 const auto saved = parse_json(read_file(retry_state.string(), 4096));
  744. 744 retry = std::min(saved.value("attempt", 0U), 100000U);
  745. 745 const auto wait = saved.value("next_retry_unix_ms", std::int64_t{0}) -
  746. 746 unix_ns() / 1000000;
  747. 747 if (wait > 0)
  748. 748 pause_retry(
  749. 749 static_cast<std::uint64_t>(std::min(wait, std::int64_t{60000})),
  750. 750 renew);
  751. 751 }
  752. 752 for (unsigned performed = 0; alive && (!renew || !canceled_); ++performed) {
  753. 753 const auto upload_deadline =
  754. 754 task.deadline_unix_ms() + (renew ? 0 : 120000);
  755. 755 if (task.deadline_unix_ms() > 0 && unix_ns() / 1000000 >= upload_deadline)
  756. 756 throw UploadFailure(false);
  757. 757 unsigned retry_after = 0;
  758. 758 try {
  759. 759 if (renew && !progress(task, "UPLOADING"))
  760. 760 throw UploadFailure(!canceled_);
  761. 761 const auto receipt =
  762. 762 upload_file(url, file.string(), artifact.content_type(), maximum);
  763. 763 atomic_write(receipt_file.string(),
  764. 764 json{{"sha256", artifact.sha256()},
  765. 765 {"etag", receipt.etag},
  766. 766 {"completed_unix_ms", receipt.completed_unix_ms}}
  767. 767 .dump());
  768. 768 atomic_write(marker.string(), artifact.sha256());
  769. 769 metrics_.add("drop_upload_bytes_total",
  770. 770 static_cast<double>(artifact.size()));
  771. 771 return receipt;
  772. 772 } catch (const UploadFailure &error) {
  773. 773 metrics_.add("drop_upload_errors_total");
  774. 774 if (!error.retryable || (max_attempts && performed + 1 >= max_attempts))
  775. 775 throw;
  776. 776 retry_after = error.retry_after_seconds;
  777. 777 } catch (const std::exception &) {
  778. 778 throw UploadFailure(
  779. 779 false); // 本地文件或配置错误是永久失败，继续做网络重试无法修复。
  780. 780 }
  781. 781 const auto delay = std::max<std::uint64_t>(
  782. 782 upload_retry_delay_ms(retry, retry_entropy()),
  783. 783 static_cast<std::uint64_t>(retry_after) * 1000);
  784. 784 atomic_write(
  785. 785 retry_state.string(),
  786. 786 json{{"attempt", ++retry},
  787. 787 {"next_retry_unix_ms",
  788. 788 unix_ns() / 1000000 + static_cast<std::int64_t>(delay)}}
  789. 789 .dump());
  790. 790 pause_retry(delay, renew);
  791. 791 }
  792. 792 throw UploadFailure(
  793. 793 true); // 退出时保留已成功采集的恢复记录，下次启动继续上传与确认。
  794. 794 }
