  21. 21def command(argv, source, python, timeout=30):
  22. 22 """只执行 argv，不启动 shell；环境不继承云凭据，私有输出仅返回安全错误枚举。"""
  23. 23 env = {'PATH': str(Path(python).parent), 'HOME': tempfile.gettempdir(), 'PYTHONDONTWRITEBYTECODE': '1',
  24. 24 'PYTHONPATH': str(source), 'PYTHONHASHSEED': '0', 'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1'}
  25. 25 if os.name == 'nt' and 'SystemRoot' in os.environ:
  26. 26 env['SystemRoot'] = os.environ['SystemRoot']
  27. 27 with tempfile.TemporaryFile() as out, tempfile.TemporaryFile() as err:
  28. 28 try:
  29. 29 result = subprocess.run(argv, cwd=source, env=env, stdout=out, stderr=err, timeout=timeout)
  30. 30 except subprocess.TimeoutExpired:
  31. 31 return False, 'EXECUTION_TIMEOUT'
  32. 32 if out.tell() > 65536 or err.tell() > 65536:
  33. 33 return False, 'EXECUTION_OUTPUT_LIMIT'
  34. 34 return result.returncode == 0, '' if result.returncode == 0 else 'EXECUTION_FAILED'

  37. 37def measure(source, manifest_path, action, out, seed, python, manifest):
  38. 38 substitutes = {'{driver}': str(HARNESS / 'measure.py'), '{source}': str(source), '{manifest}': str(manifest_path),
  39. 39 '{output}': str(out), '{seed}': str(seed), 'python': python}
  40. 40 config = manifest['commands'][action]
  41. 41 argv = [substitutes.get(arg, arg) for arg in config['argv']]

  44. 44 raise RuntimeError('MEASUREMENT_' + code)
  45. 45 if not out.is_file() or out.stat().st_size > 32768:
  46. 46 raise RuntimeError('MEASUREMENT_OUTPUT_INVALID')

  67. 67def main():

  69. 69 parser.add_argument('--action', choices=['profile', 'test'], required=True)
  70. 70 parser.add_argument('--workspace', default='/workspace')
  71. 71 parser.add_argument('--output', default='/output')
  72. 72 parser.add_argument('--python', default=sys.executable)
  73. 73 args = parser.parse_args()
  74. 74 root, out = Path(args.workspace).resolve(), Path(args.output).resolve()
  75. 75 out.mkdir(parents=True, exist_ok=True)
  76. 76 manifest_path = root / 'manifest.json'
  77. 77 manifest = json.loads(manifest_path.read_text(encoding='utf-8'))
  78. 78 seed = json.loads((root / 'job.json').read_text(encoding='utf-8'))['seed']
  79. 79 baseline, candidate = root / 'baseline', root / 'candidate'
  80. 80 begin = time.monotonic()
  81. 81 original_checks(baseline, manifest, args.python)
  82. 82 if args.action == 'profile':
  83. 83 result = measure(baseline, manifest_path, 'profile', out / 'profile.json', seed, args.python, manifest)
  84. 84 result.update({'success': True, 'original_regression': 'passed'})
  85. 85 else:
  86. 86 before = regression(baseline, root / 'regression-tests', manifest_path, out / 'regression-before.json', args.python, seed)
  87. 87 after = regression(candidate, root / 'regression-tests', manifest_path, out / 'regression-after.json', args.python, seed)
  88. 88 defect_reproduced = before['tests'] > 0 and before['failures'] > 0 and before['errors'] == 0 and before['skipped'] == 0 and before['expected_failures'] == 0
  89. 89 defect_fixed = after['tests'] == before['tests'] and after['success'] and not any(after[key] for key in ('failures', 'errors', 'skipped', 'expected_failures', 'unexpected_successes'))
  90. 90 validation = {'baseline_failed': defect_reproduced, 'candidate_passed': defect_fixed,
  91. 91 'baseline': before, 'candidate': after, 'same_test_files': True,
  92. 92 'test_sha256': {file.name: hashlib.sha256(file.read_bytes()).hexdigest() for file in sorted((root / 'regression-tests').glob('test_*.py'))}}
  93. 93 if not defect_reproduced or not defect_fixed:
  94. 94 result = {'success': False, 'accepted': False, 'error_code': 'DEFECT_REGRESSION_NOT_ESTABLISHED', 'regression_test_validation': validation}
  95. 95 else:
  96. 96 original_checks(candidate, manifest, args.python)
  97. 97 pairs, controls = [], []
  98. 98 for trial in range(6):
  99. 99 order = ['baseline', 'candidate'] if trial % 2 == 0 else ['candidate', 'baseline']
  100. 100 measured = {side: measure(root / side, manifest_path, 'benchmark', out / f'{trial}-{side}.json', seed + trial, args.python, manifest) for side in order}
  101. 101 left, right = measured['baseline'], measured['candidate']
  102. 102 if left['output_sha256'] != right['output_sha256'] or left['output_type'] != right['output_type'] or not right['mutable_output_isolated']:
  103. 103 raise RuntimeError('OUTPUT_TYPE_OR_VALUE_PARITY_FAILED')
  104. 104 if min(left['cpu_ns'], right['cpu_ns']) < 1000000:
  105. 105 raise RuntimeError('MEASUREMENT_UNDER_RESOLVED')
  106. 106 pairs.append({'baseline_cpu_ns': left['cpu_ns'], 'candidate_cpu_ns': right['cpu_ns'], 'cases': left['cases'], 'output_sha256': left['output_sha256']})
  107. 107 # sham 对照运行两次完全相同的冻结 baseline；它必须显示稳定性，不能冒充一个优化补丁。
  108. 108 control = measure(baseline, manifest_path, 'benchmark', out / f'{trial}-negative.json', seed + trial, args.python, manifest)
  109. 109 controls.append(left['cpu_ns'] / control['cpu_ns'])
  110. 110 ratios = [pair['baseline_cpu_ns'] / pair['candidate_cpu_ns'] for pair in pairs]
  111. 111 speedup, negative_median = statistics.median(ratios), statistics.median(controls)
  112. 112 gate = manifest['performance_gate']
  113. 113 stable = 1 - gate['negative_control_max_drift'] <= negative_median <= 1 + gate['negative_control_max_drift']
  114. 114 improved = sum(ratio > 1.05 for ratio in ratios)
  115. 115 accepted = stable and speedup >= gate['minimum_median_speedup'] and improved >= gate['minimum_improved_pairs']
  116. 116 result = {'success': True, 'accepted': accepted, 'functional_regression': 'passed', 'regression_test_validation': validation,
  117. 117 'original_tests_unchanged': True, 'pairs': pairs, 'output_parity_cases': sum(pair['cases'] for pair in pairs),
  118. 118 'median_cpu_speedup': speedup, 'improved_pairs': improved,
  119. 119 'negative_control': {'kind': 'identical_frozen_baseline', 'ratios': controls, 'median_ratio': negative_median, 'stable': stable},
  120. 120 'gate': gate, 'root_cause_status': 'validated_in_controlled_python_benchmark' if accepted else 'not_validated',
  121. 121 'scope': 'administrator-registered Python workload on synthetic input; isolated snapshot only; not production latency or whole-project improvement'}
  122. 122 result['candidate_profile'] = measure(candidate, manifest_path, 'profile', out / 'candidate-profile.json', seed, args.python, manifest)
  123. 123 result['elapsed_seconds'] = round(time.monotonic() - begin, 3)
  124. 124 (out / 'result.json').write_text(json.dumps(result, ensure_ascii=False), encoding='utf-8')
