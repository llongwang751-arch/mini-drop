  139. 139// NewAgentRunner 创建两条独立 Unix socket 通路，并核验 Python UID 与网关目录边界。
  140. 140// 模型密钥只属于 runner 配置；Go 在这里注册临时工具会话，不向 Python 提供数据库或执行器连接。
  141. 141func NewAgentRunner(runnerSocket, gatewaySocket string) (*AgentRunner, error) {
  142. 142 if !validAgentSocket(runnerSocket) || !validAgentSocket(gatewaySocket) || runnerSocket == gatewaySocket {
  143. 143 return nil, errors.New("invalid AgentScope socket configuration")
  144. 144 }
  145. 145 if !validAgentDirectory(runnerSocket, 10002) {
  146. 146 return nil, errors.New("AgentScope runner socket directory rejected")
  147. 147 }
  148. 148 listener, err := listenAgentGateway(gatewaySocket)
  149. 149 if err != nil {
  150. 150 return nil, errors.New("AgentScope gateway socket unavailable")
  151. 151 }
  152. 152 transport := &http.Transport{DialContext: func(ctx context.Context, _, _ string) (net.Conn, error) {
  153. 153 conn, err := (&net.Dialer{Timeout: 5 * time.Second}).DialContext(ctx, "unix", runnerSocket)
  154. 154 if err != nil {
  155. 155 return nil, err
  156. 156 }
  157. 157 if err := verifyAgentPeer(conn, 10002); err != nil {
  158. 158 conn.Close()
  159. 159 return nil, err
  160. 160 }
  161. 161 return conn, nil
  162. 162 }, MaxConnsPerHost: 4, MaxIdleConns: 4, ResponseHeaderTimeout: 425 * time.Second}
  163. 163 r := &AgentRunner{sessions: map[string]*agentSession{}, listener: listener, client: &http.Client{Transport: transport, CheckRedirect: func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }}}
  164. 164 r.server = &http.Server{Handler: r, ReadHeaderTimeout: 5 * time.Second, ReadTimeout: 10 * time.Second, IdleTimeout: 30 * time.Second, MaxHeaderBytes: 4096}
  165. 165 go func() { _ = r.server.Serve(listener) }()
  166. 166 return r, nil
  167. 167}

  179. 179func decodeAgentJSON(data []byte, out any) bool {
  180. 180 d := json.NewDecoder(bytes.NewReader(data))
  181. 181 d.DisallowUnknownFields()
  182. 182 var tail any
  183. 183 return d.Decode(out) == nil && d.Decode(&tail) == io.EOF
  184. 184}

  186. 186// canonicalAgentJSON 规范化对象键和空白，同时用 UseNumber 保留数值词法与精度。
  187. 187// 0.000001、1e-6、-0 或超 float64 精度的小数不能在 Python 重编码后悄悄改变，再被宽松浮点比较掩盖。
  188. 188func canonicalAgentJSON(data []byte) ([]byte, bool) {
  189. 189 var value any
  190. 190 d := json.NewDecoder(bytes.NewReader(data))
  191. 191 d.UseNumber()
  192. 192 var tail any
  193. 193 if d.Decode(&value) != nil || d.Decode(&tail) != io.EOF {
  194. 194 return nil, false
  195. 195 }
  196. 196 encoded, err := json.Marshal(value)
  197. 197 return encoded, err == nil
  198. 198}

  204. 204func safeAgentError(err error) string {
  205. 205 var e *domain.Error
  206. 206 if errors.As(err, &e) && len(e.Code) > 0 && len(e.Code) <= 80 && strings.IndexFunc(e.Code, func(r rune) bool { return !(r >= 'A' && r <= 'Z' || r >= '0' && r <= '9' || r == '_') }) < 0 {
  207. 207 return e.Code
  208. 208 }
  209. 209 if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {

  211. 211 }

  215. 215// ServeHTTP 处理私有工具回调，先验证 capability、请求上限和固定字段，再在会话锁内校验回放、轮次、schema 与业务栅栏。
  216. 216// 相同 call_id 与相同规范请求重放只返回缓存；相同 ID 携带不同内容会撤销整次运行，不能重复执行副作用。
  217. 217func (r *AgentRunner) ServeHTTP(w http.ResponseWriter, request *http.Request) {
  218. 218 w.Header().Set("Content-Type", "application/json")
  219. 219 deny := func(status int, code string) {
  220. 220 body, _ := json.Marshal(agentFailure(code, true))
  221. 221 w.Header().Set("Content-Length", strconv.Itoa(len(body)))
  222. 222 w.WriteHeader(status)
  223. 223 _, _ = w.Write(body)
  224. 224 }
  225. 225 if request.Method != http.MethodPost || !strings.HasPrefix(request.URL.Path, "/v1/callback/") || request.URL.RawQuery != "" {
  226. 226 deny(http.StatusNotFound, "AGENT_CALLBACK_REJECTED")
  227. 227 return
  228. 228 }
  229. 229 id := strings.TrimPrefix(request.URL.Path, "/v1/callback/")
  230. 230 r.mu.Lock()
  231. 231 session := r.sessions[id]
  232. 232 r.mu.Unlock()
  233. 233 auth := request.Header.Get("Authorization")
  234. 234 if session == nil || subtle.ConstantTimeCompare([]byte(auth), []byte("Bearer "+session.capability)) != 1 {
  235. 235 deny(http.StatusUnauthorized, "AGENT_CAPABILITY_REJECTED")
  236. 236 return
  237. 237 }
  238. 238 raw, err := io.ReadAll(io.LimitReader(request.Body, 262145))
  239. 239 if err != nil || len(raw) > 262144 {
  240. 240 deny(http.StatusRequestEntityTooLarge, "AI_BUDGET_EXCEEDED")
  241. 241 return
  242. 242 }
  243. 243 var call agentCallback
  244. 244 canonical, ok := canonicalAgentJSON(raw)
  245. 245 if !ok || !decodeAgentJSON(raw, &call) || len(call.CallID) < 1 || len(call.CallID) > 128 || strings.ContainsAny(call.CallID, "\r\n\x00") {
  246. 246 deny(http.StatusBadRequest, "AGENT_CALLBACK_REJECTED")
  247. 247 return
  248. 248 }
  249. 249 if call.Type != "checkpoint" && len(raw) > 65536 {
  250. 250 deny(http.StatusRequestEntityTooLarge, "AI_BUDGET_EXCEEDED")
  251. 251 return
  252. 252 }
  253. 253 session.mu.Lock()
  254. 254 defer session.mu.Unlock()
  255. 255 if cause := context.Cause(session.ctx); cause != nil {
  256. 256 deny(http.StatusGone, safeAgentError(cause))
  257. 257 return
  258. 258 }
  259. 259 if previous, exists := session.cache[call.CallID]; exists {
  260. 260 if !bytes.Equal(canonical, previous.Request) {
  261. 261 session.cancel(domain.Fail("AGENT_CALLBACK_REPLAY", "工具回调标识被重复用于不同请求"))
  262. 262 deny(http.StatusConflict, "AGENT_CALLBACK_REPLAY")
  263. 263 return
  264. 264 }
  265. 265 w.Header().Set("Content-Length", strconv.Itoa(len(previous.Response)))
  266. 266 _, _ = w.Write(previous.Response)
  267. 267 return
  268. 268 }
  269. 269 if call.Type == "tool" && session.spec.Checkpoint != nil {
  270. 270 replayed, found, err := session.spec.Checkpoint.replay(session.ctx, call)
  271. 271 if err != nil {
  272. 272 session.cancel(err)
  273. 273 deny(http.StatusConflict, safeAgentError(err))
  274. 274 return
  275. 275 }
  276. 276 if found {
  277. 277 var saved map[string]json.RawMessage
  278. 278 if json.Unmarshal(replayed, &saved) == nil && bytes.Equal(saved["terminal"], []byte("true")) && json.Valid(saved["result"]) {
  279. 279 session.accepted = append(json.RawMessage(nil), saved["result"]...)
  280. 280 if err := session.spec.Checkpoint.finish(session.ctx, session.accepted, session.formatOnly); err != nil {
  281. 281 session.cancel(err)
  282. 282 deny(http.StatusConflict, safeAgentError(err))
  283. 283 return
  284. 284 }
  285. 285 }
  286. 286 w.Header().Set("Content-Length", strconv.Itoa(len(replayed)))
  287. 287 _, _ = w.Write(replayed)
  288. 288 return
  289. 289 }
  290. 290 }
  291. 291 response := session.callback(call)
  292. 292 encoded, marshalErr := json.Marshal(response)
  293. 293 if marshalErr != nil || len(encoded) > 65536 {
  294. 294 session.cancel(domain.Fail("AI_BUDGET_EXCEEDED", "工具回调输出超限"))
  295. 295 encoded, _ = json.Marshal(agentFailure("AI_BUDGET_EXCEEDED", true))
  296. 296 }
  297. 297 session.cache[call.CallID] = cachedAgentCallback{Request: canonical, Response: encoded}
  298. 298 w.Header().Set("Content-Length", strconv.Itoa(len(encoded)))
  299. 299 _, _ = w.Write(encoded)
  300. 300}

  302. 302// callback 在 Go 侧保留模型调用、工具调用和最终报告的最后决定权。
  303. 303// model_start 先预留连续轮次，tool 独立校验阶段与参数，finish 通过后封存 accepted 结果；封存后不能追加模型或工具。
  304. 304func (s *agentSession) callback(call agentCallback) map[string]any {
  305. 305 fail := func(code string) map[string]any {
  306. 306 s.cancel(domain.Fail(code, "AgentScope运行协议或预算无效"))
  307. 307 return agentFailure(code, true)
  308. 308 }
  309. 309 if len(s.accepted) != 0 || call.Round < 0 || call.Round == 0 && call.Type != "checkpoint" || len(s.cache) >= 128 {
  310. 310 return fail("AGENT_CALLBACK_REJECTED")
  311. 311 }
  312. 312 if s.spec.Fence != nil {
  313. 313 if err := s.spec.Fence(s.ctx); err != nil {
  314. 314 s.cancel(err)
  315. 315 return agentFailure(safeAgentError(err), true)
  316. 316 }
  317. 317 }
  318. 318 if call.Type != "tool" && (call.Name != "" || len(call.Arguments) != 0) || call.Type != "finish" && call.Type != "checkpoint" && call.Content != "" || call.Type != "model_end" && len(call.Usage) != 0 {
  319. 319 return fail("AGENT_CALLBACK_REJECTED")
  320. 320 }
  321. 321 switch call.Type {
  322. 322 case "checkpoint":
  323. 323 var checkpoint struct {
  324. 324 Sequence int64 `json:"sequence"`
  325. 325 State json.RawMessage `json:"state"`
  326. 326 }
  327. 327 if s.spec.Checkpoint == nil || call.Round != s.round || !decodeAgentJSON([]byte(call.Content), &checkpoint) {
  328. 328 return fail("AGENT_CHECKPOINT_REJECTED")
  329. 329 }
  330. 330 if err := s.spec.Checkpoint.saveSDK(s.ctx, checkpoint.Sequence, checkpoint.State); err != nil {
  331. 331 s.cancel(err)
  332. 332 return agentFailure(safeAgentError(err), true)
  333. 333 }
  334. 334 return map[string]any{"ok": true}
  335. 335 case "model_start":
  336. 336 if call.Round != s.round+1 || call.Round > s.budget.MaxModelCalls || s.spec.Phase == "diagnosis" && call.Round == s.budget.MaxModelCalls && !s.formatOnly {
  337. 337 return fail("AI_BUDGET_EXCEEDED")
  338. 338 }
  339. 339 if s.spec.Checkpoint != nil {
  340. 340 if err := s.spec.Checkpoint.reserveModel(s.ctx, call.Round, s.budget, s.spec.Phase); err != nil {
  341. 341 s.cancel(err)
  342. 342 return agentFailure(safeAgentError(err), true)
  343. 343 }
  344. 344 }
  345. 345 s.round = call.Round
  346. 346 s.roundTools = 0
  347. 347 return map[string]any{"ok": true}
  348. 348 case "model_end":
  349. 349 if call.Round != s.round || s.endedRounds[call.Round] || len(call.Usage) > 4096 {
  350. 350 return fail("AGENT_CALLBACK_REJECTED")
  351. 351 }
  352. 352 if len(call.Usage) > 0 && !validAgentUsage(call.Usage) {
  353. 353 return fail("AGENT_CALLBACK_REJECTED")
  354. 354 }
  355. 355 if s.spec.Checkpoint != nil {
  356. 356 if err := s.spec.Checkpoint.endModel(s.ctx, call.Round, call.Usage); err != nil {
  357. 357 s.cancel(err)
  358. 358 return agentFailure(safeAgentError(err), true)
  359. 359 }
  360. 360 }
  361. 361 s.endedRounds[call.Round] = true
  362. 362 return map[string]any{"ok": true}
  363. 363 case "tool":
  364. 364 maxArgs := 1024
  365. 365 if s.spec.Phase == "planning" {
  366. 366 maxArgs = 4096
  367. 367 }
  368. 368 if s.spec.Phase == "repair" {
  369. 369 maxArgs = 32768
  370. 370 }
  371. 371 if call.Round != s.round || s.formatOnly || !phaseTools(s.spec.Phase)[call.Name] || s.toolCalls >= s.budget.MaxToolCalls || s.roundTools >= s.budget.MaxToolsPerRound || len(call.Arguments) > maxArgs {
  372. 372 return fail("AI_BUDGET_EXCEEDED")
  373. 373 }
  374. 374 var args map[string]any
  375. 375 if !decodeAgentJSON(call.Arguments, &args) || args == nil {
  376. 376 return fail("AI_TOOL_REJECTED")
  377. 377 }
  378. 378 if schema := s.schemas[call.Name]; schema == nil || schema.Validate(args) != nil {
  379. 379 return fail("AI_TOOL_REJECTED")
  380. 380 }
  381. 381 if s.spec.Checkpoint != nil {
  382. 382 if err := s.spec.Checkpoint.beginTool(s.ctx, call, s.budget); err != nil {
  383. 383 s.cancel(err)
  384. 384 return agentFailure(safeAgentError(err), true)
  385. 385 }
  386. 386 }
  387. 387 s.toolCalls++
  388. 388 s.roundTools++
  389. 389 result, err := s.spec.Invoke(s.ctx, call.Name, call.Arguments)
  390. 390 if err != nil {
  391. 391 s.cancel(err)
  392. 392 return agentFailure(safeAgentError(err), true)
  393. 393 }
  394. 394 response := map[string]any{"ok": true, "observation": result.Observation}
  395. 395 if len(result.Terminal) > 0 {
  396. 396 if s.spec.Phase != "planning" || call.Name != "propose_collection" || !json.Valid(result.Terminal) {
  397. 397 return fail("AGENT_CALLBACK_REJECTED")
  398. 398 }
  399. 399 s.accepted = append(json.RawMessage(nil), result.Terminal...)
  400. 400 response["terminal"] = true
  401. 401 response["result"] = s.accepted
  402. 402 }
  403. 403 if s.spec.Checkpoint != nil {
  404. 404 if err := s.spec.Checkpoint.completeTool(s.ctx, call, response); err != nil {
  405. 405 s.cancel(err)
  406. 406 return agentFailure(safeAgentError(err), true)
  407. 407 }
  408. 408 if len(s.accepted) > 0 {
  409. 409 if err := s.spec.Checkpoint.finish(s.ctx, s.accepted, s.formatOnly); err != nil {
  410. 410 s.cancel(err)
  411. 411 return agentFailure(safeAgentError(err), true)
  412. 412 }
  413. 413 }
  414. 414 }
  415. 415 return response
  416. 416 case "finish":
  417. 417 if call.Round != s.round || len(call.Content) > 12000 || s.spec.Finish == nil {
  418. 418 return fail("AGENT_CALLBACK_REJECTED")
  419. 419 }
  420. 420 accepted, err := s.spec.Finish(s.ctx, call.Content)
  421. 421 if err != nil {
  422. 422 // 只记录固定错误枚举及已计数的调用量，便于区分报告修订和工具循环；不保存候选正文。
  423. 423 slog.Warn("AgentScope report rejected", "reason", "report_rejected", "error_code", safeAgentError(err), "stage", "finish_callback", "model_calls", s.round, "tool_calls", s.toolCalls)
  424. 424 response := agentFailure(safeAgentError(err), false)
  425. 425 // 只接受专用类型产生并重新核验的安全反馈；绝不把err.Error/domain.Message或上游内容回填给模型。
  426. 426 if provider, ok := err.(diagnosisReportBatchFeedbackProvider); ok {
  427. 427 if feedback, valid := provider.safeDiagnosisReportBatchFeedback().safeJSON(); valid {
  428. 428 response["feedback"] = feedback
  429. 429 }
  430. 430 } else if provider, ok := err.(diagnosisReportFeedbackProvider); ok {
  431. 431 if feedback, valid := provider.safeDiagnosisReportFeedback().safeJSON(); valid {
  432. 432 response["feedback"] = feedback
  433. 433 }
  434. 434 }
  435. 435 if s.spec.Phase == "diagnosis" && call.Round == s.budget.MaxModelCalls-1 && safeAgentError(err) == "AI_INVALID_REPORT" {
  436. 436 s.formatOnly = true
  437. 437 response["format_only"] = true
  438. 438 }
  439. 439 if s.spec.Checkpoint != nil {
  440. 440 if persistErr := s.spec.Checkpoint.finish(s.ctx, nil, s.formatOnly); persistErr != nil {
  441. 441 s.cancel(persistErr)
  442. 442 return agentFailure(safeAgentError(persistErr), true)
  443. 443 }
  444. 444 }
  445. 445 return response
  446. 446 }
  447. 447 if !json.Valid(accepted) {
  448. 448 return fail("AGENT_CALLBACK_REJECTED")
  449. 449 }
  450. 450 s.accepted = append(json.RawMessage(nil), accepted...)
  451. 451 if s.spec.Checkpoint != nil {
  452. 452 if err := s.spec.Checkpoint.finish(s.ctx, s.accepted, s.formatOnly); err != nil {
  453. 453 s.cancel(err)
  454. 454 return agentFailure(safeAgentError(err), true)
  455. 455 }
  456. 456 }
  457. 457 return map[string]any{"ok": true, "accepted": true, "result": s.accepted}
  458. 458 }
  459. 459 return fail("AGENT_CALLBACK_REJECTED")
  460. 460}

  462. 462func validAgentUsage(raw json.RawMessage) bool {
  463. 463 var value map[string]json.RawMessage
  464. 464 if !decodeAgentJSON(raw, &value) || value == nil || len(value) > 16 {
  465. 465 return false
  466. 466 }
  467. 467 for name, number := range value {
  468. 468 if len(name) > 64 || strings.IndexFunc(name, func(r rune) bool { return !(r >= 'a' && r <= 'z' || r == '_') }) >= 0 {
  469. 469 return false
  470. 470 }
  471. 471 var n int64
  472. 472 if json.Unmarshal(number, &n) != nil || n < 0 || n > 1000000000 {
  473. 473 return false
  474. 474 }
  475. 475 }
  476. 476 return true
  477. 477}

  479. 479// Run 为一次阶段生成随机 capability 和公开 run_id，启动来源栅栏轮询并调用独立 runner。
  480. 480// 取消或撤权会关闭上游 context 并注销能力；返回时严格核对 accepted JSON、框架证明和独立计数，不能信任 Python 单方面宣称成功。
  481. 481func (r *AgentRunner) Run(ctx context.Context, spec agentRunSpec) (json.RawMessage, *AgentRuntime, error) {
  482. 482 if r == nil {
  483. 483 return nil, nil, domain.Fail("AGENT_RUNTIME_UNAVAILABLE", "AgentScope未配置")
  484. 484 }
  485. 485 budget, valid := phaseBudget(spec.Phase)
  486. 486 if !valid || spec.Invoke == nil || spec.Finish == nil {
  487. 487 return nil, nil, domain.Fail("AGENT_RUNTIME_ERROR", "AgentScope阶段配置无效")
  488. 488 }
  489. 489 if spec.Budget != nil {
  490. 490 requested := *spec.Budget
  491. 491 if requested.MaxModelCalls < 1 || requested.MaxModelCalls > budget.MaxModelCalls || requested.MaxToolCalls < 1 || requested.MaxToolCalls > budget.MaxToolCalls || requested.MaxToolsPerRound < 1 || requested.MaxToolsPerRound > budget.MaxToolsPerRound || requested.TimeoutMS < 1 || requested.TimeoutMS > budget.TimeoutMS {
  492. 492 return nil, nil, domain.Fail("AI_BUDGET_EXCEEDED", "AgentScope请求预算越界")
  493. 493 }
  494. 494 budget = requested
  495. 495 }
  496. 496 schemas, err := compileAgentTools(spec.Phase, spec.Tools)
  497. 497 if err != nil {
  498. 498 return nil, nil, err
  499. 499 }
  500. 500 var restored *agentCheckpointSnapshot
  501. 501 if spec.Checkpoint != nil {
  502. 502 encoded, _ := json.Marshal(map[string]any{"phase": spec.Phase, "model": spec.Model, "system_prompt": spec.SystemPrompt, "messages": spec.Messages, "tools": spec.Tools, "budget": budget})
  503. 503 snapshot, loadErr := spec.Checkpoint.load(ctx, repository.DigestBytes(encoded), budget)
  504. 504 if loadErr != nil {
  505. 505 return nil, nil, loadErr
  506. 506 }
  507. 507 restored = &snapshot
  508. 508 remaining := budget.TimeoutMS - snapshot.ActiveMS
  509. 509 if remaining < 1 && len(snapshot.Accepted) == 0 {
  510. 510 return nil, nil, agentCheckpointError("AI_BUDGET_EXCEEDED")
  511. 511 }
  512. 512 if remaining < 1 {
  513. 513 remaining = 1000
  514. 514 }
  515. 515 if remaining < budget.TimeoutMS {
  516. 516 budget.TimeoutMS = remaining
  517. 517 }
  518. 518 }
  519. 519 work, cancel := context.WithTimeout(ctx, time.Duration(budget.TimeoutMS)*time.Millisecond)
  520. 520 defer cancel()
  521. 521 work, revoke := context.WithCancelCause(work)
  522. 522 defer revoke(context.Canceled)
  523. 523 if spec.Checkpoint != nil && len(restored.Accepted) == 0 {
  524. 524 originalBudget, _ := phaseBudget(spec.Phase)
  525. 525 if spec.Budget != nil {
  526. 526 originalBudget = *spec.Budget
  527. 527 }
  528. 528 granted, reserveErr := spec.Checkpoint.reserveActive(work, originalBudget.TimeoutMS)
  529. 529 if reserveErr != nil {
  530. 530 return nil, nil, reserveErr
  531. 531 }
  532. 532 // 独立截止计时器与续片分开，数据库续片变慢不能让会话越过已经持久预留的时间。
  533. 533 watchdog := time.AfterFunc(time.Duration(granted)*time.Millisecond, func() { revoke(context.DeadlineExceeded) })
  534. 534 defer watchdog.Stop()
  535. 535 go func() {
  536. 536 reservedActive := restored.ActiveMS + granted
  537. 537 for {
  538. 538 if reservedActive >= originalBudget.TimeoutMS {
  539. 539 return
  540. 540 }
  541. 541 wait := time.Duration(granted)*time.Millisecond - 250*time.Millisecond
  542. 542 if wait < time.Millisecond {
  543. 543 wait = time.Millisecond
  544. 544 }
  545. 545 timer := time.NewTimer(wait)
  546. 546 select {
  547. 547 case <-work.Done():
  548. 548 timer.Stop()
  549. 549 return
  550. 550 case <-timer.C:
  551. 551 }
  552. 552 reserveCtx, stop := context.WithTimeout(work, 200*time.Millisecond)
  553. 553 next, err := spec.Checkpoint.reserveActive(reserveCtx, originalBudget.TimeoutMS)
  554. 554 stop()
  555. 555 if err != nil {
  556. 556 revoke(err)
  557. 557 return
  558. 558 }
  559. 559 if !watchdog.Reset(time.Duration(next) * time.Millisecond) {
  560. 560 revoke(context.DeadlineExceeded)
  561. 561 return
  562. 562 }
  563. 563 granted = next
  564. 564 reservedActive += next
  565. 565 }
  566. 566 }()
  567. 567 }
  568. 568 id, capability := uuid.NewString(), RandomToken()
  569. 569 session := &agentSession{ctx: work, cancel: revoke, capability: capability, spec: spec, budget: budget, cache: map[string]cachedAgentCallback{}, endedRounds: map[int]bool{}, schemas: schemas}
  570. 570 if restored != nil {
  571. 571 session.round = restored.ModelCalls
  572. 572 session.toolCalls = restored.ToolCalls
  573. 573 session.roundTools = restored.RoundTools
  574. 574 session.formatOnly = restored.FormatOnly
  575. 575 session.accepted = append(json.RawMessage(nil), restored.Accepted...)
  576. 576 for _, round := range restored.EndedRounds {
  577. 577 session.endedRounds[round] = true
  578. 578 }
  579. 579 }
  580. 580 r.mu.Lock()
  581. 581 r.sessions[id] = session
  582. 582 r.mu.Unlock()
  583. 583 defer func() {
  584. 584 revoke(context.Canceled)
  585. 585 r.mu.Lock()
  586. 586 delete(r.sessions, id)
  587. 587 r.mu.Unlock()
  588. 588 // 工作器收尾前，等待仍在执行的工具回调退出，避免终态落库后旧回调继续写入。
  589. 589 session.mu.Lock()
  590. 590 session.mu.Unlock()
  591. 591 }()
  592. 592 if spec.Fence != nil {
  593. 593 if err := spec.Fence(work); err != nil {
  594. 594 return nil, nil, err
  595. 595 }
  596. 596 go func() {
  597. 597 ticker := time.NewTicker(500 * time.Millisecond)
  598. 598 defer ticker.Stop()
  599. 599 for {
  600. 600 select {
  601. 601 case <-work.Done():
  602. 602 return
  603. 603 case <-ticker.C:
  604. 604 if err := spec.Fence(work); err != nil {
  605. 605 revoke(err)
  606. 606 return
  607. 607 }
  608. 608 }
  609. 609 }
  610. 610 }()
  611. 611 }
  612. 612 payload := map[string]any{"version": 1, "run_id": id, "capability": capability, "phase": spec.Phase, "model": spec.Model, "system_prompt": spec.SystemPrompt, "messages": spec.Messages, "tools": spec.Tools, "budget": budget}
  613. 613 if restored != nil {
  614. 614 payload["checkpoint"] = restored
  615. 615 }
  616. 616 body, err := json.Marshal(payload)
  617. 617 if err != nil || len(body) > 262144 {
  618. 618 return nil, nil, domain.Fail("AI_BUDGET_EXCEEDED", "AgentScope输入超限")
  619. 619 }
  620. 620 request, _ := http.NewRequestWithContext(work, http.MethodPost, "http://agentscope/v1/run", bytes.NewReader(body))
  621. 621 request.Header.Set("Content-Type", "application/json")
  622. 622 response, err := r.client.Do(request)
  623. 623 if err != nil {
  624. 624 if cause := context.Cause(work); cause != nil {
  625. 625 return nil, nil, cause
  626. 626 }
  627. 627 return nil, nil, domain.Fail("AGENT_RUNTIME_UNAVAILABLE", "AgentScope服务调用失败")
  628. 628 }
  629. 629 defer response.Body.Close()
  630. 630 raw, err := io.ReadAll(io.LimitReader(response.Body, 131073))
  631. 631 if cause := context.Cause(work); cause != nil {
  632. 632 return nil, nil, cause
  633. 633 }
  634. 634 if err != nil || len(raw) > 131072 {
  635. 635 return nil, nil, domain.Fail("AGENT_RUNTIME_ERROR", "AgentScope响应无效")
  636. 636 }
  637. 637 if response.StatusCode != http.StatusOK {
  638. 638 return nil, nil, agentRuntimeFailure(response.StatusCode, raw, budget)
  639. 639 }
  640. 640 var answer struct {
  641. 641 Version int `json:"version"`
  642. 642 RunID string `json:"run_id"`
  643. 643 Result json.RawMessage `json:"result"`
  644. 644 Runtime AgentRuntime `json:"runtime"`
  645. 645 }
  646. 646 if !decodeAgentJSON(raw, &answer) {
  647. 647 slog.Error("AgentScope execution failed", "reason", "response_protocol_invalid")
  648. 648 return nil, nil, domain.Fail("AGENT_RUNTIME_PROTOCOL_ERROR", "AgentScope响应协议无效")
  649. 649 }
  650. 650 session.mu.Lock()
  651. 651 defer session.mu.Unlock()
  652. 652 a, aOK := canonicalAgentJSON(answer.Result)
  653. 653 b, bOK := canonicalAgentJSON(session.accepted)
  654. 654 m := answer.Runtime
  655. 655 if !aOK || !bOK || !bytes.Equal(a, b) {
  656. 656 slog.Error("AgentScope execution failed", "reason", "accepted_result_mismatch", "model_calls", session.round, "tool_calls", session.toolCalls)
  657. 657 return nil, nil, domain.Fail("AGENT_RUNTIME_RESULT_MISMATCH", "AgentScope结果与Go验收内容不一致")
  658. 658 }
  659. 659 if answer.Version != 1 || answer.RunID != id || m.Framework != "agentscope" || m.FrameworkVersion != "2.0.10" || m.AgentClass != "agentscope.agent.Agent" || m.RunID != id || m.ModelCalls != session.round || m.ToolCalls != session.toolCalls || m.ElapsedMS < 0 || m.ElapsedMS > budget.TimeoutMS+5000 || len(m.Usage) > 0 && !validAgentUsage(m.Usage) {
  660. 660 slog.Error("AgentScope execution failed", "reason", "runtime_metadata_mismatch", "model_calls", session.round, "tool_calls", session.toolCalls)
  661. 661 return nil, nil, domain.Fail("AGENT_RUNTIME_METADATA_MISMATCH", "AgentScope运行身份或计数无效")
  662. 662 }
  663. 663 return append(json.RawMessage(nil), session.accepted...), &m, nil
  664. 664}

  666. 666// 私有 runner 失败时只返回固定的应用错误枚举。
  667. 667// 不得保存响应正文、SDK 异常原文、任意枚举或额外字段；诊断字段也必须单独经过白名单校验。
  668. 668func agentRuntimeFailure(status int, raw []byte, budget agentBudget) error {
  669. 669 fallback := domain.Fail("AGENT_RUNTIME_ERROR", "AgentScope响应无效")
  670. 670 if status < 400 || status > 599 || len(raw) > 512 {
  671. 671 return fallback
  672. 672 }
  673. 673 var response struct {
  674. 674 ErrorCode string `json:"error_code"`
  675. 675 Diagnostics *agentFailureDiagnostics `json:"diagnostics,omitempty"`
  676. 676 }
  677. 677 if !decodeAgentJSON(raw, &response) {
  678. 678 return fallback
  679. 679 }
  680. 680 var fields map[string]json.RawMessage
  681. 681 _ = json.Unmarshal(raw, &fields)
  682. 682 if diagnostic, present := fields["diagnostics"]; present {
  683. 683 var members map[string]json.RawMessage
  684. 684 if response.Diagnostics == nil || !response.Diagnostics.valid(budget) || json.Unmarshal(diagnostic, &members) != nil || len(members) != 4 {
  685. 685 return fallback
  686. 686 }
  687. 687 for _, key := range []string{"stage", "exception_kind", "model_calls", "tool_calls"} {
  688. 688 if value, ok := members[key]; !ok || bytes.Equal(bytes.TrimSpace(value), []byte("null")) {
  689. 689 return fallback
  690. 690 }
  691. 691 }
  692. 692 }
  693. 693 switch response.ErrorCode {
  694. 694 case "AI_CONFIGURATION_ERROR", "AI_TOOL_REJECTED", "AI_BUDGET_EXCEEDED", "AI_INVALID_RESPONSE", "AI_UPSTREAM_ERROR", "AI_INVALID_REPORT", "AI_INSUFFICIENT_EVIDENCE",
  695. 695 "AGENT_CHECKPOINT_REJECTED", "AGENT_CHECKPOINT_AMBIGUOUS", "AGENT_CHECKPOINT_TOO_LARGE", "AGENT_CHECKPOINT_STALE", "AGENT_PROVIDER_CIRCUIT_OPEN",
  696. 696 "AGENT_RUNTIME_BUSY", "AGENT_RUNTIME_DUPLICATE_RUN", "AGENT_RUNTIME_VERSION_MISMATCH", "AGENT_RUNTIME_PROTOCOL_ERROR", "AGENT_RUNTIME_BODY_TOO_LARGE", "AGENT_RUNTIME_PEER_REJECTED", "AGENT_RUNTIME_GATEWAY_REJECTED", "AGENT_RUNTIME_GATEWAY_UNAVAILABLE", "AGENT_RUNTIME_FAILED",
  697. 697 "STALE_LEASE", "STALE_ATTEMPT", "AUTHORIZATION_REVOKED", "DIAGNOSIS_CANCELED", "INPUT_CHANGED":
  698. 698 if d := response.Diagnostics; d != nil {
  699. 699 slog.Error("AgentScope execution failed", "reason", "runner_non_200", "error_code", response.ErrorCode, "stage", d.Stage, "exception_kind", d.ExceptionKind, "model_calls", d.ModelCalls, "tool_calls", d.ToolCalls)
  700. 700 } else {
  701. 701 slog.Error("AgentScope execution failed", "reason", "runner_non_200", "error_code", response.ErrorCode)
  702. 702 }
  703. 703 return domain.Fail(response.ErrorCode, "AgentScope阶段执行失败，请检查受控工具证据")
  704. 704 }
  705. 705 return fallback
  706. 706}

  736. 736// compileAgentTools 把 Go 权威工具定义编译为独立 JSON Schema，确保 Python 和 Go 两侧都验证参数。
  737. 737// 工具名称限定本阶段，不能从模型输出扩展；先转换 Go 容器类型再编译，避免 required 列表被误解。
  738. 738func compileAgentTools(phase string, tools []map[string]any) (map[string]*jsonschema.Schema, error) {
  739. 739 schemas := map[string]*jsonschema.Schema{}
  740. 740 for _, tool := range tools {
  741. 741 function, ok := tool["function"].(map[string]any)
  742. 742 if !ok {
  743. 743 return nil, domain.Fail("AGENT_RUNTIME_ERROR", "工具定义无效")
  744. 744 }
  745. 745 name, _ := function["name"].(string)
  746. 746 if !phaseTools(phase)[name] || schemas[name] != nil {
  747. 747 return nil, domain.Fail("AGENT_RUNTIME_ERROR", "工具定义越过阶段边界")
  748. 748 }
  749. 749 compiler := jsonschema.NewCompiler()
  750. 750 uri := "https://drop.invalid/agent-tools/" + name
  751. 751 // JSON Schema 编译器只接受标准 JSON 值类型。Go 中手写的 required 使用 []string，
  752. 752 // 必须先序列化再反序列化为 JSON 的 []any，才能正确编译 planning 与 repair 工具约束。
  753. 753 raw, err := json.Marshal(function["parameters"])
  754. 754 var document any
  755. 755 if err != nil || json.Unmarshal(raw, &document) != nil || compiler.AddResource(uri, document) != nil {
  756. 756 return nil, domain.Fail("AGENT_RUNTIME_ERROR", "工具参数定义无效")
  757. 757 }
  758. 758 schema, err := compiler.Compile(uri)
  759. 759 if err != nil {
  760. 760 return nil, domain.Fail("AGENT_RUNTIME_ERROR", "工具参数定义无效")
  761. 761 }
  762. 762 schemas[name] = schema
  763. 763 }
  764. 764 if len(schemas) == 0 {
  765. 765 return nil, domain.Fail("AGENT_RUNTIME_ERROR", "工具定义为空")
  766. 766 }
  767. 767 return schemas, nil
