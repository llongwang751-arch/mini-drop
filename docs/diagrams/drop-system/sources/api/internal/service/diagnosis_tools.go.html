  57. 57func diagnosisSymbolHint(name string) string {
  58. 58 name = strings.ToLower(name)
  59. 59 for _, group := range []struct {
  60. 60 hint string
  61. 61 matches []string
  62. 62 }{
  63. 63 {"lock_or_wait", []string{"futex", "mutex", "pthread_cond", "locksupport", "park", "semaphore"}},
  64. 64 {"garbage_collection", []string{"gcworker", "garbage", "g1collect", "zgc", "gc.collect", "runtime.gc"}},
  65. 65 {"allocation", []string{"malloc", "calloc", "realloc", "operator new", "allocobject", "runtime.mallocgc"}},
  66. 66 {"io_or_network", []string{"epoll", "poll_wait", "recv", "sendmsg", "read_syscall", "sys_read", "sys_write", "socket"}},
  67. 67 {"scheduler", []string{"schedule", "context_switch", "sched_yield", "runtime.gosched"}},
  68. 68 } {
  69. 69 for _, match := range group.matches {
  70. 70 if strings.Contains(name, match) {
  71. 71 return group.hint
  72. 72 }
  73. 73 }
  74. 74 }
  75. 75 return "unknown"
  76. 76}

  223. 223 case "propose_verification":
  224. 224 if params.Kind != "" && params.Kind != task.Kind {
