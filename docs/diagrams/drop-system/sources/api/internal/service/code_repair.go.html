  167. 167func (s *Suggestions) processCodeRepair(ctx context.Context) error {

  169. 169 return nil
  170. 170 }
  171. 171 var run CodeRepair

  178. 178 return tx.Raw("UPDATE code_repairs SET status='running',lease_owner=?,lease_expires_at=now()+interval '480 seconds',attempts=attempts+1 WHERE id=(SELECT id FROM code_repairs WHERE (status='pending' OR (status='running' AND lease_expires_at<now())) AND attempts<3 ORDER BY id FOR UPDATE SKIP LOCKED LIMIT 1) RETURNING *", owner).Scan(&run).Error
  179. 179 })

  181. 181 return err
  182. 182 }

  209. 209 }

  211. 211 if report != nil {
  212. 212 data = string(report)
  213. 213 }

  215. 215 if res.Error != nil {

  217. 217 }

  375. 375func (state *repairToolState) finish(ctx context.Context, content string) (json.RawMessage, error) {
