  44. 44func (s *Suggestions) AuthorizeDiagnosisSignals(ctx context.Context, u domain.User, run DiagnosisRun) error {
  45. 45 if run.TelemetryServiceID == "" {
  46. 46 return nil
  47. 47 }
  48. 48 current, err := s.Repo.User(ctx, u.ID)
  49. 49 if err != nil {
  50. 50 return err
  51. 51 }
  52. 52 if s.Observability == nil {
  53. 53 return domain.Fail("FORBIDDEN", "业务信号登记不可用，不能读取关联诊断")
  54. 54 }
  55. 55 var trace []DiagnosisStep
  56. 56 if json.Unmarshal(run.Trace, &trace) != nil {
  57. 57 return domain.Fail("INVALID_EVIDENCE", "诊断轨迹无法读取")
  58. 58 }
  59. 59 refs, err := diagnosticSignalRefs(trace)
  60. 60 if err != nil {
  61. 61 return err
  62. 62 }
  63. 63 services := s.Observability.registeredScope(run.TelemetryServiceID)
  64. 64 for _, ref := range refs {
  65. 65 if len(ref) != 2 || ref[0] != 'S' || ref[1] < '1' || ref[1] > '8' {
  66. 66 return domain.Fail("INVALID_EVIDENCE", "业务信号别名无效")
  67. 67 }
  68. 68 index := int(ref[1] - '1')
  69. 69 if index >= len(services) {
  70. 70 return domain.Fail("FORBIDDEN", "原业务信号作用域已改变")
  71. 71 }
  72. 72 service := services[index]
  73. 73 if err = s.authorizeTelemetryService(ctx, current, service.ID, service.AgentID, false); err != nil {
  74. 74 return domain.Fail("FORBIDDEN", "无权读取关联业务信号")
  75. 75 }
  76. 76 }
  77. 77 for _, step := range trace {
  78. 78 if step.Tool == "read_profile_summary" {
  79. 79 var metadata struct {
  80. 80 Revision string `json:"observability_revision"`
  81. 81 }
  82. 82 if json.Unmarshal(step.Observation, &metadata) != nil || metadata.Revision != s.Observability.Digest {
  83. 83 return domain.Fail("FORBIDDEN", "原业务信号登记版本已改变")
  84. 84 }
  85. 85 }
  86. 86 }
  87. 87 return nil
  88. 88}

  90. 90// AuthorizeVerificationSignals 覆盖复测报告中的基线/候选诊断副本。
  91. 91// 任务共享只授予profile操作权限，不能顺带授予被复制业务信号的读取权限；每个实际来源都走原诊断的当前授权检查。
  92. 92// 调用方仍须保留基线和候选任务的原有权限检查。没有业务信号的旧来源可读；缺少来源或登记配置则失败关闭。
  93. 93func (s *Suggestions) AuthorizeVerificationSignals(ctx context.Context, u domain.User, verification VerificationRun) error {
  94. 94 if s == nil || s.Repo == nil {
  95. 95 return domain.Fail("FORBIDDEN", "无法核验复测诊断来源")
  96. 96 }
  97. 97 sources := []struct{ diagnosisID, taskID int64 }{{verification.DiagnosisID, verification.BaselineTaskID}}
  98. 98 if verification.CandidateDiagnosisID != nil {
  99. 99 sources = append(sources, struct{ diagnosisID, taskID int64 }{*verification.CandidateDiagnosisID, verification.CandidateTaskID})
  100. 100 }
  101. 101 for _, source := range sources {
  102. 102 if source.diagnosisID <= 0 || source.taskID <= 0 {
  103. 103 return domain.Fail("INVALID_EVIDENCE", "复测诊断来源无效")
  104. 104 }
  105. 105 var run DiagnosisRun
  106. 106 if err := s.Repo.Context(ctx).Where("id=? AND task_id=?", source.diagnosisID, source.taskID).First(&run).Error; err != nil {
  107. 107 return domain.Fail("FORBIDDEN", "复测诊断来源无法核实")
  108. 108 }
  109. 109 if err := s.AuthorizeDiagnosisSignals(ctx, u, run); err != nil {
  110. 110 return err
  111. 111 }
  112. 112 }
  113. 113 return nil
  114. 114}

  116. 116func (s *Suggestions) authorizeInvestigationSignals(ctx context.Context, u domain.User, run Investigation) error {
  117. 117 var rows []DiagnosisRun
  118. 118 if err := s.Repo.Context(ctx).Raw(`SELECT * FROM diagnosis_runs WHERE id=? OR id IN
  119. 119 (SELECT candidate_diagnosis_id FROM verification_runs WHERE id=?)`, run.DiagnosisID, run.VerificationID).Scan(&rows).Error; err != nil {
  120. 120 return err
  121. 121 }
  122. 122 for _, diagnosis := range rows {
  123. 123 if err := s.AuthorizeDiagnosisSignals(ctx, u, diagnosis); err != nil {
  124. 124 return err
  125. 125 }
  126. 126 }
