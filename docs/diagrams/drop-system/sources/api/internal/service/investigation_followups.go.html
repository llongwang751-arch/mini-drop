  40. 40// boundInvestigation 只沿数据库中的真实来源关系查找调查，不从问题文本猜测目标。
  41. 41// 复测、补证分支均继承根调查身份；歧义关系必须拒绝，不能随便选一条。
  42. 42func (s *Suggestions) boundInvestigation(ctx context.Context, run DiagnosisRun) (Investigation, bool, error) {
  43. 43 var rows []Investigation
  44. 44 err := s.Repo.Context(ctx).Raw(`SELECT DISTINCT i.* FROM investigations i
  45. 45 LEFT JOIN verification_runs v ON v.id=i.verification_id
  46. 46 LEFT JOIN investigation_followups f ON f.investigation_id=i.id
  47. 47 WHERE (i.baseline_task_id=? OR v.candidate_task_id=? OR f.task_id=?)
  48. 48 AND i.requested_by=? LIMIT 2`, run.TaskID, run.TaskID, run.TaskID, run.RequestedBy).Scan(&rows).Error
  49. 49 if err != nil || len(rows) == 0 {
  50. 50 return Investigation{}, false, err
  51. 51 }
  52. 52 if len(rows) != 1 {
  53. 53 return Investigation{}, false, domain.Fail("INVALID_EVIDENCE", "诊断对应多个调查")
  54. 54 }
  55. 55 return rows[0], true, nil
  56. 56}

  113. 113func (s *Suggestions) proposeFollowup(ctx context.Context, run DiagnosisRun, args json.RawMessage, trace []DiagnosisStep) (json.RawMessage, error) {
  114. 114 var in struct {
  115. 115 CollectionPlan
  116. 116 EvidenceIDs []string `json:"evidence_ids"`
  117. 117 }
  118. 118 if !decodeAgentJSON(args, &in) || strings.TrimSpace(string(args)) == "null" || len(in.EvidenceIDs) < 1 || len(in.EvidenceIDs) > 4 {
  119. 119 return nil, domain.Fail("AI_TOOL_REJECTED", "追加采集需要有效计划和本轮证据")
  120. 120 }
  121. 121 for _, id := range in.EvidenceIDs {
  122. 122 if !slices.ContainsFunc(trace, func(step DiagnosisStep) bool {
  123. 123 return step.EvidenceID == id && (slices.Contains([]string{"inspect_hotspots", "inspect_call_paths", "check_collection_quality", "query_business_signals"}, step.Tool) || followupResourceMeasurement(step))
  124. 124 }) {
  125. 125 return nil, domain.Fail("AI_TOOL_REJECTED", "追加采集必须引用本轮实际观测")
  126. 126 }
  127. 127 }
  128. 128 if slices.Contains([]string{"SCRIPT", "BOLT", "BOLT_PROFILE", "COMPOSITE"}, in.Kind) {
  129. 129 return nil, domain.Fail("AI_TOOL_REJECTED", "此类工具需要独立配置，不接受自动补证建议")
  130. 130 }
  131. 131 i, found, err := s.boundInvestigation(ctx, run)
  132. 132 if err != nil {
  133. 133 return nil, err
  134. 134 }
  135. 135 if !found {
  136. 136 return json.Marshal(map[string]any{"available": false, "reason": "investigation_required"})
  137. 137 }
  138. 138 if i.Status == "canceled" || i.Status == "failed" {
  139. 139 return nil, domain.Fail("INVALID_STATE", "原调查已结束或取消")
  140. 140 }
  141. 141 var n int64
  142. 142 if err := s.Repo.Context(ctx).Model(&InvestigationFollowup{}).Where("investigation_id=?", i.ID).Count(&n).Error; err != nil {
  143. 143 return nil, err
  144. 144 }
  145. 145 if n >= 2 {
  146. 146 return json.Marshal(map[string]any{"available": false, "reason": "followup_budget_exhausted"})
  147. 147 }
  148. 148 u, err := s.Repo.User(ctx, run.RequestedBy)
  149. 149 if err != nil {
  150. 150 return nil, err
  151. 151 }
  152. 152 current, err := s.Repo.Task(ctx, u, run.TaskID)
  153. 153 if err != nil {
  154. 154 return nil, err
  155. 155 }
  156. 156 for _, step := range trace {
  157. 157 if slices.Contains(in.EvidenceIDs, step.EvidenceID) && step.Tool == "read_profile_summary" && current.Kind != "RESOURCE_SNAPSHOT" {
  158. 158 return nil, domain.Fail("AI_TOOL_REJECTED", "资源测量摘要必须与当前任务采集类型一致")
  159. 159 }
  160. 160 }
  161. 161 if _, err = s.validateCollectionPlan(ctx, u, i, in.CollectionPlan); err != nil {
  162. 162 return json.Marshal(map[string]any{"available": true, "validated": false, "error_code": safeAgentError(err), "next": "按实际采集能力和参数修正草案；缺少端口或工具配置时说明需要配置，不猜测。"})
  163. 163 }
  164. 164 if in.Kind == current.Kind {
  165. 165 return json.Marshal(map[string]any{"available": true, "validated": false, "reason": "current_collector_already_observed", "next": "当前采集器已有观测；同类复采请调用propose_verification提出受控复测，其他信号才使用补证采集。"})
  166. 166 }
  167. 167 return json.Marshal(FollowupProposal{Plan: in.CollectionPlan, EvidenceIDs: in.EvidenceIDs, RequiresConfirmation: true})
  168. 168}

  203. 203// DecideInvestigationFollowup 把审阅决定和审计放在同一事务。
  204. 204// 统一按调查→来源任务→分支加锁；确认后仅入队，后台仍重新核验能力、目标启动身份和来源轮次。
  205. 205func (s *Suggestions) DecideInvestigationFollowup(ctx context.Context, u domain.User, id, followupID int64, confirm bool, request string) (InvestigationFollowup, error) {
  206. 206 var out InvestigationFollowup
  207. 207 err := s.Repo.Context(ctx).Transaction(func(tx *gorm.DB) error {
  208. 208 local := *s
  209. 209 local.Repo = repository.New(tx)
  210. 210 var root Investigation
  211. 211 if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).Where("id=?", id).First(&root).Error; err != nil {
  212. 212 return err
  213. 213 }
  214. 214 current, err := local.Repo.User(ctx, u.ID)
  215. 215 if err != nil {
  216. 216 return err
  217. 217 }
  218. 218 if _, err = local.AuthorizeInvestigation(ctx, current, id); err != nil {
  219. 219 return err
  220. 220 }
  221. 221 if !current.CanWrite() || root.Status == "canceled" || root.Status == "failed" {
  222. 222 return domain.Fail("INVALID_STATE", "当前调查不接受追加采集")
  223. 223 }
  224. 224 if err = tx.Where("id=? AND investigation_id=?", followupID, id).First(&out).Error; err != nil {
  225. 225 return err
  226. 226 }
  227. 227 var parent domain.Task
  228. 228 if err = tx.Clauses(clause.Locking{Strength: "UPDATE"}).Where("id=?", out.ParentTaskID).First(&parent).Error; err != nil {
  229. 229 return err
  230. 230 }
  231. 231 if err = tx.Clauses(clause.Locking{Strength: "UPDATE"}).Where("id=? AND investigation_id=?", followupID, id).First(&out).Error; err != nil {
  232. 232 return err
  233. 233 }
  234. 234 if out.Status != "proposed" {
  235. 235 if confirm && out.Status != "rejected" && out.Status != "canceled" || !confirm && out.Status == "rejected" {
  236. 236 return nil
  237. 237 }
  238. 238 return domain.Fail("INVALID_STATE", "此补证计划已经处理")
  239. 239 }
  240. 240 if confirm {
  241. 241 if time.Since(root.CreatedAt) > 20*time.Minute {
  242. 242 return domain.Fail("DIAGNOSIS_TIMEOUT", "追加采集超出调查时间预算")
  243. 243 }
  244. 244 var diagnosis DiagnosisRun
  245. 245 if err = tx.Where("id=? AND status='success'", out.SourceDiagnosisID).First(&diagnosis).Error; err != nil {
  246. 246 return err
  247. 247 }
  248. 248 if err = local.ValidateDiagnosisSource(ctx, diagnosis); err != nil {
  249. 249 return err
  250. 250 }
  251. 251 var proposal FollowupProposal
  252. 252 if json.Unmarshal(out.Plan, &proposal) != nil {
  253. 253 return domain.Fail("INVALID_EVIDENCE", "补证计划不可读")
  254. 254 }
  255. 255 if _, err = local.validateCollectionPlan(ctx, current, root, proposal.Plan); err != nil {
  256. 256 return err
  257. 257 }
  258. 258 out.Status = "queued"
  259. 259 } else {
  260. 260 out.Status = "rejected"
  261. 261 }
  262. 262 out.UpdatedAt = time.Now().UTC()
  263. 263 if err = tx.Model(&out).Updates(map[string]any{"status": out.Status, "updated_at": out.UpdatedAt}).Error; err != nil {
  264. 264 return err
  265. 265 }
  266. 266 return tx.Create(ptrAuditDiagnosis(Audit(current, "investigation.followup_decision", "investigation", fmt.Sprint(id), request))).Error
  267. 267 })
  268. 268 return out, err
  269. 269}
