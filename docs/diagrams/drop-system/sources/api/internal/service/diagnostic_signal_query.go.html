  83. 83// collectionSignalWindows 只为真实执行区间聚合指标。上下文扩窗只提供元数据，
  84. 84// 不能将此前正常请求混进本轮P95，也不能把attempt包含的准备/上传时间称为精确采样时钟。
  85. 85func collectionSignalWindows(attempt domain.Attempt, now time.Time) (diagnosticSignalWindows, error) {
  86. 86 fail := func() (diagnosticSignalWindows, error) {
  87. 87 return diagnosticSignalWindows{}, domain.Fail("SIGNAL_WINDOW_UNKNOWN", "没有可核实且有界的采集执行时钟")
  88. 88 }
  89. 89 if attempt.StartedAt == nil || attempt.FinishedAt == nil || !attempt.FinishedAt.After(*attempt.StartedAt) || attempt.FinishedAt.Sub(*attempt.StartedAt) > 13*time.Minute || attempt.FinishedAt.After(now.Add(time.Minute)) {
  90. 90 return fail()
  91. 91 }
  92. 92 result := diagnosticSignalWindows{Collection: SignalWindow{attempt.StartedAt.UTC(), attempt.FinishedAt.UTC()}, Context: SignalWindow{attempt.StartedAt.Add(-time.Minute).UTC(), attempt.FinishedAt.Add(time.Minute).UTC()}, Scope: "attempt_execution"}
  93. 93 var metadata map[string]json.RawMessage
  94. 94 raw := bytes.TrimSpace(attempt.RunnerMetadata)
  95. 95 if len(attempt.RunnerMetadata) > 32768 || len(attempt.RunnerMetadata) != 0 && len(raw) == 0 || len(raw) != 0 && (json.Unmarshal(raw, &metadata) != nil || metadata == nil) {
  96. 96 return fail()
  97. 97 }
  98. 98 startRaw, hasStart := metadata["collection_started_unix_ms"]
  99. 99 endRaw, hasEnd := metadata["collection_finished_unix_ms"]
  100. 100 if hasStart != hasEnd {
  101. 101 return fail()
  102. 102 }
  103. 103 if !hasStart {
  104. 104 return result, nil
  105. 105 }
  106. 106 var startMS, endMS *int64
  107. 107 if json.Unmarshal(startRaw, &startMS) != nil || json.Unmarshal(endRaw, &endMS) != nil || startMS == nil || endMS == nil || *startMS <= 0 || *endMS <= *startMS {
  108. 108 return fail()
  109. 109 }
  110. 110 start, end := time.UnixMilli(*startMS).UTC(), time.UnixMilli(*endMS).UTC()
  111. 111 if *startMS < attempt.StartedAt.UnixMilli() || *endMS > attempt.FinishedAt.UnixMilli() {
  112. 112 return fail()
  113. 113 }
  114. 114 // Agent/helper的时钟为毫秒精度；首毫秒与精确attempt起点取交集，
  115. 115 // 不因量化误差读到attempt以前，也不引入任意的时钟漂移容差。
  116. 116 if start.Before(*attempt.StartedAt) {
  117. 117 start = attempt.StartedAt.UTC()
  118. 118 }
  119. 119 if !end.After(start) {
  120. 120 return fail()
  121. 121 }
  122. 122 result.Collection, result.Scope = SignalWindow{start, end}, "collector_execution"
  123. 123 return result, nil
  124. 124}

  139. 139func (s *Suggestions) queryBusinessSignals(ctx context.Context, run DiagnosisRun, args json.RawMessage) (json.RawMessage, error) {
  140. 140 var in struct {
  141. 141 ServiceRef string `json:"service_ref"`
  142. 142 Signal string `json:"signal"`
  143. 143 }
  144. 144 if !decodeAgentJSON(args, &in) || len(in.ServiceRef) != 2 || in.ServiceRef[0] != 'S' || in.ServiceRef[1] < '1' || in.ServiceRef[1] > '8' || !slices.Contains([]string{"metrics", "logs", "traces", "changes"}, in.Signal) {
  145. 145 return nil, domain.Fail("AI_TOOL_REJECTED", "业务信号参数无效")
  146. 146 }
  147. 147 services, err := s.telemetryScope(ctx, run)
  148. 148 if err != nil {
  149. 149 return nil, err
  150. 150 }
  151. 151 out := BusinessSignalObservation{ServiceRef: in.ServiceRef, Signal: in.Signal, Limitations: []string{"关联窗口内同时出现不证明因果；缺失和截断不能解释为正常。"}}
  152. 152 index := int(in.ServiceRef[1] - '1')
  153. 153 if index >= len(services) {
  154. 154 out.Reason = "service_not_bound"
  155. 155 return json.Marshal(out)
  156. 156 }
  157. 157 if services[index].ID == "" {
  158. 158 out.Reason = "service_not_authorized"
  159. 159 return json.Marshal(out)
  160. 160 }
  161. 161 connector, exists := services[index].Connectors[in.Signal]
  162. 162 if !exists {
  163. 163 out.Reason = "signal_not_configured"
  164. 164 return json.Marshal(out)
  165. 165 }
  166. 166 windows, err := s.diagnosticSignalWindows(ctx, run)
  167. 167 if err != nil {
  168. 168 out.Reason = "collection_window_unknown"
  169. 169 out.Limitations = append(out.Limitations, "缺少有效采集执行起止时钟，不使用创建时间或分析完成时间替代。")
  170. 170 return json.Marshal(out)
  171. 171 }
  172. 172 out.Window, out.WindowScope, out.ContextWindow = windows.Collection, windows.Scope, &windows.Context
  173. 173 out.Limitations = append(out.Limitations, "指标只按window内的采集执行区间聚合；context_window是旧扩展上下文范围，仅供说明，本次未查询或混入其指标。")
  174. 174 if windows.Scope == "attempt_execution" {
  175. 175 out.Limitations = append(out.Limitations, "旧任务缺少collector起止元数据，使用attempt执行区间；它可能含准备与上传，不是精确采样时钟。")
  176. 176 } else {
  177. 177 out.Limitations = append(out.Limitations, "collector_execution来自Agent/helper记录的有界执行起止；它仍不是逐个sample的采样时间戳。")
  178. 178 }
  179. 179 parent := ctx
  180. 180 ctx, cancel := context.WithTimeout(ctx, 8*time.Second)
  181. 181 defer cancel()
  182. 182 out.SourceKind = connector.Kind
  183. 183 switch connector.Kind {
  184. 184 case "prometheus":
  185. 185 err = s.queryPrometheus(ctx, connector, &out)
  186. 186 case "loki":
  187. 187 err = s.queryLoki(ctx, connector, &out)
  188. 188 case "tempo":
  189. 189 err = s.queryTempo(ctx, connector, &out)
  190. 190 case "drop-adapter":
  191. 191 err = s.querySignalAdapter(ctx, connector, services[index].ID, &out)
  192. 192 default:
  193. 193 err = domain.Fail("SIGNAL_SOURCE_INVALID", "数据源类型无效")
  194. 194 }
  195. 195 if err != nil {
  196. 196 if parent.Err() != nil {
  197. 197 return nil, parent.Err()
  198. 198 }
  199. 199 out.Available = false
  200. 200 out.Metrics = nil
  201. 201 out.Logs = nil
  202. 202 out.Traces = nil
  203. 203 out.Changes = nil
  204. 204 out.Reason = safeAgentError(err)
  205. 205 out.Limitations = append(out.Limitations, "本数据源本次未提供可用观测，不能据此判断正常。")
  206. 206 return json.Marshal(out)
  207. 207 }
  208. 208 out.Available = true
  209. 209 if len(out.Metrics)+len(out.Logs)+len(out.Traces)+len(out.Changes) == 0 {
  210. 210 out.Reason = "no_matching_data"
  211. 211 out.Limitations = append(out.Limitations, "当前窗口未返回记录，不表示业务没有异常。")
  212. 212 }
  213. 213 return json.Marshal(out)
  214. 214}

  216. 216func (s *Suggestions) inspectServiceTopology(ctx context.Context, run DiagnosisRun, args json.RawMessage) (json.RawMessage, error) {
  217. 217 if err := emptyDiagnosticKnowledgeArgs(args); err != nil {
  218. 218 return nil, err
  219. 219 }
  220. 220 services, err := s.telemetryScope(ctx, run)
  221. 221 if err != nil {
  222. 222 return nil, err
  223. 223 }
  224. 224 nodes := []map[string]any{}
  225. 225 edges := []map[string]string{}
  226. 226 refs := map[string]string{}
  227. 227 for index, service := range services {
  228. 228 if service.ID == "" {
  229. 229 continue
  230. 230 }
  231. 231 refs[service.ID] = fmt.Sprintf("S%d", index+1)
  232. 232 }
  233. 233 for _, service := range services {
  234. 234 if service.ID == "" {
  235. 235 continue
  236. 236 }
  237. 237 types := []string{}
  238. 238 for signal := range service.Connectors {
  239. 239 types = append(types, signal)
  240. 240 }
  241. 241 slices.Sort(types)
  242. 242 nodes = append(nodes, map[string]any{"service_ref": refs[service.ID], "signal_types": types})
  243. 243 for _, dependency := range service.Dependencies {
  244. 244 if to, exists := refs[dependency]; exists {
  245. 245 edges = append(edges, map[string]string{"from": refs[service.ID], "to": to, "source": "registered_dependency"})
  246. 246 }
  247. 247 }
  248. 248 }
  249. 249 return json.Marshal(map[string]any{"available": len(nodes) > 0, "nodes": nodes, "edges": edges, "current_service_ref": "S1", "causality_proven": false, "limitations": []string{"登记的一跳依赖用于选择调查方向；必须再用同时段信号验证，不能仅由拓扑确认故障传播。"}})
  250. 250}

  252. 252func (s *Suggestions) queryPrometheus(ctx context.Context, c TelemetryConnector, out *BusinessSignalObservation) error {
  253. 253 names := []string{}
  254. 254 for name := range c.Queries {
  255. 255 names = append(names, name)
  256. 256 }
  257. 257 slices.Sort(names)
  258. 258 for _, name := range names {
  259. 259 query := c.Queries[name]
  260. 260 parameters := url.Values{"query": {query}, "start": {strconv.FormatInt(out.Window.Start.Unix(), 10)}, "end": {strconv.FormatInt(out.Window.End.Unix(), 10)}, "step": {"15"}, "timeout": {"3s"}}
  261. 261 data, err := s.Observability.fetch(ctx, c, "/api/v1/query_range", parameters)
  262. 262 if err != nil {
  263. 263 return err
  264. 264 }
  265. 265 var response struct {
  266. 266 Status string `json:"status"`
  267. 267 Data struct {
  268. 268 ResultType string `json:"resultType"`
  269. 269 Result []struct {
  270. 270 Values [][]json.RawMessage `json:"values"`
  271. 271 } `json:"result"`
  272. 272 } `json:"data"`
  273. 273 }
  274. 274 if json.Unmarshal(data, &response) != nil || response.Status != "success" || response.Data.ResultType != "matrix" || len(response.Data.Result) > 1 {
  275. 275 return domain.Fail("INVALID_EVIDENCE", "业务指标必须返回已聚合的单一时序")
  276. 276 }
  277. 277 if len(response.Data.Result) == 0 {
  278. 278 continue
  279. 279 }
  280. 280 values := response.Data.Result[0].Values
  281. 281 if len(values) > 64 {
  282. 282 return domain.Fail("INVALID_EVIDENCE", "业务指标点数超出预算")
  283. 283 }
  284. 284 if len(values) == 0 {
  285. 285 continue
  286. 286 }
  287. 287 last := values[len(values)-1]
  288. 288 if len(last) != 2 {
  289. 289 return domain.Fail("INVALID_EVIDENCE", "业务指标格式无效")
  290. 290 }
  291. 291 var raw string
  292. 292 if json.Unmarshal(last[1], &raw) != nil {
  293. 293 return domain.Fail("INVALID_EVIDENCE", "业务指标格式无效")
  294. 294 }
  295. 295 value, err := strconv.ParseFloat(raw, 64)
  296. 296 if err != nil || !finiteSignal(value) || name == "error_rate" && value > 1 {
  297. 297 return domain.Fail("INVALID_EVIDENCE", "业务指标单位或范围无效")
  298. 298 }
  299. 299 var seconds float64
  300. 300 if json.Unmarshal(last[0], &seconds) != nil || !finiteSignal(seconds) {
  301. 301 return domain.Fail("INVALID_EVIDENCE", "业务指标时间无效")
  302. 302 }
  303. 303 stamp := time.UnixMilli(int64(seconds * 1000)).UTC()
  304. 304 if stamp.Before(out.Window.Start) || stamp.After(out.Window.End) {
  305. 305 return domain.Fail("INVALID_EVIDENCE", "业务指标不在当前时间窗口")
  306. 306 }
  307. 307 out.Metrics = append(out.Metrics, TelemetryMetric{Name: name, Value: value, Unit: metricUnit(name), ObservedAt: &stamp})
  308. 308 }
  309. 309 out.QueryDigest = repository.DigestBytes(mustSignalJSON(c.Queries))
  310. 310 out.Limitations = append(out.Limitations, "指标为管理员登记聚合查询在窗口末端返回的值，不与profile权重直接换算。")
  311. 311 return nil
  312. 312}

  316. 316func normalizeTelemetryLog(timestamp time.Time, raw []byte, window SignalWindow) (TelemetryLog, bool) {
  317. 317 if timestamp.Before(window.Start) || timestamp.After(window.End) || len(raw) > 8192 {
  318. 318 return TelemetryLog{}, false
  319. 319 }
  320. 320 var input struct {
  321. 321 Level string `json:"level"`
  322. 322 EventCode string `json:"event_code"`
  323. 323 DurationMS *float64 `json:"duration_ms"`
  324. 324 StatusCode *int `json:"status_code"`
  325. 325 TraceID string `json:"trace_id"`
  326. 326 }
  327. 327 if json.Unmarshal(raw, &input) != nil {
  328. 328 return TelemetryLog{}, false
  329. 329 }
  330. 330 level := strings.ToLower(input.Level)
  331. 331 if !slices.Contains([]string{"debug", "info", "warn", "error", "fatal"}, level) {
  332. 332 level = "unknown"
  333. 333 }
  334. 334 code := input.EventCode
  335. 335 if !slices.Contains([]string{"REQUEST_COMPLETE", "UPSTREAM_TIMEOUT", "DB_SLOW_QUERY", "CACHE_MISS", "GC_PRESSURE", "DEPLOYMENT", "UPSTREAM_ERROR"}, code) {
  336. 336 code = "UNKNOWN"
  337. 337 }
  338. 338 out := TelemetryLog{Timestamp: timestamp.UTC(), Level: level, EventCode: code}
  339. 339 if input.DurationMS != nil && finiteSignal(*input.DurationMS) {
  340. 340 out.DurationMS = input.DurationMS
  341. 341 }
  342. 342 if input.StatusCode != nil && *input.StatusCode >= 100 && *input.StatusCode <= 599 {
  343. 343 out.StatusCode = input.StatusCode
  344. 344 }
  345. 345 if len(input.TraceID) >= 16 && len(input.TraceID) <= 64 {
  346. 346 out.TraceRef = "trace_" + repository.Digest(input.TraceID)[:12]
  347. 347 }
  348. 348 return out, true
  349. 349}

  383. 383 out.Logs = append(out.Logs, row)

  385. 385 }

  387. 387 out.Truncated = out.Truncated || len(out.Logs) == 30
  388. 388 out.QueryDigest = repository.Digest(c.Selector)
  389. 389 out.Limitations = append(out.Limitations, "仅解析JSON日志中的固定结构字段；原日志正文及自由属性不发送模型。")
  390. 390 return nil
