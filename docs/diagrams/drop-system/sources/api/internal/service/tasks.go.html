  218. 218// DownloadGrant 重新核验任务和制品权限、归属及就绪状态，并生成带审计的短期下载授权。
  219. 219// 过期时间属于本次请求，浏览器不能把一个 URL 当作永久资源权限。
  220. 220func (s *Tasks) DownloadGrant(ctx context.Context, u domain.User, id, artifactID int64, request string) (string, time.Time, error) {
  221. 221 if _, err := s.Authorize(ctx, u, id); err != nil {
  222. 222 return "", time.Time{}, err
  223. 223 }
  224. 224 as, err := s.Repo.Artifacts(ctx, id)
  225. 225 if err != nil {
  226. 226 return "", time.Time{}, err
  227. 227 }
  228. 228 for _, a := range as {
  229. 229 if a.ID == artifactID && a.Status == "ready" {
  230. 230 if !a.DownloadAllowed || (a.ExpiresAt != nil && !a.ExpiresAt.After(time.Now())) {
  231. 231 return "", time.Time{}, domain.Fail("ARTIFACT_EXPIRED", "制品已过期或禁止下载")
  232. 232 }
  233. 233 ttl := 5 * time.Minute
  234. 234 if a.ExpiresAt != nil && time.Until(*a.ExpiresAt) < ttl {
  235. 235 ttl = time.Until(*a.ExpiresAt)
  236. 236 }
  237. 237 if ttl < time.Second {
  238. 238 return "", time.Time{}, domain.Fail("ARTIFACT_EXPIRED", "制品保留期即将结束")
  239. 239 }
  240. 240 expires := time.Now().UTC().Add(ttl)
  241. 241 var url string
  242. 242 var err error
  243. 243 if store, ok := s.Storage.(interface {
  244. 244 PresignGetFor(context.Context, string, time.Duration) (string, error)
  245. 245 }); ok {
  246. 246 url, err = store.PresignGetFor(ctx, a.ObjectKey, ttl)
  247. 247 } else {
  248. 248 if ttl < 5*time.Minute {
  249. 249 return "", time.Time{}, domain.Fail("ARTIFACT_EXPIRED", "剩余保留期不足")
  250. 250 }
  251. 251 url, err = s.Storage.PresignGet(ctx, a.ObjectKey)
  252. 252 }
  253. 253 if err != nil {
  254. 254 return "", time.Time{}, err
  255. 255 }
  256. 256 if err := s.Repo.Audit(ctx, Audit(u, "artifact.download", "artifact", strconv.FormatInt(a.ID, 10), request)); err != nil {
  257. 257 return "", time.Time{}, err
  258. 258 }
  259. 259 return url, expires, nil
  260. 260 }
  261. 261 }
  262. 262 return "", time.Time{}, domain.Fail("NOT_FOUND", "制品不存在或尚未就绪")
  263. 263}

  265. 265// Run 启动派发、定时计划、组合推进和清理等有界后台循环；共享 context 用于优雅退出，工作事实仍以数据库为准。
  266. 266func (s *Tasks) Run(ctx context.Context) {
  267. 267 go tick(ctx, time.Second, "COMPOSITE_ERROR", s.TickComposite)
  268. 268 go tick(ctx, time.Second, "SCHEDULE_ERROR", s.schedule)
  269. 269 go tick(ctx, time.Second, "IMPORT_RECOVERY_ERROR", s.Repo.SweepStaleImport)
  270. 270 go tick(ctx, time.Minute, "EPHEMERAL_CLEANUP_ERROR", s.Repo.CleanupEphemeral)
  271. 271 ticker := time.NewTicker(time.Second)
  272. 272 defer ticker.Stop()
  273. 273 for {
  274. 274 select {
  275. 275 case <-ctx.Done():
  276. 276 return
  277. 277 case <-ticker.C:
  278. 278 for i := 0; i < 10; i++ {
  279. 279 if err := s.dispatch(ctx); err != nil {
  280. 280 slog.Error("outbox dispatch failed", "code", "OUTBOX_ERROR")
  281. 281 break
  282. 282 }
  283. 283 }
  284. 284 }
  285. 285 }
  286. 286}

  288. 288// tick 用固定间隔驱动一次性工作函数，隔离单轮失败；循环只记录安全错误类别，退出由 context 统一控制。
  289. 289func tick(ctx context.Context, interval time.Duration, code string, fn func(context.Context) error) {
  290. 290 ticker := time.NewTicker(interval)
  291. 291 defer ticker.Stop()
  292. 292 for {
  293. 293 select {
  294. 294 case <-ctx.Done():
  295. 295 return
  296. 296 case <-ticker.C:
  297. 297 if err := fn(ctx); err != nil && ctx.Err() == nil {
  298. 298 slog.Error("background operation failed", "code", code)
  299. 299 }
  300. 300 }
  301. 301 }
  302. 302}

  304. 304// dispatch 根据 Outbox 中的任务和 attempt 构造类型化描述，补充临时对象授权后调用 Core。
  305. 305// Core 可能已接受但 ACK 未达，最终状态由仓储的重读栅栏决定，不能只看这次 RPC 返回值。
  306. 306func (s *Tasks) dispatch(ctx context.Context) error {
  307. 307 return s.Repo.DispatchOne(ctx, func(o domain.Outbox, t domain.Task, a domain.Attempt) error {
  308. 308 rpcCtx, cancel := context.WithTimeout(ctx, 4*time.Second)
  309. 309 defer cancel()
  310. 310 linked := otel.GetTextMapPropagator().Extract(ctx, propagation.MapCarrier{"traceparent": t.TraceParent})
  311. 311 spanCtx := trace.SpanContextFromContext(linked)
  312. 312 opts := []trace.SpanStartOption{}
  313. 313 if spanCtx.IsValid() {
  314. 314 opts = append(opts, trace.WithLinks(trace.Link{SpanContext: spanCtx}))
  315. 315 }
  316. 316 rpcCtx, span := otel.Tracer("drop/api/outbox").Start(rpcCtx, "task.dispatch", opts...)
  317. 317 defer span.End()
  318. 318 span.SetAttributes(telemetry.TaskAttributes(t.ID)...)
  319. 319 if o.EventType == "CANCEL" {
  320. 320 return s.Control.Cancel(rpcCtx, t.ID, o.AttemptID, t.RequestID)
  321. 321 }
  322. 322 url, err := s.Storage.PresignPut(rpcCtx, a.RawObjectKey)
  323. 323 if err != nil {
  324. 324 return err
  325. 325 }
  326. 326 if err := s.Repo.SetUploadURL(rpcCtx, a.ID, url); err != nil {
  327. 327 return err
  328. 328 }
  329. 329 var p domain.Parameters
  330. 330 if err := json.Unmarshal(t.Parameters, &p); err != nil {
  331. 331 return err
  332. 332 }
  333. 333 agent := ""
  334. 334 if t.AgentID != nil {
  335. 335 agent = *t.AgentID
  336. 336 }
  337. 337 deadline := time.Now().Add(10 * time.Minute)
  338. 338 if t.DeadlineAt != nil {
  339. 339 deadline = *t.DeadlineAt
  340. 340 }
  341. 341 desc := &pb.TaskDesc{TaskId: t.ID, AttemptId: a.ID, Kind: t.Kind, Priority: int32(t.Priority), TargetAgentId: agent, Parameters: &pb.TaskParameters{Pid: int32(p.PID), DurationSeconds: int32(p.DurationSeconds), FrequencyHz: int32(p.FrequencyHz), Scope: p.Scope}, DeadlineUnixMs: deadline.UnixMilli(), RawUploadUrl: url, RawObjectKey: a.RawObjectKey, RequestId: t.RequestID, MaxFileBytes: 32 << 20}
  342. 342 kind, _ := domain.Kind(t.Kind)
  343. 343 desc.Traceparent = t.TraceParent
  344. 344 if err := s.enrichDispatch(rpcCtx, desc, p, kind, a); err != nil {
  345. 345 return err
  346. 346 }

  348. 348 })
  349. 349}
