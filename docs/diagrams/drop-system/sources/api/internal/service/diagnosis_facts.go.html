  1086. 1086 if err := diagnosisReportErrors(validationErrors); err != nil {
  1087. 1087 return nil, err // 一处错误也不能返回局部报告或由程序代替模型修改内容。
  1088. 1088 }
  1089. 1089 report.Summary, report.Limitations, err = diagnosisV2Summary(trace)
  1090. 1090 if err != nil {
  1091. 1091 return nil, err
  1092. 1092 }
  1093. 1093 if err := validateDiagnosticSkillUse(trace); err != nil {
