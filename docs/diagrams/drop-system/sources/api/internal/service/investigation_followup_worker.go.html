  20. 20 EXISTS(SELECT 1 FROM investigation_followups f WHERE f.investigation_id=i.id AND f.status IN ('queued','collecting','diagnosing'))
  21. 21 OR ((i.created_at<=now()-interval '20 minutes' OR i.status IN ('failed','canceled')) AND
  22. 22 EXISTS(SELECT 1 FROM investigation_followups f WHERE f.investigation_id=i.id AND f.status='proposed'))

  24. 24 AND (SELECT count(*) FROM investigation_followups f WHERE f.investigation_id=i.id)<2

  29. 29 AND NOT EXISTS(SELECT 1 FROM investigation_followups f WHERE f.investigation_id=i.id AND f.source_diagnosis_id=d.id)))

  98. 98func (s *Suggestions) stageFollowupPlans(ctx context.Context, tx *gorm.DB, root Investigation) error {
  99. 99 var count int64
  100. 100 if err := tx.Model(&InvestigationFollowup{}).Where("investigation_id=?", root.ID).Count(&count).Error; err != nil {
  101. 101 return err
  102. 102 }
  103. 103 if count >= 2 {
  104. 104 return nil
  105. 105 }
  106. 106 var runs []DiagnosisRun
  107. 107 if err := tx.Raw(`SELECT DISTINCT d.* FROM diagnosis_runs d WHERE d.status='success' AND d.requested_by=? AND
  108. 108 (d.id=? OR d.id IN (SELECT candidate_diagnosis_id FROM verification_runs WHERE id=?)
  109. 109 OR d.id IN (SELECT diagnosis_id FROM investigation_followups WHERE investigation_id=?)) ORDER BY d.id LIMIT 6`, root.RequestedBy, root.DiagnosisID, root.VerificationID, root.ID).Scan(&runs).Error; err != nil {
  110. 110 return err
  111. 111 }
  112. 112 for _, run := range runs {
  113. 113 var report DiagnosisReport
  114. 114 if json.Unmarshal(run.Report, &report) != nil || len(report.FollowupPlans) == 0 {
  115. 115 continue
  116. 116 }
  117. 117 if err := s.ValidateDiagnosisSource(ctx, run); err != nil {
  118. 118 continue
  119. 119 }
  120. 120 for _, plan := range report.FollowupPlans {
  121. 121 if count >= 2 {
  122. 122 return nil
  123. 123 }
  124. 124 if !plan.RequiresConfirmation {
  125. 125 continue
  126. 126 }
  127. 127 data, _ := json.Marshal(plan)
  128. 128 digest := repository.DigestBytes(data)
  129. 129 var exists int64
  130. 130 if err := tx.Model(&InvestigationFollowup{}).Where("investigation_id=? AND source_diagnosis_id=? AND plan_digest=?", root.ID, run.ID, digest).Count(&exists).Error; err != nil {
  131. 131 return err
  132. 132 }
  133. 133 if exists > 0 {
  134. 134 continue
  135. 135 }
  136. 136 row := InvestigationFollowup{InvestigationID: root.ID, ParentTaskID: run.TaskID, SourceDiagnosisID: run.ID, Plan: data, PlanDigest: digest, Status: "proposed"}
  137. 137 if err := tx.Create(&row).Error; err != nil {
  138. 138 return err
  139. 139 }
  140. 140 if err := tx.Create(ptrAuditDiagnosis(Audit(domain.User{ID: root.RequestedBy}, "investigation.followup_proposed", "investigation", fmt.Sprint(root.ID), "investigation-worker"))).Error; err != nil {
  141. 141 return err
  142. 142 }

  144. 144 }
  145. 145 }

  149. 149func (s *Suggestions) advanceFollowup(ctx context.Context, tx *gorm.DB, u domain.User, root Investigation, row *InvestigationFollowup) error {
  150. 150 if time.Since(root.CreatedAt) > 20*time.Minute {
  151. 151 return domain.Fail("DIAGNOSIS_TIMEOUT", "补证超出调查时间预算")
  152. 152 }
  153. 153 if err := tx.Exec("SELECT pg_advisory_xact_lock(hashtextextended(?,417626))", root.RequestedBy).Error; err != nil {
  154. 154 return err
  155. 155 }
  156. 156 var source DiagnosisRun
  157. 157 if err := tx.Where("id=? AND status='success'", row.SourceDiagnosisID).First(&source).Error; err != nil {
  158. 158 return err

  160. 160 var parent domain.Task
  161. 161 if err := tx.Raw("SELECT * FROM tasks WHERE id=? FOR UPDATE", row.ParentTaskID).Scan(&parent).Error; err != nil {
  162. 162 return err
  163. 163 }
  164. 164 if _, err := s.Repo.AuthorizeTaskWrite(ctx, u, row.ParentTaskID); err != nil {
  165. 165 return err
  166. 166 }
  167. 167 if err := s.ValidateDiagnosisSource(ctx, source); err != nil {
  168. 168 return err
  169. 169 }
  170. 170 if row.Status == "queued" {
  171. 171 var p FollowupProposal
  172. 172 if json.Unmarshal(row.Plan, &p) != nil {
  173. 173 return domain.Fail("INVALID_EVIDENCE", "补证计划不可读")
  174. 174 }
  175. 175 input, err := s.validateCollectionPlan(ctx, u, root, p.Plan)
  176. 176 if err != nil {
  177. 177 return err
  178. 178 }

  180. 180 task, _, err := s.Tasks.Create(ctx, u, input, fmt.Sprintf("investigation:%d:followup:%d", root.ID, row.ID), "investigation-worker")
  181. 181 if err != nil {
  182. 182 return err
  183. 183 }
  184. 184 row.TaskID = &task.ID
  185. 185 row.Status = "collecting"
  186. 186 return nil
  187. 187 }
  188. 188 if row.TaskID == nil {
  189. 189 return domain.Fail("INVALID_STATE", "补证任务缺失")
  190. 190 }
  191. 191 task, err := s.Repo.AuthorizeTaskWrite(ctx, u, *row.TaskID)
  192. 192 if err != nil {
  193. 193 return err
  194. 194 }
  195. 195 if task.CollectionStatus == "FAILED" || task.CollectionStatus == "CANCELED" || task.AnalysisStatus == "FAILED" {
  196. 196 return domain.Fail("COLLECTION_FAILED", "补证采集或分析失败")
  197. 197 }
  198. 198 if row.Status == "collecting" {
  199. 199 if task.CollectionStatus != "COLLECTED" || task.AnalysisStatus != "SUCCESS" {
  200. 200 return nil
  201. 201 }
  202. 202 diagnosis, err := s.RequestDiagnosis(ctx, u, task.ID, DiagnosisInput{Goal: root.Goal, Mode: "model", Consent: root.Consent, TelemetryServiceID: root.TelemetryServiceID}, "investigation-worker", fmt.Sprintf("followup:%d:diagnosis", row.ID))
  203. 203 if err != nil {
  204. 204 return err
  205. 205 }
  206. 206 row.DiagnosisID = &diagnosis.ID
  207. 207 row.Status = "diagnosing"
  208. 208 return nil
  209. 209 }
  210. 210 if row.DiagnosisID == nil {
  211. 211 return domain.Fail("INVALID_STATE", "补证诊断缺失")
  212. 212 }
  213. 213 var diagnosis DiagnosisRun
  214. 214 if err := tx.Where("id=?", *row.DiagnosisID).First(&diagnosis).Error; err != nil {
  215. 215 return err
  216. 216 }
  217. 217 if diagnosis.Status == "failed" || diagnosis.Status == "canceled" {
  218. 218 return domain.Fail("DIAGNOSIS_FAILED", "补证诊断未通过")
  219. 219 }
  220. 220 if diagnosis.Status == "success" {
  221. 221 if err := s.ValidateDiagnosisSource(ctx, diagnosis); err != nil {
  222. 222 return err
  223. 223 }
  224. 224 row.Status = "success"
  225. 225 }
  226. 226 return nil
