  116. 116// CreateTask 在同一事务中创建 task、首个 attempt、事件、审计和可选派发 Outbox。

  118. 118// 尚未完成的导入仍可用原对象键补齐上传，这与创建新任务不同。
  119. 119func (r *Repository) CreateTask(ctx context.Context, t domain.Task, queue bool, audit domain.Audit) (domain.Task, domain.Attempt, bool, error) {
  120. 120 var result domain.Task
  121. 121 var a domain.Attempt
  122. 122 replay := false
  123. 123 err := r.Context(ctx).Transaction(func(tx *gorm.DB) error {
  124. 124 if err := tx.Exec("SELECT pg_advisory_xact_lock(hashtextextended(?,0))", "task-quota:"+t.OwnerID).Error; err != nil {
  125. 125 return err
  126. 126 }
  127. 127 err := tx.Where("creator_id = ? AND idempotency_key = ?", t.OwnerID, t.IdempotencyKey).First(&result).Error
  128. 128 if err == nil {
  129. 129 replay = true
  130. 130 if result.RequestHash != t.RequestHash {
  131. 131 return domain.Fail("IDEMPOTENCY_CONFLICT", "该幂等键已用于不同请求")
  132. 132 }
  133. 133 return tx.Where("task_id = ?", result.ID).Order("attempt DESC").First(&a).Error
  134. 134 }
  135. 135 if !errors.Is(err, gorm.ErrRecordNotFound) {
  136. 136 return err
  137. 137 }
  138. 138 var count int64
  139. 139 if err := tx.Model(&domain.Task{}).Where("creator_id = ? AND collection_status IN ('CREATED','QUEUED','DELIVERED','RUNNING','UPLOADING')", t.OwnerID).Count(&count).Error; err != nil {
  140. 140 return err
  141. 141 }
  142. 142 if count >= 10 {
  143. 143 return domain.Fail("QUOTA_EXCEEDED", "最多同时运行 10 个采集任务")
  144. 144 }
  145. 145 if err := tx.Create(&t).Error; err != nil {
  146. 146 return err
  147. 147 }
  148. 148 k, _ := domain.Kind(t.Kind)
  149. 149 rawName := k.RawName
  150. 150 if rawName == "" {
  151. 151 rawName = "profile.folded"
  152. 152 }
  153. 153 a = domain.Attempt{TaskID: t.ID, Attempt: 1, AgentID: t.AgentID, Status: t.CollectionStatus, RunnerVersion: "1.0.0", RawObjectKey: domain.RawKey(t.ID, 1)[:len(domain.RawKey(t.ID, 1))-len("profile.folded")] + rawName, CreatedAt: time.Now().UTC()}
  154. 154 if err := tx.Create(&a).Error; err != nil {
  155. 155 return err
  156. 156 }
  157. 157 event := domain.EventFor(t, &a.ID, "TASK_CREATED", map[string]any{"kind": t.Kind, "collection_status": t.CollectionStatus})
  158. 158 if err := tx.Create(&event).Error; err != nil {
  159. 159 return err
  160. 160 }
  161. 161 if queue {
  162. 162 out := domain.Outbox{TaskID: t.ID, AttemptID: a.ID, EventType: "DISPATCH", Payload: []byte("{}"), AvailableAt: time.Now().UTC(), CreatedAt: time.Now().UTC()}
  163. 163 if err := tx.Create(&out).Error; err != nil {
  164. 164 return err
  165. 165 }
  166. 166 }
  167. 167 audit.ResourceID = strconv.FormatInt(t.ID, 10)
  168. 168 if err := tx.Create(&audit).Error; err != nil {
  169. 169 return err
  170. 170 }
  171. 171 result = t
  172. 172 return nil
  173. 173 })
  174. 174 return result, a, replay, err
  175. 175}

  189. 189func (r *Repository) Cancel(ctx context.Context, u domain.User, id int64, audit domain.Audit) (domain.Task, error) {
  190. 190 var t domain.Task
  191. 191 err := r.Context(ctx).Transaction(func(tx *gorm.DB) error {
  192. 192 if err := ScopeWrite(tx, u).Clauses(clause.Locking{Strength: "UPDATE"}).Where("id = ?", id).First(&t).Error; err != nil {
  193. 193 return err
  194. 194 }
  195. 195 if t.Terminal() {
  196. 196 // 取消操作必须幂等：已终止的任务保留原终态，
  197. 197 // 不能重复生成事件、审计、上传授权或取消 Outbox 记录。
  198. 198 return nil
  199. 199 }
  200. 200 if err := tx.Model(&t).Updates(map[string]any{"collection_status": "CANCELED", "analysis_status": "CANCELED", "updated_at": time.Now().UTC(), "version": gorm.Expr("version + 1")}).Error; err != nil {
  201. 201 return err
  202. 202 }
  203. 203 var a domain.Attempt
  204. 204 if err := tx.Where("task_id = ?", id).Order("attempt DESC").First(&a).Error; err != nil {
  205. 205 return err
  206. 206 }
  207. 207 if err := tx.Model(&a).Updates(map[string]any{"status": "CANCELED", "finished_at": time.Now().UTC(), "raw_upload_url": "", "upload_targets": string(TerminalUploadMetadata(a))}).Error; err != nil {
  208. 208 return err
  209. 209 }
  210. 210 if err := tx.Exec("UPDATE analysis_jobs SET status='failed',lease_owner=NULL,lease_expires_at=NULL,error_code='TASK_CANCELED',error_message='Task was canceled',updated_at=now() WHERE task_id=? AND attempt_id=? AND status IN ('pending','running','retry')", id, a.ID).Error; err != nil {
  211. 211 return err
  212. 212 }
  213. 213 if err := tx.Exec("UPDATE ai_requests SET status='failed',lease_owner=NULL,lease_expires_at=NULL,error_code='TASK_CANCELED',updated_at=now() WHERE task_id=? AND attempt_id=? AND status IN ('pending','running')", id, a.ID).Error; err != nil {
  214. 214 return err
  215. 215 }
  216. 216 event := domain.EventFor(t, &a.ID, "TASK_CANCELED", map[string]string{"collection_status": "CANCELED"})
  217. 217 if err := tx.Create(&event).Error; err != nil {
  218. 218 return err
  219. 219 }
  220. 220 out := domain.Outbox{TaskID: t.ID, AttemptID: a.ID, EventType: "CANCEL", Payload: []byte("{}"), AvailableAt: time.Now().UTC(), CreatedAt: time.Now().UTC()}
  221. 221 if err := tx.Create(&out).Error; err != nil {
  222. 222 return err
  223. 223 }
  224. 224 return tx.Create(&audit).Error
  225. 225 })
  226. 226 return t, err
  227. 227}

  229. 229// Retry 先核验写权限和旧终态，再创建新的 attempt 与派发消息。
  230. 230// 同一幂等键重放必须返回同一轮次，旧轮次上传授权与迟到结果立即失去发布资格。
  231. 231func (r *Repository) Retry(ctx context.Context, u domain.User, id int64, key, requestID string) (domain.Task, bool, error) {
  232. 232 var t domain.Task
  233. 233 replay := false
  234. 234 err := r.Context(ctx).Transaction(func(tx *gorm.DB) error {
  235. 235 var owned domain.Task
  236. 236 if err := ScopeWrite(tx, u).Where("id=?", id).First(&owned).Error; err != nil {
  237. 237 return err
  238. 238 }
  239. 239 if err := tx.Exec("SELECT pg_advisory_xact_lock(hashtextextended(?,0))", "task-quota:"+owned.OwnerID).Error; err != nil {
  240. 240 return err
  241. 241 }
  242. 242 if err := ScopeWrite(tx, u).Clauses(clause.Locking{Strength: "UPDATE"}).Where("id = ?", id).First(&t).Error; err != nil {
  243. 243 return err
  244. 244 }
  245. 245 var marker int64
  246. 246 if err := tx.Table("task_events").Where("task_id = ? AND type = 'TASK_RETRIED' AND payload->>'idempotency_key' = ?", id, key).Count(&marker).Error; err != nil {
  247. 247 return err
  248. 248 }
  249. 249 if marker > 0 {
  250. 250 replay = true
  251. 251 return nil
  252. 252 }
  253. 253 if !t.Terminal() {
  254. 254 return domain.Fail("INVALID_STATE", "只有终态任务可以重试")
  255. 255 }
  256. 256 var active int64
  257. 257 if err := tx.Model(&domain.Task{}).Where("creator_id = ? AND collection_status IN ('CREATED','QUEUED','DELIVERED','RUNNING','UPLOADING')", t.OwnerID).Count(&active).Error; err != nil {
  258. 258 return err
  259. 259 }
  260. 260 if active >= 10 {
  261. 261 return domain.Fail("QUOTA_EXCEEDED", "最多同时运行 10 个采集任务")
  262. 262 }
  263. 263 if t.AgentID == nil {
  264. 264 return domain.Fail("INVALID_STATE", "请重新导入原始文本以建立独立的分析任务")
  265. 265 }
  266. 266 var previous domain.Attempt
  267. 267 if err := tx.Where("task_id = ?", id).Order("attempt DESC").First(&previous).Error; err != nil {
  268. 268 return err
  269. 269 }
  270. 270 // 重试会创建新的 attempt；旧 attempt 此后连迟到的日志也不能发布，
  271. 271 // 也不能继续保留上传 URL 或上传授权元数据。
  272. 272 if err := tx.Model(&previous).Updates(map[string]any{"raw_upload_url": "", "upload_targets": "[]"}).Error; err != nil {
  273. 273 return err
  274. 274 }
  275. 275 k, _ := domain.Kind(t.Kind)
  276. 276 rawKey := domain.RawKey(id, previous.Attempt+1)
  277. 277 if k.RawName != "" {
  278. 278 rawKey = rawKey[:len(rawKey)-len("profile.folded")] + k.RawName
  279. 279 }
  280. 280 a := domain.Attempt{TaskID: id, Attempt: previous.Attempt + 1, AgentID: t.AgentID, Status: "CREATED", RunnerVersion: "1.0.0", RawObjectKey: rawKey, CreatedAt: time.Now().UTC()}
  281. 281 if err := tx.Create(&a).Error; err != nil {
  282. 282 return err
  283. 283 }
  284. 284 deadline := time.Now().Add(10 * time.Minute)
  285. 285 if err := tx.Model(&t).Updates(map[string]any{"collection_status": "CREATED", "analysis_status": "PENDING", "error_code": nil, "error_message": nil, "request_id": requestID, "collection_deadline": deadline, "updated_at": time.Now().UTC(), "version": gorm.Expr("version + 1")}).Error; err != nil {
  286. 286 return err
  287. 287 }
  288. 288 e := domain.EventFor(t, &a.ID, "TASK_RETRIED", map[string]string{"idempotency_key": key})
  289. 289 if err := tx.Create(&e).Error; err != nil {
  290. 290 return err
  291. 291 }
  292. 292 out := domain.Outbox{TaskID: id, AttemptID: a.ID, EventType: "DISPATCH", Payload: []byte("{}"), AvailableAt: time.Now().UTC(), CreatedAt: time.Now().UTC()}
  293. 293 if err := tx.Create(&out).Error; err != nil {
  294. 294 return err
  295. 295 }
  296. 296 return nil
  297. 297 })
  298. 298 return t, replay, err
  299. 299}

  353. 353// DispatchOne 领取一条可投递 Outbox，在网络调用期间不持有任务锁，返回后重新锁定并核验当前 attempt。
  354. 354// RPC 失败可能只是 ACK 丢失；重放依赖接收端幂等，而耗尽重试后的失败也不能覆盖其他事务已经推进的状态。
  355. 355func (r *Repository) DispatchOne(ctx context.Context, fn func(domain.Outbox, domain.Task, domain.Attempt) error) error {
  356. 356 return r.Context(ctx).Transaction(func(tx *gorm.DB) error {
  357. 357 var o domain.Outbox
  358. 358 err := tx.Select("outbox.*").Joins("JOIN tasks AS dispatch_task ON dispatch_task.id=outbox.task_id").Clauses(clause.Locking{Strength: "UPDATE", Table: clause.Table{Name: "outbox"}, Options: "SKIP LOCKED"}).Where("outbox.published_at IS NULL AND outbox.available_at <= ?", time.Now()).Order("CASE WHEN outbox.event_type='CANCEL' THEN 1 ELSE 0 END DESC").Order("dispatch_task.priority DESC").Order("outbox.created_at ASC, outbox.id ASC").First(&o).Error
  359. 359 if errors.Is(err, gorm.ErrRecordNotFound) {
  360. 360 return nil
  361. 361 }
  362. 362 if err != nil {
  363. 363 return err
  364. 364 }
  365. 365 var t domain.Task
  366. 366 if err := tx.Where("id = ?", o.TaskID).First(&t).Error; err != nil {
  367. 367 return err
  368. 368 }
  369. 369 var a domain.Attempt
  370. 370 if err := tx.Where("id = ?", o.AttemptID).First(&a).Error; err != nil {
  371. 371 return err
  372. 372 }
  373. 373 var latest domain.Attempt
  374. 374 if err := tx.Where("task_id = ?", t.ID).Order("attempt DESC").First(&latest).Error; err != nil {
  375. 375 return err
  376. 376 }
  377. 377 if !Dispatchable(o, t, a, latest.ID) {
  378. 378 return tx.Model(&o).Update("published_at", time.Now().UTC()).Error
  379. 379 }
  380. 380 err = fn(o, t, a)
  381. 381 if err == nil {
  382. 382 return tx.Model(&o).Update("published_at", time.Now().UTC()).Error
  383. 383 }
  384. 384 attempt := o.Attempt + 1
  385. 385 delay := time.Duration(min(attempt*attempt, 60)) * time.Second
  386. 386 if attempt >= 10 || (t.DeadlineAt != nil && time.Now().After(*t.DeadlineAt)) {
  387. 387 if o.EventType == "DISPATCH" {
  388. 388 // RPC 返回时，队列 ACK、取消或重试可能已经在其他事务中提交。
  389. 389 // 因此只在 RPC 结束后加锁并重读当前状态；若在网络调用期间持锁，
  390. 390 // Core 自己提交状态时会被阻塞，形成跨服务死锁。
  391. 391 var current domain.Task
  392. 392 if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).First(&current, t.ID).Error; err != nil {
  393. 393 return err
  394. 394 }
  395. 395 var currentAttempt domain.Attempt
  396. 396 if err := tx.Clauses(clause.Locking{Strength: "UPDATE"}).Where("task_id=?", t.ID).Order("attempt DESC").First(&currentAttempt).Error; err != nil {
  397. 397 return err
  398. 398 }
  399. 399 if !Dispatchable(o, current, currentAttempt, currentAttempt.ID) {
  400. 400 return tx.Model(&o).Updates(map[string]any{"published_at": time.Now().UTC(), "attempt": attempt}).Error
  401. 401 }
  402. 402 t, a = current, currentAttempt
  403. 403 if err := tx.Model(&t).Updates(map[string]any{"collection_status": "FAILED", "error_code": "DISPATCH_UNAVAILABLE", "error_message": "采集控制面不可用，调度重试已耗尽", "updated_at": time.Now().UTC()}).Error; err != nil {
  404. 404 return err
  405. 405 }
  406. 406 if err := tx.Model(&a).Updates(map[string]any{"status": "FAILED", "finished_at": time.Now().UTC(), "error_code": "DISPATCH_UNAVAILABLE", "raw_upload_url": "", "upload_targets": TerminalUploadMetadata(a)}).Error; err != nil {
  407. 407 return err
  408. 408 }
  409. 409 e := domain.EventFor(t, &a.ID, "TASK_FAILED", map[string]string{"error_code": "DISPATCH_UNAVAILABLE"})
  410. 410 if err := tx.Create(&e).Error; err != nil {
  411. 411 return err
  412. 412 }
  413. 413 }
  414. 414 return tx.Model(&o).Updates(map[string]any{"published_at": time.Now().UTC(), "attempt": attempt, "last_error": "DISPATCH_UNAVAILABLE"}).Error
  415. 415 }
  416. 416 return tx.Model(&o).Updates(map[string]any{"attempt": attempt, "last_error": "DISPATCH_UNAVAILABLE", "available_at": time.Now().Add(delay)}).Error
  417. 417 })
  418. 418}

  446. 446// TickSchedule 锁定到期调度记录并创建本次执行，更新下一触发时间和关联记录。
  447. 447// 多个 API 实例并发时，数据库锁负责唯一领取，不能让每个进程的定时器各自创建一份任务。
  448. 448func (r *Repository) TickSchedule(ctx context.Context, fn func(*Repository, domain.Schedule) (int64, error)) error {
  449. 449 return r.Context(ctx).Transaction(func(tx *gorm.DB) error {
  450. 450 var s domain.Schedule
  451. 451 err := tx.Clauses(clause.Locking{Strength: "UPDATE", Options: "SKIP LOCKED"}).Where("enabled=true AND next_run_at<=?", time.Now()).Order("next_run_at ASC").First(&s).Error
  452. 452 if errors.Is(err, gorm.ErrRecordNotFound) {
  453. 453 return nil
  454. 454 }
  455. 455 if err != nil {
  456. 456 return err
  457. 457 }
  458. 458 taskID, callErr := fn(New(tx), s)
  459. 459 if callErr != nil {
  460. 460 var de *domain.Error
  461. 461 if !errors.As(callErr, &de) {
  462. 462 return callErr
  463. 463 }
  464. 464 if err := tx.Exec("INSERT INTO schedule_records(schedule_id,scheduled_at,status,error_message,template_snapshot) VALUES(?,?,'FAILED',?,?) ON CONFLICT DO NOTHING", s.ID, s.NextRunAt, de.Code, string(s.Template)).Error; err != nil {
  465. 465 return err
  466. 466 }
  467. 467 } else {
  468. 468 if err := tx.Exec("INSERT INTO schedule_records(schedule_id,scheduled_at,task_id,status,template_snapshot) VALUES(?,?,?,'CREATED',?) ON CONFLICT DO NOTHING", s.ID, s.NextRunAt, taskID, string(s.Template)).Error; err != nil {
  469. 469 return err
  470. 470 }
  471. 471 }
  472. 472 next, err := domain.NextSchedule(s.Cron, s.Timezone, s.IntervalMinutes, time.Now())
  473. 473 if err != nil {
  474. 474 return err
  475. 475 }
  476. 476 return tx.Model(&s).Updates(map[string]any{"next_run_at": next, "last_run_at": s.NextRunAt, "updated_at": time.Now().UTC()}).Error
  477. 477 })
