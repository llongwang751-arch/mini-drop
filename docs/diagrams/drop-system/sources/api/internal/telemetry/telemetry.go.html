  46. 46func (m *Metrics) UnaryInterceptor(ctx context.Context, method string, req, reply any, cc *grpc.ClientConn, invoke grpc.UnaryInvoker, opts ...grpc.CallOption) error {
  47. 47 start := time.Now()
  48. 48 err := invoke(ctx, method, req, reply, cc, opts...)
  49. 49 m.RPCRequests.WithLabelValues(method, status.Code(err).String()).Inc()

  52. 52}
  53. 53func Configure(ctx context.Context, endpoint, service, version string) (func(context.Context) error, error) {
  54. 54 otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(propagation.TraceContext{}, propagation.Baggage{}))
  55. 55 if endpoint == "" {
  56. 56 return func(context.Context) error { return nil }, nil
  57. 57 }
  58. 58 u, err := url.Parse(endpoint)
  59. 59 if err != nil || u.Host == "" {
  60. 60 return nil, fmt.Errorf("invalid OTLP endpoint")
  61. 61 }
  62. 62 if u.Path == "" || u.Path == "/" {
  63. 63 u.Path = "/v1/traces"
  64. 64 }
  65. 65 endpoint = strings.TrimRight(u.String(), "/")
  66. 66 exp, err := otlptracehttp.New(ctx, otlptracehttp.WithEndpointURL(endpoint))
  67. 67 if err != nil {
  68. 68 return nil, err
  69. 69 }
  70. 70 provider := sdktrace.NewTracerProvider(sdktrace.WithBatcher(exp, sdktrace.WithMaxQueueSize(512), sdktrace.WithMaxExportBatchSize(128), sdktrace.WithBatchTimeout(2*time.Second)), sdktrace.WithSampler(sdktrace.ParentBased(sdktrace.TraceIDRatioBased(.1))), sdktrace.WithResource(resource.NewSchemaless(attribute.String("service.name", service), attribute.String("service.version", version))))
  71. 71 otel.SetTracerProvider(provider)
  72. 72 return provider.Shutdown, nil
  73. 73}

  81. 81func (c *databaseCollector) Collect(ch chan<- prometheus.Metric) {
  82. 82 ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
  83. 83 defer cancel()
  84. 84 db := c.DB.WithContext(ctx)
  85. 85 failed := false
  86. 86 gauge := func(name, help string, v float64, labels []string, values ...string) {
  87. 87 ch <- prometheus.MustNewConstMetric(prometheus.NewDesc(name, help, labels, nil), prometheus.GaugeValue, v, values...)
  88. 88 }
  89. 89 scalar := func(name, query string) {
  90. 90 var v float64
  91. 91 if err := db.Raw(query).Scan(&v).Error; err != nil {
  92. 92 failed = true
  93. 93 return
  94. 94 }
  95. 95 gauge(name, name, v, nil)
  96. 96 }
  97. 97 scalar("drop_agents_online", `SELECT count(*) FROM agents WHERE last_seen_at>now()-interval '45 seconds' AND status NOT IN ('disabled','offline')`)
  98. 98 scalar("drop_task_queue_depth", `SELECT count(*) FROM tasks WHERE collection_status IN ('CREATED','QUEUED') AND target_agent_id IS NOT NULL`)
  99. 99 scalar("drop_task_queue_oldest_age_seconds", `SELECT coalesce(max(extract(epoch FROM now()-created_at)),0) FROM tasks WHERE collection_status IN ('CREATED','QUEUED') AND target_agent_id IS NOT NULL`)
  100. 100 scalar("drop_tasks_overdue", `SELECT count(*) FROM tasks WHERE collection_deadline<now() AND collection_status IN ('CREATED','QUEUED','DELIVERED','RUNNING','UPLOADING')`)
  101. 101 scalar("drop_analysis_oldest_age_seconds", `SELECT coalesce(max(extract(epoch FROM now()-created_at)),0) FROM analysis_jobs WHERE status IN ('pending','retry')`)
  102. 102 scalar("drop_analysis_expired_leases", `SELECT count(*) FROM analysis_jobs WHERE status='running' AND lease_expires_at<now()`)
  103. 103 scalar("drop_artifacts_expired", `SELECT count(*) FROM artifacts WHERE status='ready' AND expires_at<now()`)
  104. 104 scalar("drop_outbox_pending", `SELECT count(*) FROM outbox WHERE published_at IS NULL`)
  105. 105 scalar("drop_composites_running", `SELECT count(*) FROM composite_runs WHERE status IN ('PENDING','RUNNING')`)
  106. 106 var agents []struct {
  107. 107 ID string
  108. 108 Age float64
  109. 109 }
  110. 110 if err := db.Raw(`SELECT id,extract(epoch FROM now()-last_seen_at) AS age FROM agents ORDER BY id LIMIT 500`).Scan(&agents).Error; err != nil {
  111. 111 failed = true
  112. 112 }
  113. 113 for _, a := range agents {
  114. 114 gauge("drop_agent_heartbeat_age_seconds", "Seconds since actual Agent heartbeat.", max(0, a.Age), []string{"agent"}, a.ID)
  115. 115 }
  116. 116 var jobs []struct {
  117. 117 Status, Pipeline string
  118. 118 Count float64
  119. 119 }
  120. 120 if err := db.Raw(`SELECT status,pipeline,count(*) AS count FROM analysis_jobs GROUP BY status,pipeline`).Scan(&jobs).Error; err != nil {
  121. 121 failed = true
  122. 122 }
  123. 123 for _, j := range jobs {
  124. 124 gauge("drop_analysis_jobs", "Persisted analysis jobs by status and pipeline.", j.Count, []string{"status", "pipeline"}, j.Status, j.Pipeline)
  125. 125 }
  126. 126 if sqlDB, err := c.DB.DB(); err == nil {
  127. 127 stats := sqlDB.Stats()
  128. 128 ch <- prometheus.MustNewConstMetric(prometheus.NewDesc("drop_db_pool_wait_seconds_total", "Total database pool wait seconds.", nil, nil), prometheus.CounterValue, stats.WaitDuration.Seconds())
  129. 129 gauge("drop_db_pool_connections", "Open database connections.", float64(stats.OpenConnections), nil)
  130. 130 }
  131. 131 for role, path := range c.Certs {
  132. 132 b, err := os.ReadFile(path)
  133. 133 if err != nil {
  134. 134 continue
  135. 135 }
  136. 136 block, _ := pem.Decode(b)
  137. 137 if block == nil {
  138. 138 continue
  139. 139 }
  140. 140 cert, err := x509.ParseCertificate(block.Bytes)
  141. 141 if err != nil {
  142. 142 continue
  143. 143 }
  144. 144 gauge("drop_certificate_expiry_seconds", "Seconds until mTLS certificate expiry.", time.Until(cert.NotAfter).Seconds(), []string{"role"}, role)
  145. 145 }
  146. 146 value := 0.
  147. 147 if failed {
  148. 148 value = 1
  149. 149 }
  150. 150 gauge("drop_metrics_collection_error", "A database metric query failed.", value, nil)
  151. 151}
