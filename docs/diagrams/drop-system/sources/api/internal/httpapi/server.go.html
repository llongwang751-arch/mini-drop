  104. 104// recoverSafely 不输出请求头；Cookie、会话令牌和 CSRF 值绝不能进入异常日志。
  105. 105func (s *Server) recoverSafely() gin.HandlerFunc {
  106. 106 return func(c *gin.Context) {
  107. 107 defer func() {
  108. 108 if recover() != nil {
  109. 109 slog.Error("request panic", "stage", "http.request", "request_id", c.GetString("request_id"), "error_code", "INTERNAL_ERROR")
  110. 110 s.fail(c, domain.Fail("INTERNAL_ERROR", "服务暂时不可用"))
  111. 111 }
  112. 112 }()
  113. 113 c.Next()
  114. 114 }
  115. 115}

  117. 117// requestID 建立有限且可关联的请求标识，供响应、审计与服务日志连接同一次调用；标识不是认证凭据。
  118. 118func (s *Server) requestID() gin.HandlerFunc {
  119. 119 return func(c *gin.Context) {
  120. 120 id := c.GetHeader("X-Request-ID")
  121. 121 if len(id) < 8 || len(id) > 100 || strings.ContainsAny(id, "\r\n") {
  122. 122 id = uuid.NewString()
  123. 123 }
  124. 124 c.Set("request_id", id)
  125. 125 c.Header("X-Request-ID", id)
  126. 126 c.Header("X-Content-Type-Options", "nosniff")
  127. 127 c.Header("Cache-Control", "no-store")
  128. 128 s.requests.Add(1)
  129. 129 start := time.Now()
  130. 130 c.Next()
  131. 131 if c.Writer.Status() >= 500 {
  132. 132 s.errors.Add(1)
  133. 133 }
  134. 134 sc := trace.SpanContextFromContext(c.Request.Context())
  135. 135 taskID, agentID := "", ""
  136. 136 if strings.HasPrefix(c.FullPath(), "/api/v1/tasks/:id") {
  137. 137 taskID = c.Param("id")
  138. 138 }
  139. 139 if strings.HasPrefix(c.FullPath(), "/api/v1/agents/:id") {
  140. 140 agentID = c.Param("id")
  141. 141 }
  142. 142 slog.Info("http request", "stage", "http.request", "request_id", id, "task_id", taskID, "agent_id", agentID, "trace_id", sc.TraceID().String(), "method", c.Request.Method, "route", c.FullPath(), "status", c.Writer.Status(), "duration_ms", time.Since(start).Milliseconds(), "error_code", c.GetString("error_code"))
  143. 143 }
  144. 144}

  149. 149// fail 将领域错误映射为稳定 HTTP 错误契约，未知内部错误不应把数据库、执行器或模型原始消息透给浏览器。
  150. 150func (s *Server) fail(c *gin.Context, err error) {
  151. 151 var de *domain.Error
  152. 152 if errors.Is(err, gorm.ErrRecordNotFound) {
  153. 153 de = domain.Fail("NOT_FOUND", "资源不存在或无权访问")
  154. 154 } else if !errors.As(err, &de) {
  155. 155 slog.Error("request failed", "stage", "http.request", "request_id", c.GetString("request_id"), "error_code", "INTERNAL_ERROR")
  156. 156 de = domain.Fail("INTERNAL_ERROR", "服务暂时不可用")
  157. 157 }
  158. 158 c.Set("error_code", de.Code)
  159. 159 status := 400
  160. 160 switch de.Code {
  161. 161 case "UNAUTHENTICATED", "INVALID_CREDENTIALS":
  162. 162 status = 401
  163. 163 case "FORBIDDEN", "CSRF_INVALID", "TARGET_NOT_ALLOWED":
  164. 164 status = 403
  165. 165 case "NOT_FOUND", "AGENT_NOT_FOUND":
  166. 166 status = 404
  167. 167 case "ARTIFACT_EXPIRED":
  168. 168 status = 410
  169. 169 case "INVALID_STATE", "IDEMPOTENCY_CONFLICT", "RESULT_NOT_READY", "USER_EXISTS", "STALE_ATTEMPT", "DIAGNOSIS_NOT_ACTIVE", "KNOWLEDGE_CHANGED":
  170. 170 status = 409
  171. 171 case "RATE_LIMITED", "QUOTA_EXCEEDED", "DIAGNOSIS_BUSY":
  172. 172 status = 429
  173. 173 case "AGENT_OFFLINE", "CAPABILITY_UNAVAILABLE", "STORAGE_UNAVAILABLE", "AI_UNAVAILABLE", "OIDC_UNAVAILABLE", "CODE_EXECUTOR_UNAVAILABLE", "QUALITY_REPORTS_UNAVAILABLE":
  174. 174 status = 503
  175. 175 de.Retryable = true
  176. 176 case "INTERNAL_ERROR":
  177. 177 status = 500
  178. 178 case "KNOWLEDGE_TIMEOUT":
  179. 179 status = 504
  180. 180 de.Retryable = true
  181. 181 }
  182. 182 c.AbortWithStatusJSON(status, gin.H{"request_id": c.GetString("request_id"), "data": nil, "error": de})
  183. 183}

  185. 185// bind 限制请求体并拒绝未知字段与尾随 JSON；尽早拒绝无界或歧义输入，不能依靠模型或数据库替入口补救。
  186. 186func (s *Server) bind(c *gin.Context, v any) bool {
  187. 187 limit := int64(4 << 20)
  188. 188 if c.FullPath() == "/api/v1/tasks" && c.Request.Method == "POST" {
  189. 189 limit = 64 << 20
  190. 190 }
  191. 191 c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, limit)
  192. 192 dec := json.NewDecoder(c.Request.Body)
  193. 193 dec.DisallowUnknownFields()
  194. 194 if err := dec.Decode(v); err != nil {
  195. 195 s.fail(c, domain.Fail("INVALID_PARAMETERS", "请求 JSON 无效或含不支持的字段"))
  196. 196 return false
  197. 197 }
  198. 198 if err := dec.Decode(new(any)); err != io.EOF {
  199. 199 s.fail(c, domain.Fail("INVALID_PARAMETERS", "请求必须包含单个 JSON 对象"))
  200. 200 return false
  201. 201 }
  202. 202 return true
  203. 203}

  206. 206// authenticate 每次从持久化会话读取当前用户，写请求额外校验 CSRF 并执行共享限流。
  207. 207// Cookie 自动附带不等于用户主动提交；认证成功也不等于有权读取某个 task，资源处理器还要单独授权。
  208. 208func (s *Server) authenticate() gin.HandlerFunc {
  209. 209 return func(c *gin.Context) {
  210. 210 token, err := c.Cookie("drop_session")
  211. 211 if err != nil || len(token) != 64 {
  212. 212 s.fail(c, domain.Fail("UNAUTHENTICATED", "请先登录"))
  213. 213 return
  214. 214 }
  215. 215 session, u, err := s.Repo.Session(c.Request.Context(), token)
  216. 216 if err != nil {
  217. 217 s.fail(c, domain.Fail("UNAUTHENTICATED", "登录状态已过期"))
  218. 218 return
  219. 219 }
  220. 220 c.Set("user", u)
  221. 221 c.Set("csrf", session.CSRFToken)
  222. 222 if c.Request.Method != "GET" && c.Request.Method != "HEAD" {
  223. 223 if subtle.ConstantTimeCompare([]byte(c.GetHeader("X-CSRF-Token")), []byte(session.CSRFToken)) != 1 {
  224. 224 s.fail(c, domain.Fail("CSRF_INVALID", "安全校验失败，请刷新页面"))
  225. 225 return
  226. 226 }
  227. 227 allowed, rateErr := s.allowRate(c, "write:"+u.ID, 60, time.Minute)
  228. 228 if rateErr != nil {
  229. 229 s.fail(c, rateErr)
  230. 230 return
  231. 231 }
  232. 232 if !allowed {
  233. 233 s.fail(c, domain.Fail("RATE_LIMITED", "操作过于频繁，请稍后再试"))
  234. 234 return
  235. 235 }
  236. 236 }
  237. 237 c.Next()
  238. 238 }
  239. 239}
  240. 240func (s *Server) login(c *gin.Context) {
  241. 241 allowed, rateErr := s.allowRate(c, "login:"+c.ClientIP(), 10, 15*time.Minute)
  242. 242 if rateErr != nil {
  243. 243 s.fail(c, rateErr)
  244. 244 return
  245. 245 }
  246. 246 if !allowed {
  247. 247 c.Header("Retry-After", "900")
  248. 248 s.fail(c, domain.Fail("RATE_LIMITED", "登录尝试过于频繁，请 15 分钟后再试"))
  249. 249 return
  250. 250 }
  251. 251 if origin := c.GetHeader("Origin"); origin != "" && strings.TrimRight(origin, "/") != strings.TrimRight(s.PublicURL, "/") {
  252. 252 s.fail(c, domain.Fail("CSRF_INVALID", "登录来源校验失败"))
  253. 253 return
  254. 254 }
  255. 255 var in struct {
  256. 256 Username string `json:"username"`
  257. 257 Password string `json:"password"`
  258. 258 }
  259. 259 if !s.bind(c, &in) {
  260. 260 return
  261. 261 }
  262. 262 if len(in.Username) > 64 || len(in.Password) > 72 {
  263. 263 s.fail(c, domain.Fail("INVALID_CREDENTIALS", "用户名或密码错误"))
  264. 264 return
  265. 265 }
  266. 266 u, token, csrf, err := s.Accounts.Login(c.Request.Context(), in.Username, in.Password, c.GetString("request_id"))
  267. 267 if err != nil {
  268. 268 s.fail(c, err)
  269. 269 return
  270. 270 }
  271. 271 c.SetSameSite(http.SameSiteLaxMode)
  272. 272 c.SetCookie("drop_session", token, 12*3600, "/", "", s.SecureCookie, true)
  273. 273 s.ok(c, 200, gin.H{"user": u, "csrf_token": csrf})
  274. 274}
  275. 275func (s *Server) logout(c *gin.Context) {
  276. 276 token, _ := c.Cookie("drop_session")
  277. 277 if err := s.Repo.DeleteSession(c.Request.Context(), token); err != nil {
  278. 278 s.fail(c, err)
  279. 279 return
  280. 280 }
  281. 281 _ = s.Repo.Audit(c.Request.Context(), service.Audit(user(c), "auth.logout", "user", user(c).ID, c.GetString("request_id")))
  282. 282 c.SetSameSite(http.SameSiteLaxMode)
  283. 283 c.SetCookie("drop_session", "", -1, "/", "", s.SecureCookie, true)
  284. 284 s.ok(c, 200, gin.H{"logged_out": true})
  285. 285}
