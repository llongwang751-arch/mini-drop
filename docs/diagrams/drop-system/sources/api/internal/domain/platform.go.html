  70. 70// NextSchedule 按时区和 cron 或固定间隔计算下一次触发点；定时器使用持久化时间，进程重启不应重置计划语义。
  71. 71func NextSchedule(expr, zone string, interval int, after time.Time) (time.Time, error) {
  72. 72 loc, err := time.LoadLocation(zone)
  73. 73 if err != nil {
  74. 74 return time.Time{}, Fail("INVALID_PARAMETERS", "IANA时区无效")
  75. 75 }
  76. 76 if expr == "" {
  77. 77 if interval < 1 || interval > 10080 {
  78. 78 return time.Time{}, Fail("INVALID_PARAMETERS", "间隔须为1至10080分钟")
  79. 79 }
  80. 80 return after.Add(time.Duration(interval) * time.Minute).UTC(), nil
  81. 81 }
  82. 82 if len(strings.Fields(expr)) != 5 || strings.ContainsAny(expr, "@=\r\n") {
  83. 83 return time.Time{}, Fail("INVALID_PARAMETERS", "Cron必须为标准五段表达式")
  84. 84 }
  85. 85 spec, err := cron.NewParser(cron.Minute | cron.Hour | cron.Dom | cron.Month | cron.Dow).Parse("CRON_TZ=" + loc.String() + " " + expr)
  86. 86 if err != nil {
  87. 87 return time.Time{}, Fail("INVALID_PARAMETERS", "Cron表达式无效")
  88. 88 }
  89. 89 next := spec.Next(after)
  90. 90 if next.IsZero() {
  91. 91 return next, Fail("INVALID_PARAMETERS", "Cron在未来五年内无触发时点")
  92. 92 }
  93. 93 return next.UTC(), nil
  94. 94}

  117. 117func (c CreateComposite) Validate() *Error {
  118. 118 if c.Name == "" || len(c.Name) > 120 || c.DeadlineSeconds < 1 || c.DeadlineSeconds > 3600 || len(c.Nodes) < 1 || len(c.Nodes) > 16 {
  119. 119 return Fail("INVALID_PARAMETERS", "复合任务名称、期限或节点数无效")
  120. 120 }
  121. 121 nodes := map[string]CompositeNodeInput{}
  122. 122 for _, n := range c.Nodes {
  123. 123 if !regexp.MustCompile(`^[a-zA-Z][a-zA-Z0-9_-]{0,47}$`).MatchString(n.Key) {
  124. 124 return Fail("INVALID_DAG", "节点key无效")
  125. 125 }
  126. 126 if _, ok := nodes[n.Key]; ok {
  127. 127 return Fail("INVALID_DAG", "节点key重复")
  128. 128 }
  129. 129 if n.Task.Kind == "COMPOSITE" || n.Task.Import != nil || n.Task.CollapsedText != "" || n.Task.Kind == "IMPORT_COLLAPSED" {
  130. 130 return Fail("INVALID_DAG", "复合任务只允许现场采集子任务")
  131. 131 }
  132. 132 nodes[n.Key] = n
  133. 133 }
  134. 134 state := map[string]int{}
  135. 135 var visit func(string) bool
  136. 136 visit = func(key string) bool {
  137. 137 if state[key] == 1 {
  138. 138 return false
  139. 139 }
  140. 140 if state[key] == 2 {
  141. 141 return true
  142. 142 }
  143. 143 n, ok := nodes[key]
  144. 144 if !ok {
  145. 145 return false
  146. 146 }
  147. 147 state[key] = 1
  148. 148 for _, d := range n.DependsOn {
  149. 149 if !visit(d) {
  150. 150 return false
  151. 151 }
  152. 152 }
  153. 153 state[key] = 2
  154. 154 return true
  155. 155 }
  156. 156 for key := range nodes {
  157. 157 if !visit(key) {
  158. 158 return Fail("INVALID_DAG", "依赖缺失或存在环")
  159. 159 }
  160. 160 }
  161. 161 return nil
  162. 162}

  196. 196// AggregateComposite 根据子节点状态汇总父组合状态及计数；必选节点与可选节点的失败语义不同，部分结果不能包装成全部成功。
  197. 197func AggregateComposite(nodes []CompositeNode, partial bool) (string, map[string]int) {
  198. 198 summary := map[string]int{"total": len(nodes), "waiting": 0, "running": 0, "succeeded": 0, "failed": 0, "skipped": 0}
  199. 199 requiredFailed := false
  200. 200 for _, n := range nodes {
  201. 201 switch n.Status {
  202. 202 case "WAITING":
  203. 203 summary["waiting"]++
  204. 204 case "RUNNING":
  205. 205 summary["running"]++
  206. 206 case "SUCCESS":
  207. 207 summary["succeeded"]++
  208. 208 case "FAILED":
  209. 209 summary["failed"]++
  210. 210 requiredFailed = requiredFailed || n.Required
  211. 211 case "SKIPPED":
  212. 212 summary["skipped"]++
  213. 213 requiredFailed = requiredFailed || n.Required
  214. 214 }
  215. 215 }
  216. 216 if summary["waiting"]+summary["running"] > 0 {
  217. 217 return "RUNNING", summary
  218. 218 }
  219. 219 if requiredFailed || (!partial && summary["failed"]+summary["skipped"] > 0) || summary["succeeded"] == 0 {
  220. 220 return "FAILED", summary
  221. 221 }
  222. 222 if summary["failed"]+summary["skipped"] > 0 {
  223. 223 return "PARTIAL_SUCCESS", summary
  224. 224 }
  225. 225 return "SUCCESS", summary
  226. 226}
