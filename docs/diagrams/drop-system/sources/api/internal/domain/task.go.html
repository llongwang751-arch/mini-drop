  92. 92// Validate 校验创建参数的协议边界，包括目录能力、目标、时长、大小与导入限制。
  93. 93// 它是纯规则校验，不查询当前用户、目标在线状态或数据库，因此服务层还要继续授权。

  95. 95 if c.Priority < -10 || c.Priority > 10 {

  108. 108 if c.Visibility != "" && c.Visibility != "private" && c.Visibility != "group" {

  110. 110 }
  111. 111 if c.Visibility == "group" && c.GroupID == nil {
  112. 112 return Fail("INVALID_PARAMETERS", "组内可见任务必须指定group_id")
  113. 113 }
  114. 114 if c.Import != nil {
  115. 115 if c.AgentID != "" || c.CollapsedText != "" {
  116. 116 return Fail("INVALID_PARAMETERS", "导入与Agent目标不能混用")
  117. 117 }
  118. 118 if err := ValidateImport(k, *c.Import); err != nil {
  119. 119 return err
  120. 120 }
  121. 121 return ValidateKindParameters(c.Kind, c.Parameters)
  122. 122 }
  123. 123 if c.Kind == "IMPORT_COLLAPSED" {
  124. 124 if c.CollapsedText == "" || len(c.CollapsedText) > 2<<20 {
  125. 125 return Fail("INVALID_PARAMETERS", "导入文本须为 1 字节至 2 MiB")
  126. 126 }
  127. 127 return nil
  128. 128 }
  129. 129 if err := ValidateKindParameters(c.Kind, c.Parameters); err != nil {
  130. 130 return err
  131. 131 }
  132. 132 if len(k.Targets) > 0 {
  133. 133 allowed := false
  134. 134 for _, target := range k.Targets {
  135. 135 allowed = allowed || target == c.Parameters.Scope
  136. 136 }
  137. 137 if !allowed {
  138. 138 return Fail("INVALID_PARAMETERS", "该TaskKind不支持此目标范围")
  139. 139 }
  140. 140 }
  141. 141 if c.Parameters.Scope == "container" && c.Parameters.Target != nil && c.Parameters.Target.ContainerID != "" && c.AgentID != "" {
  142. 142 return nil
  143. 143 }
  144. 144 if c.Kind != "PERF_CPU" && c.Kind != "PROC_CPU" {
  145. 145 if c.AgentID == "" || !k.AgentRunner {
  146. 146 return Fail("INVALID_PARAMETERS", "必须指定采集Agent")
  147. 147 }
  148. 148 if c.Parameters.Scope == "host" {
  149. 149 return nil
  150. 150 }
  151. 151 if c.Parameters.Scope == "container" && c.Parameters.Target != nil && c.Parameters.Target.ContainerID != "" {
  152. 152 return nil
  153. 153 }
  154. 154 if k.RequiresPID && c.Parameters.PID < 2 {
  155. 155 return Fail("INVALID_PARAMETERS", "必须指定合法PID")
  156. 156 }
  157. 157 return nil
  158. 158 }
  159. 159 maxFreq := 199
  160. 160 if c.Kind == "PROC_CPU" {
  161. 161 maxFreq = 10
  162. 162 }

  164. 164 return Fail("INVALID_PARAMETERS", "PID、采样时长、频率或采样范围无效")
