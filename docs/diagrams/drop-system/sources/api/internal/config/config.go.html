  94. 94// boundedFile 只读取有界普通文件，并校验实际打开的描述符。
  95. 95// 秘密投射卷可能包含符号链接，因此不能只在打开之前检查路径；错误必须脱敏。
  96. 96func boundedFile(name string) (data []byte, err error) {
  97. 97 if name == "" || !valid(name) {
  98. 98 return nil, ErrInvalid
  99. 99 }
  100. 100 file, err := openRegular(name)
  101. 101 if err != nil {
  102. 102 return nil, ErrInvalid
  103. 103 }
  104. 104 defer file.Close()
  105. 105 info, err := file.Stat()
  106. 106 if err != nil || !info.Mode().IsRegular() || info.Size() > maxBytes {
  107. 107 return nil, ErrInvalid
  108. 108 }
  109. 109 data, err = io.ReadAll(io.LimitReader(file, maxBytes+1))
  110. 110 if err != nil || len(data) > maxBytes {
  111. 111 return nil, ErrInvalid
  112. 112 }
  113. 113 return data, nil
  114. 114}

  116. 116func fileSettings(name string) (map[string]string, error) {
  117. 117 data, err := boundedFile(name)
  118. 118 if err != nil {
  119. 119 return nil, ErrInvalid
  120. 120 }
  121. 121 decoder := json.NewDecoder(bytes.NewReader(data))
  122. 122 token, err := decoder.Token()

  124. 124 return nil, ErrInvalid
  125. 125 }
  126. 126 values := map[string]string{}
  127. 127 for decoder.More() {
  128. 128 token, err := decoder.Token()
  129. 129 key, ok := token.(string)
  130. 130 if err != nil || !ok {
  131. 131 return nil, ErrInvalid
  132. 132 }
  133. 133 if _, duplicate := values[key]; duplicate {
  134. 134 return nil, ErrInvalid
  135. 135 }
  136. 136 base, reference, known := logicalKey(key)
  137. 137 var raw any
  138. 138 if !known || decoder.Decode(&raw) != nil {
  139. 139 return nil, ErrInvalid
  140. 140 }
  141. 141 value, stringValue := raw.(string)
  142. 142 if !stringValue || !valid(value) || (definitions[base].secret && !reference) {
  143. 143 return nil, ErrInvalid
  144. 144 }
  145. 145 values[key] = value
  146. 146 }
  147. 147 if token, err = decoder.Token(); err != nil || token != json.Delim('}') {
  148. 148 return nil, ErrInvalid
  149. 149 }
  150. 150 if decoder.Decode(new(any)) != io.EOF {
  151. 151 return nil, ErrInvalid
  152. 152 }

  156. 156// Load 按 CLI > 环境变量 > JSON 文件 > 默认值的顺序解析。

  172. 172 return nil, ErrInvalid

  179. 179 if flag != "--config" && flag != "--set" {

  182. 182 if !hasValue {

  186. 186 }

  199. 199 return nil, ErrInvalid
  200. 200 }
  201. 201 seen[base] = true
  202. 202 cli[key] = setting
  203. 203 }
  204. 204 fileValues := map[string]string{}
  205. 205 if filename != "" {

  207. 207 fileValues, err = fileSettings(filename)

  227. 227 }
  228. 228 }

  232. 232 return nil, ErrInvalid
  233. 233 }
  234. 234 value = strings.TrimSpace(string(data))
  235. 235 if !valid(value) {
  236. 236 return nil, ErrInvalid
  237. 237 }
  238. 238 }
  239. 239 if value == "" && spec.value != "" {
  240. 240 value = spec.value
  241. 241 }
  242. 242 result.values[key] = value
  243. 243 }
  244. 244 if result.values["OIDC_REDIRECT_URL"] == "" {
  245. 245 result.values["OIDC_REDIRECT_URL"] = result.values["PUBLIC_URL"] + "/api/v1/auth/oidc/callback"
  246. 246 }

  248. 248}

  271. 271func (c *Config) Effective() SafeValues {
  272. 272 values := map[string]string{}
  273. 273 for key, value := range c.values {
  274. 274 spec := definitions[key]
  275. 275 if value == "" {
  276. 276 values[key] = "<unset>"

  281. 281 values[key] = "<configured>"

  283. 283 values[key] = "<configured>"
  284. 284 } else {
  285. 285 values[key] = value
  286. 286 }
  287. 287 }
  288. 288 return SafeValues{values: values}

  291. 291var endpointAddress = regexp.MustCompile(`^[A-Za-z0-9.\[\]:-]{1,255}$`)

  293. 293func safeEndpoint(value string) string {
  294. 294 if endpointAddress.MatchString(value) {
  295. 295 return value
  296. 296 }
  297. 297 parsed, err := url.Parse(value)
  298. 298 if err == nil && (parsed.Scheme == "http" || parsed.Scheme == "https") && parsed.Host != "" && endpointAddress.MatchString(parsed.Host) {
  299. 299 return parsed.Scheme + "://" + parsed.Host
  300. 300 }
  301. 301 return "<configured>"
  302. 302}
