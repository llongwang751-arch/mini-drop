  102. 102 width: node.x1 - node.x0,
  103. 103 depth: node.depth,
  104. 104 value: node.value || 0,
  105. 105 self: node.data.self,
  106. 106 node: node.data,
  107. 107 });
  108. 108 });

  110. 110 rects,
  111. 111 rows,
  112. 112 height: root.height + 1,
