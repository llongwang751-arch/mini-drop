  4. 4 * 对象请求不携带应用 Cookie；预签名 URL 只在本流程临时使用，不放进错误消息或日志。

  8. 8 ERROR_BODY_LIMIT,

  25. 25 signal: AbortSignal,
  26. 26) {

  28. 28 requestId = "";

  33. 33 `/api/v1/tasks/${message.taskId}/artifacts/${message.artifactId}/download`,

  35. 35 credentials: "same-origin",
  36. 36 cache: "no-store",
  37. 37 redirect: "error",
  38. 38 signal,

  63. 63 refreshed = true;
  64. 64 continue;
  65. 65 }
  66. 66 throw new ProfileDataError("PROFILE_URL_EXPIRED");

  71. 71 cache: "no-store",
  72. 72 redirect: "error",
  73. 73 referrerPolicy: "no-referrer",
  74. 74 signal,

  88. 88 continue;
  89. 89 }

  92. 92 throw profileError(response.status, requestId);
  93. 93 }

  96. 96 length &&
  97. 97 (!/^\d+$/.test(length) || Number(length) > COMPRESSED_LIMIT)
  98. 98 )

  103. 103 throw new ProfileDataError("PROFILE_SIZE_MISMATCH");

  105. 105 }
  106. 106 } catch (error) {

  108. 108 error.requestId ||= requestId;
  109. 109 throw error;
  110. 110 }
  111. 111 if (signal.aborted)
  112. 112 throw new ProfileDataError("REQUEST_TIMEOUT", 0, requestId);
  113. 113 throw new ProfileDataError("PROFILE_DOWNLOAD_FAILED", 0, requestId);
  114. 114 }
  115. 115}
  116. 116// Worker 只接受两类受限操作；失败只返回受控错误码与请求 ID，不序列化 URL 或原生异常。

  119. 119 const controller = new AbortController();

  122. 122 let data;
  123. 123 if (message.op === "result") {
  124. 124 if (
  125. 125 !(message.buffer instanceof ArrayBuffer) ||
  126. 126 message.buffer.byteLength > EXPANDED_LIMIT
  127. 127 )

  130. 130 message.buffer,
  131. 131 message.status,
  132. 132 message.requestId,
  133. 133 );
  134. 134 } else {
  135. 135 if (
  136. 136 !/^[1-9]\d{0,18}$/.test(message.taskId) ||
  137. 137 !/^[1-9]\d{0,18}$/.test(message.artifactId) ||
  138. 138 !Number.isSafeInteger(message.expectedSize) ||
  139. 139 message.expectedSize <= 0 ||
  140. 140 message.expectedSize > COMPRESSED_LIMIT
  141. 141 )
  142. 142 throw new ProfileDataError("PROFILE_TOO_LARGE");
  143. 143 data = await artifact(message, controller.signal);
  144. 144 }
  145. 145 self.postMessage({ data });
  146. 146 } catch (error) {
  147. 147 const problem =
  148. 148 error instanceof ProfileDataError
  149. 149 ? error
  150. 150 : new ProfileDataError("PROFILE_WORKER_FAILED");
  151. 151 self.postMessage({
  152. 152 error: {
  153. 153 code: problem.code,
  154. 154 status: problem.status,
  155. 155 requestId: problem.requestId,
  156. 156 },
  157. 157 });
  158. 158 } finally {

  160. 160 }
