  1. 1/**
  2. 2 * 任务实时事件通道：将网络字节流拆成完整事件，再通知缓存失效。
  3. 3 * 它不执行采集，也不直接改任务状态；状态最终来自授权的任务详情接口。
  4. 4 */
  5. 5import { newIdempotencyKey } from "./api";
  6. 6
  7. 7export interface StreamEvent {
  8. 8 type: string;
  9. 9 data: string;
  10. 10 lastEventId: string;
  11. 11}
  12. 12export type StreamState = "connecting" | "live" | "polling" | "paused";
  13. 13class StreamProtocolError extends Error {}
  14. 14
  15. 15/**
  16. 16 * 按流解码 UTF-8 和 SSE 帧：网络分块不等于事件边界，只有遇到空行才提交完整事件。
  17. 17 * 限额包含心跳注释、字段名和分隔符，防止小 JSON 之外的大量无用字节绕过内存限制。
  18. 18 */
  19. 19export class EventStreamDecoder {
  20. 20 private decoder = new TextDecoder("utf-8", { fatal: true });
  21. 21 private line = "";
  22. 22 private bytes = 0;
  23. 23 private afterCR = false;
  24. 24 private data: string[] = [];
  25. 25 private type = "message";
  26. 26 private pendingId: string | undefined;
  27. 27 private lastEventId: string;
  28. 28 constructor(
  29. 29 private emit: (event: StreamEvent) => void,
  30. 30 cursor = "",
  31. 31 ) {
  32. 32 this.lastEventId = cursor;
  33. 33 }
  34. 34 push(chunk: Uint8Array) {
  35. 35 this.consume(this.decode(chunk));
  36. 36 }
  37. 37 finish() {
  38. 38 this.consume(this.decode());
  39. 39 if (this.afterCR) {
  40. 40 this.afterCR = false;
  41. 41 this.consumeLine();
  42. 42 }
  43. 43 // 连接结束时丢弃未封口的事件；必须有空行结束标记，才能把数据交给页面。
  44. 44 }
  45. 45 private decode(chunk?: Uint8Array) {
  46. 46 try {
  47. 47 return this.decoder.decode(chunk, { stream: !!chunk });
  48. 48 } catch {
  49. 49 throw new StreamProtocolError("Invalid SSE UTF-8");
  50. 50 }
  51. 51 }
  52. 52 // 分块可能恰好切在 CRLF 或多字节字符之间；TextDecoder 与 afterCR 共同保存跨块状态。
  53. 53 private consume(text: string) {
  54. 54 for (const character of text) {
  55. 55 if (this.afterCR) {
  56. 56 this.afterCR = false;
  57. 57 if (character === "\n") {
  58. 58 this.addBytes(1);
  59. 59 this.consumeLine();
  60. 60 continue;
  61. 61 }
  62. 62 this.consumeLine();
  63. 63 }
  64. 64 const point = character.codePointAt(0)!;
  65. 65 this.addBytes(
  66. 66 point < 0x80 ? 1 : point < 0x800 ? 2 : point < 0x10000 ? 3 : 4,
  67. 67 );
  68. 68 if (character === "\r") this.afterCR = true;
  69. 69 else if (character === "\n") this.consumeLine();
  70. 70 else this.line += character;
  71. 71 }
  72. 72 }
  73. 73 private addBytes(count: number) {
  74. 74 this.bytes += count;
  75. 75 if (this.bytes > 65536)
  76. 76 throw new StreamProtocolError("SSE frame exceeds byte limit");
  77. 77 }
  78. 78 private consumeLine() {
  79. 79 const line = this.line;
  80. 80 this.line = "";
  81. 81 if (!line) {
  82. 82 // 只有完整帧到达，才提交新游标；半个事件不能推进断点续读位置。
  83. 83 if (this.pendingId !== undefined) this.lastEventId = this.pendingId;
  84. 84 if (this.data.length)
  85. 85 this.emit({
  86. 86 type: this.type || "message",
  87. 87 data: this.data.join("\n"),
  88. 88 lastEventId: this.lastEventId,
  89. 89 });
  90. 90 this.bytes = 0;
  91. 91 this.data = [];
  92. 92 this.type = "message";
  93. 93 this.pendingId = undefined;
  94. 94 return;
  95. 95 }
  96. 96 if (line.startsWith(":")) return;
  97. 97 const colon = line.indexOf(":"),
  98. 98 field = colon < 0 ? line : line.slice(0, colon);
  99. 99 let value = colon < 0 ? "" : line.slice(colon + 1);
  100. 100 if (value.startsWith(" ")) value = value.slice(1);
  101. 101 if (field === "data") this.data.push(value);
  102. 102 else if (field === "event") this.type = value;
  103. 103 else if (field === "id" && !value.includes("\0")) {
  104. 104 if (value.length > 128)
  105. 105 throw new StreamProtocolError("SSE cursor exceeds length limit");
  106. 106 this.pendingId = value;
  107. 107 }
  108. 108 // 不采纳服务端 retry 字段，重连仍由客户端有上限的退避策略统一控制。
  109. 109 }
  110. 110}
  111. 111
  112. 112// int64 上限用 BigInt 比较，整个传输与缓存路径仍保存字符串，避免 number 精度损失。
  113. 113function validCursor(value: string) {
  114. 114 return /^[1-9]\d{0,18}$/.test(value) && BigInt(value) <= 9223372036854775807n;
  115. 115}
  116. 116
  117. 117/**
  118. 118 * 为一个任务维护鉴权事件流，游标保留为字符串，避免 int64 ID 转成 JS number 后丢精度。
  119. 119 * 使用 fetch 才能在断线后显式携带 Last-Event-ID；页面隐藏和路由退出都可立即取消连接。
  120. 120 * SSE 仅补充实时体验，任务的 REST 轮询独立存在，不把流连通状态当成任务执行状态。
  121. 121 */
  122. 122export function subscribeTaskEvents(
  123. 123 taskId: string,
  124. 124 options: {
  125. 125 onEvent: (event: StreamEvent) => void;
  126. 126 onState?: (state: StreamState) => void;
  127. 127 },
  128. 128): () => void {
  129. 129 if (!validCursor(taskId)) throw new RangeError("Invalid task ID");
  130. 130 let disposed = false,
  131. 131 blocked = false,
  132. 132 cursor = "",
  133. 133 failures = 0;
  134. 134 let active: AbortController | undefined;
  135. 135 let reconnect: ReturnType<typeof setTimeout> | undefined;
  136. 136 let timeout: ReturnType<typeof setTimeout> | undefined;
  137. 137 const clearTimeouts = () => {
  138. 138 clearTimeout(reconnect);
  139. 139 clearTimeout(timeout);
  140. 140 reconnect = timeout = undefined;
  141. 141 };
  142. 142 // 连接身份围栏：旧连接的异步回调不得覆盖替换后的连接或路由退出后的状态。
  143. 143 const current = (controller: AbortController) =>
  144. 144 !disposed && active === controller && !controller.signal.aborted;
  145. 145 const deadline = (controller: AbortController, ms: number) => {
  146. 146 clearTimeout(timeout);
  147. 147 timeout = setTimeout(
  148. 148 () =>
  149. 149 controller.abort(new DOMException("Stream timed out", "TimeoutError")),
  150. 150 ms,
  151. 151 );
  152. 152 };
  153. 153 const start = () => {
  154. 154 if (disposed || blocked || active || document.visibilityState === "hidden")
  155. 155 return;
  156. 156 clearTimeout(reconnect);
  157. 157 reconnect = undefined;
  158. 158 const controller = new AbortController();
  159. 159 active = controller;
  160. 160 options.onState?.("connecting");
  161. 161 deadline(controller, 20000);
  162. 162 void connect(controller);
  163. 163 };
  164. 164 const connect = async (controller: AbortController) => {
  165. 165 let permanent = false;
  166. 166 try {
  167. 167 const headers = new Headers({
  168. 168 Accept: "text/event-stream",
  169. 169 "X-Request-ID": newIdempotencyKey(),
  170. 170 });
  171. 171 if (cursor) headers.set("Last-Event-ID", cursor);
  172. 172 const response = await fetch(`/api/v1/tasks/${taskId}/events/stream`, {
  173. 173 credentials: "same-origin",
  174. 174 cache: "no-store",
  175. 175 headers,
  176. 176 signal: controller.signal,
  177. 177 });
  178. 178 if (!current(controller)) return;
  179. 179 if (!response.ok || response.status === 204) {
  180. 180 // 无内容或不可恢复的客户端错误停止重连；408/429 仍按暂时性故障处理。
  181. 181 permanent =
  182. 182 response.status === 204 ||
  183. 183 (response.status >= 400 &&
  184. 184 response.status < 500 &&
  185. 185 ![408, 429].includes(response.status));
  186. 186 if (response.status === 401)
  187. 187 window.dispatchEvent(new Event("drop:unauthorized"));
  188. 188 if (permanent) blocked = true;
  189. 189 await response.body?.cancel();
  190. 190 return;
  191. 191 }
  192. 192 if (
  193. 193 response.headers
  194. 194 .get("Content-Type")
  195. 195 ?.split(";")[0]
  196. 196 .trim()
  197. 197 .toLowerCase() !== "text/event-stream" ||
  198. 198 !response.body
  199. 199 )
  200. 200 throw new StreamProtocolError("Invalid SSE response");
  201. 201 options.onState?.("live");
  202. 202 deadline(controller, 45000);
  203. 203 const parser = new EventStreamDecoder((event) => {
  204. 204 if (!current(controller)) return;
  205. 205 if (!validCursor(event.lastEventId))
  206. 206 throw new StreamProtocolError("Invalid SSE cursor");
  207. 207 if (cursor && BigInt(event.lastEventId) <= BigInt(cursor)) return;
  208. 208 cursor = event.lastEventId;
  209. 209 failures = 0;
  210. 210 options.onEvent(event);
  211. 211 }, cursor);
  212. 212 const reader = response.body.getReader();
  213. 213 try {
  214. 214 while (current(controller)) {
  215. 215 const chunk = await reader.read();
  216. 216 if (!current(controller)) return;
  217. 217 if (chunk.done) {
  218. 218 parser.finish();
  219. 219 break;
  220. 220 }
  221. 221 deadline(controller, 45000);
  222. 222 parser.push(chunk.value);
  223. 223 }
  224. 224 } finally {
  225. 225 await reader.cancel().catch(() => {});
  226. 226 reader.releaseLock();
  227. 227 }
  228. 228 } catch (error) {
  229. 229 permanent = error instanceof StreamProtocolError;
  230. 230 } finally {
  231. 231 if (active !== controller) return;
  232. 232 active = undefined;
  233. 233 clearTimeout(timeout);
  234. 234 timeout = undefined;
  235. 235 if (disposed || document.visibilityState === "hidden") return;
  236. 236 blocked = permanent;
  237. 237 options.onState?.("polling");
  238. 238 if (!blocked) {
  239. 239 // 指数退避限制故障时的连接风暴；收到有效新事件后 failures 会归零。
  240. 240 const delay = Math.min(30000, 1000 * 2 ** Math.min(failures, 5));
  241. 241 failures = Math.min(failures + 1, 16);
  242. 242 reconnect = setTimeout(start, delay);
  243. 243 }
  244. 244 }
  245. 245 };
  246. 246 // 页面隐藏主动释放流与重连定时器；恢复可见时重新从已提交游标连接。
  247. 247 const visibility = () => {
  248. 248 if (document.visibilityState === "hidden") {
  249. 249 clearTimeouts();
  250. 250 const connection = active;
  251. 251 active = undefined;
  252. 252 connection?.abort(new DOMException("Page hidden", "AbortError"));
  253. 253 options.onState?.("paused");
  254. 254 } else start();
  255. 255 };
  256. 256 document.addEventListener("visibilitychange", visibility);
  257. 257 start();
  258. 258 // React effect 的清理函数负责路由退出；仅停止浏览器订阅，不代表取消后台采集任务。
  259. 259 return () => {
  260. 260 disposed = true;
  261. 261 document.removeEventListener("visibilitychange", visibility);
  262. 262 clearTimeouts();
  263. 263 const connection = active;
  264. 264 active = undefined;
  265. 265 connection?.abort(new DOMException("Navigation canceled", "AbortError"));
  266. 266 };
  267. 267}
