cmake_minimum_required(VERSION 3.22)
project(drop_core VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)

set(PROTO_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../contracts)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})
add_custom_command(OUTPUT ${GEN_DIR}/drop.pb.cc ${GEN_DIR}/drop.pb.h ${GEN_DIR}/drop.grpc.pb.cc ${GEN_DIR}/drop.grpc.pb.h
  COMMAND protobuf::protoc --proto_path=${PROTO_DIR} --cpp_out=${GEN_DIR} --grpc_out=${GEN_DIR}
    --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO_DIR}/drop.proto
  DEPENDS ${PROTO_DIR}/drop.proto)
add_library(drop_proto ${GEN_DIR}/drop.pb.cc ${GEN_DIR}/drop.grpc.pb.cc)
target_include_directories(drop_proto PUBLIC ${GEN_DIR})
target_link_libraries(drop_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(drop_common src/process.cpp src/config.cpp src/cooperative.cpp src/artifact.cpp src/manifest.cpp src/helper.cpp src/sampling.cpp src/catalog.cpp src/target.cpp src/allocator_probe.cpp src/runner.cpp src/bolt.cpp src/lifecycle.cpp src/telemetry.cpp)
target_include_directories(drop_common PUBLIC include)
target_link_libraries(drop_common PUBLIC nlohmann_json::nlohmann_json CURL::libcurl OpenSSL::Crypto Threads::Threads)
target_compile_options(drop_common PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_server src/server.cpp)
target_link_libraries(drop_server PRIVATE drop_proto drop_common PostgreSQL::PostgreSQL)
target_compile_options(drop_server PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_agent src/agent.cpp)
target_link_libraries(drop_agent PRIVATE drop_proto drop_common)
target_compile_options(drop_agent PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_collector_helper src/helper_main.cpp)
target_link_libraries(drop_collector_helper PRIVATE drop_common)
target_compile_options(drop_collector_helper PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_gdb_ptrace src/gdb_ptrace_main.cpp)
target_link_libraries(drop_gdb_ptrace PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_options(drop_gdb_ptrace PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_workload src/workload.cpp)
target_link_libraries(drop_workload PRIVATE Threads::Threads)
target_compile_options(drop_workload PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-omit-frame-pointer -g)
target_link_options(drop_workload PRIVATE -rdynamic)
add_library(drop_native_profiler SHARED src/native_profiler.cpp)
target_include_directories(drop_native_profiler PUBLIC include)
target_link_libraries(drop_native_profiler PRIVATE nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(drop_native_profiler PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(drop_native_workload fixtures/native_workload.cpp)
target_link_libraries(drop_native_workload PRIVATE drop_native_profiler Threads::Threads)
target_compile_options(drop_native_workload PRIVATE -Wall -Wextra -Wpedantic -Werror -g -fno-omit-frame-pointer)
target_link_options(drop_native_workload PRIVATE -rdynamic -Wl,--emit-relocs)

  add_executable(${test_name}_test tests/${test_name}_test.cpp)
